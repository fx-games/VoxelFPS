 
ACTOR FDAlienVendettaRocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAlienVendettaRocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was blasted by %k's hail of boomy bois."
  Decal ""
  Tag "Mini-missile Launcher"
  States
  {
  Ready:
    VROK A 1 A_WeaponReady
    Loop
	
  Select:
    VROK A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    VROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    VROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaRocket", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,4)
    TNT1 A 0 A_PlaySound("alienvendettarocketlauncher/fire", CHAN_WEAPON, 1, 0, 0.9)
    TNT1 A 0 A_PlaySound("alienvendettarocketlauncher/fire", 7, 0.05, 0, 0.1)
	TNT1 A 0 A_FireCustomMissile ("FDAlienVendettaRocketProjectile", frandom(-1.0,1.0), 1)
	
	VROK B 1 Bright A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	VROK B 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0)) // 3.25
	VROK C 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	VROK C 1 Bright A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	VROK D 1 Bright A_SetPitch(pitch + (0.6 * CallACS("FD_recoil")/65536.0))
	VROK D 1 Bright A_SetPitch(pitch + (0.39 * CallACS("FD_recoil")/65536.0))
	VROK E 1 A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	VROK E 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	VROK F 1 A_SetPitch(pitch + (0.19 * CallACS("FD_recoil")/65536.0))
	VROK F 1 A_SetPitch(pitch + (0.13 * CallACS("FD_recoil")/65536.0))
	VROK G 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VROK G 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0)) // 3.25
    VROK AAAAAAAAAA 1 A_Refire
    Goto Ready
	
  Flash:
	TNT1 A 3 A_Light2
	TNT1 A 3 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}

ACTOR FDAlienVendettaRocketProjectile : FastProjectile
{
  Radius 7
  Height 5
  SeeSound "alienvendettarocketlauncher/fire"
  Speed 50
  Damage (60)
  DamageType "Explosive"
  Projectile
  Scale 0.6
  Decal "DoomImpScorch"
  Species "Player"
  +THRUSPECIES
  +THRUGHOST
  +FORCEXYBILLBOARD
  +EXPLODEONWATER
  States
  {
  Spawn:
    VMMP A 1 Bright Light("FDAVMiniMissileGlow") A_SpawnItemEx ("FDAlienVendettaRocketTrail", -8,0,0, -5,0,0, 0, SXF_NOCHECKPOSITION)
    Loop
  Death:

    TNT1 A 0 A_Explode (21, 110, 0)
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "QuakeTime")
    TNT1 A 0 A_Jump (256, 1,3,6)
	TNT1 AAAAAAA 0 A_SpawnParticle (730000, SPF_FULLBRIGHT | SPF_RELATIVE, random(30,70), random(1,2), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,3,6)
	TNT1 AAAAAAAAA 0 A_SpawnParticle (ef0000, SPF_FULLBRIGHT | SPF_RELATIVE, random(25,60), random(1,2), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,3,6)
	TNT1 AAAAAAAAAAA 0 A_SpawnParticle (f37317, SPF_FULLBRIGHT | SPF_RELATIVE, random(20,50), random(1,2), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,3,6)
	TNT1 AAAAAAAAAAAAA 0 A_SpawnParticle (ffa35b, SPF_FULLBRIGHT | SPF_RELATIVE, random(15,40), random(1,2), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,2), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, "QuakeTime")
	Goto QuakeTime
  QuakeTime:
	TNT1 A 0 A_JumpIf(CallACS("FD_quakes") == 0, "DeathAnimation")
    TNT1 A 0 A_Quake (1, 12, 0, 768, "")
    TNT1 A 0 A_Quake (3, 8, 0, 256, "")
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:
    TNT1 A 0 A_SetScale (0.88)
    TNT1 A 0 A_SetTranslucent (0.98, 1)
    TNT1 A 0 A_PlaySound("alienvendettarocketlauncher/explosion", CHAN_WEAPON, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("alienvendettarocketlauncher/explosion", 7, 0.10, 0, 0.01)
    MEXP A 3 Bright Light("FDAVMiniMissileExplosion1")
    MEXP B 3 Bright Light("FDAVMiniMissileExplosion2")
    MEXP C 2 Bright Light("FDAVMiniMissileExplosion3")
    MEXP D 2 Bright Light("FDAVMiniMissileExplosion4")
    MEXP E 1 Bright Light("FDAVMiniMissileExplosion5")
    MEXP F 1 Bright Light("FDAVMiniMissileExplosion6")
    Stop
  }
}
