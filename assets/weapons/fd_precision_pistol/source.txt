 
ACTOR FDAlienVendettaPistol : FDWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAlienVendettaPistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%k shredded %o with a hail of stored gunfire."
  Decal "BulletChip"
  Tag "Precision Pistol"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReadyUseAmmo")
    VPIS A 1 A_WeaponReady
    Loop
  ReadyFull:
    VPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Loop
	
  ReadyUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto ReadyFull
    VPIS A 1 A_WeaponReady
    Goto Ready
	
  Select:
    VPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    VPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    VPIS A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready

// aaaagh
// next person who asks for yet another pistol cvar gets fucking shanked

  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto FireNoAmmoCheckDone
  FireNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire")
    Goto FireNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, "FireNoAmmoCheckDone")
    Goto Dryfire
	
  FireNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (1, 1, 1, randompick(9,11,13), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (1, 1, 1, randompick(5,6,7), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireComplete
	
  FireNoAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,5)
    TNT1 A 0 A_PlaySound("alienvendettapistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienvendettapistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoMagConsumption") // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (1, 1, -1, randompick(9,11,13), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FireComplete:
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD9mmCasingSpawner", 0,0,frandom(0.5,0.6),-1, FPF_NOAUTOAIM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, 1) // If you've got no reloads
	Goto FireEnd
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireEndAmmo") // and chaingun ammo, then we need to do more checks
	Goto FireEnd
  FireEnd:
    VPIS B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire1
    VPIS B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire2
    VPIS C 1 Bright A_SetPitch (pitch + (0.08 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire3
    VPIS C 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire4
    VPIS D 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire5
    VPIS D 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire6
    VPIS E 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire7
    VPIS E 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire8
    VPIS F 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire9
    VPIS F 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire10
    VPIS A 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire11
    VPIS A 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefire12
    TNT1 A 0
	Goto Ready
	
  FireEndRefire1:
    VPIS B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire ("FireReset1")
  FireEndRefire2:
    VPIS C 1 Bright A_SetPitch (pitch + (0.08 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire ("FireReset2")
  FireEndRefire3:
    VPIS C 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire ("FireReset3")
  FireEndRefire4:
    VPIS D 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire5:
    VPIS D 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire6:
    VPIS E 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire7:
    VPIS E 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire8:
    VPIS F 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire9:
    VPIS F 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire10:
    VPIS A 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire11:
    VPIS A 1
    TNT1 A 0 A_Refire ("Fire")
  FireEndRefire12:
    VPIS A 1
	Goto Ready
	
	
  FireReset1:
	TNT1 A 0 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
    Goto Fire
  FireReset2:
	TNT1 A 0 A_SetPitch (pitch + (0.42 * CallACS("FD_recoil")/65536.0))
    Goto Fire
  FireReset3:
	TNT1 A 0 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    Goto Fire

  AltFire:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 0, "Reloadery")
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, "Reloadery")
	Goto DryFire // Normally we'd return to Ready, but we disable altfire if full anyway, so we can go to dryfire without problems
  Reloadery:
    VPIS FGH 2
    TNT1 A 0 A_PlaySound("alienvendettapistol/unload", 6, 1, 0, 1.25)
    VPIS IJKLMN 2
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReloadUseAmmo")
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaPistolAmmo", 16)
	Goto ReloadFinish
  ReloadUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaPistolAmmo", 0, "ReloadFinish")
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto ReloadFinish
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaPistolAmmo", 1)
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaBullets", 1)
	Goto ReloadUseAmmo
  ReloadFinish:
    TNT1 A 0 A_PlaySound("alienvendettapistol/load", 5, 1, 0, 1.25)
    VPIS OPQRS 2
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
	
// Extra repeated bullshit for fucking non-reloading bullet-using mode aaagh
  FireEndAmmo:
    VPIS B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo1
    VPIS B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo2
    VPIS C 1 Bright A_SetPitch (pitch + (0.08 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo3
    VPIS C 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo4
    VPIS D 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo5
    VPIS D 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo6
    VPIS E 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo7
    VPIS E 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo8
    VPIS F 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo9
    VPIS F 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo10
    VPIS A 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo11
    VPIS A 1
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, 1)
	Goto FireEndRefireAmmo12
    TNT1 A 0
	Goto Ready
	
  FireEndRefireAmmo1:
    VPIS B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo2
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "FireReset1")
  FireEndRefireAmmo2:
    VPIS C 1 Bright A_SetPitch (pitch + (0.08 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo3
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "FireReset2")
  FireEndRefireAmmo3:

    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo4
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "FireReset1")
  FireEndRefireAmmo4:
    VPIS D 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo5
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo5:
    VPIS D 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo6
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo6:
    VPIS E 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo7
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo7:
    VPIS E 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo8
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo8:
    VPIS F 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo9
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo9:
    VPIS F 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)

    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo10:
    VPIS A 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo11
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")
  FireEndRefireAmmo11:
    VPIS A 1
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBullets", 1, 1)
	Goto FireEndRefireAmmo12
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "Fire")

  }
