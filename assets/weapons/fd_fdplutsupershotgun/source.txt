 
ACTOR FDPlutSuperShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDPlutPlayer"
  Inventory.Icon "PLSUPERS"
  Weapon.UpSound ""
  Weapon.AmmoType "FDPlutShells"
  Weapon.AmmoUse 2
  Weapon.SlotNumber 3
  Obituary "%o got splattered along the barrels of %k's super shotgun."
  Decal "BulletChip"
  Tag "Super Shotgun"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDPlutShells", 2, "ReadyAmmo")
    PSSG A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    PSSG A 1 A_WeaponReady
    Loop
	
  Select:
    PSSG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    PSSG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    PSSG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDPlutShells", 2, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,1,6)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-1,6)
    TNT1 A 0 A_PlaySound("plutsupershotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("plutsupershotgun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (13, 10, 24, 5, "FDBulletPuff", FBF_USEAMMO)
    PSSG B 1 Bright A_SetPitch (pitch - (1.6 * CallACS("FD_recoil")/65536.0))
    PSSG B 1 Bright A_SetPitch (pitch - (1.1 * CallACS("FD_recoil")/65536.0))
    PSSG C 1 Bright A_SetPitch (pitch - (0.5 * CallACS("FD_recoil")/65536.0))
    PSSG C 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    PSSG D 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    PSSG D 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    PSSG E 1 A_SetPitch (pitch + (0.75 * CallACS("FD_recoil")/65536.0))
    PSSG E 1 A_SetPitch (pitch + (1.5 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDPlutShells", 2, "GonnaReload")
    PSSG A 1 A_SetPitch (pitch + (1.0 * CallACS("FD_recoil")/65536.0))
    PSSG A 1 A_SetPitch (pitch + (0.20 * CallACS("FD_recoil")/65536.0))
    Goto Ready
  GonnaReload:
    PSSG F 1 A_SetPitch (pitch + (1.0 * CallACS("FD_recoil")/65536.0))
    PSSG F 1 A_SetPitch (pitch + (0.20 * CallACS("FD_recoil")/65536.0))
    PSSG GHI 2
    PSSG J 4
    PSSG KLM 2
    TNT1 A 0 A_PlaySound("plutsupershotgun/open", 5, 1, 0, 1.25)
	TNT1 A 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 AA 0 A_FireCustomMissile ("FDShellCasingSpawner2", 0,0,frandom(-3.6,-3.2),-9, FPF_NOAUTOAIM)
	PSSG N 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	PSSG N 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
    PSSG OP 2
    TNT1 A 0 A_PlaySound("plutsupershotgun/load", 6, 1, 0, 1.25)
    PSSG QR 2 //40
    TNT1 A 0 A_PlaySound("plutsupershotgun/close", 5, 1, 0, 1.25)
	TNT1 A 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	PSSG S 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	PSSG S 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire ("QuickFire")
    PSSG T 3
    PSSG U 4
    PSSG V 3
    PSSG A 2 // Gun fires at a speed of 58 tics
    Goto Ready
  QuickFire:
    PSSG TUVAA 1 // Gun refires at a speed of 51 tics
    Goto Fire
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}
