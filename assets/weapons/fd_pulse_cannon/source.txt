ACTOR FDBTSXRocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDBTSXPlayer"
  Inventory.Icon "BXROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXRocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was repulsed by %k."
  Decal ""
  Tag "Pulse Cannon"
  States
  {
  Ready:
    XROK A 1 A_WeaponReady
    Loop
	
  Select:
    XROK A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    XROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    XROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXRocket", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_PlaySound("btsxrocketlauncher/charge", 6, 1, 0, 1.25)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "OverdriveCharge")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "OverdriveCharge")
	XROK BCDEFGH 2
  ChargeDone:
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_AlertMonsters
//    TNT1 A 0 A_PlaySound("btsxrocketlauncher/fire", CHAN_WEAPON, 1, 0, 0.7)
//    TNT1 A 0 A_PlaySound("btsxrocketlauncher/fire", 7, 1, 0, 0.4)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 2)
	TNT1 A 0 A_FireCustomMissile ("FDBTSXRocketProjectile", 0, 1)
	Goto FireFinish
	TNT1 A 0 A_FireCustomMissile ("FDBTSXOverdriveRocketProjectile", 0, 1)
	Goto FireFinish
  FireFinish:
	// -2.25
	XROK I 1 Bright A_SetPitch(pitch - (1.75 * CallACS("FD_recoil")/65536.0))
	XROK I 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	// 2.25
	XROK J 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	XROK J 1 Bright A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	XROK K 1 Bright A_SetPitch(pitch + (0.6 * CallACS("FD_recoil")/65536.0))
	XROK K 1 Bright A_SetPitch(pitch + (0.39 * CallACS("FD_recoil")/65536.0))
	XROK L 1 Bright A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	XROK L 1 Bright A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	XROK M 1 Bright A_SetPitch(pitch + (0.19 * CallACS("FD_recoil")/65536.0))
	XROK M 1 Bright A_SetPitch(pitch + (0.13 * CallACS("FD_recoil")/65536.0))
	XROK C 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	XROK C 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
	XROK A 2
    XROK AAAAAAAAAA 1 A_Refire
    Goto Ready
  OverdriveCharge:
	XROK BCDEFGH 1
	Goto ChargeDone
	
  Flash:
	TNT1 A 2 A_Light (7)
	TNT1 A 2 A_Light (4)
	TNT1 A 2 A_Light (3)
	TNT1 A 2 A_Light2
	TNT1 A 2 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
	
  FancyFire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 2)
	TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyRocketProjectile", 0, 1)
	Goto FireFinish
	TNT1 A 0 A_FireCustomMissile ("FDBTSXOverdriveFancyRocketProjectile", 0, 1)
	Goto FireFinish
  }
}
