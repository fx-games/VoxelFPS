 
ACTOR FDDoom2BFG9000 : FDBFGWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2BFG900"
  Weapon.UpSound ""
  Weapon.AmmoType "FDDoom2BFGCharge"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 7
  Obituary "%o was disintegrated by %k."
  Decal ""
  Tag "BFG 9000"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("doom2bfg9000/idle", 7, 1, 1, 1.4)
    DBFG A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    DBFG A 1 A_WeaponReady
    Loop
	
  Select:
    DBFG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    "####" "#" 0 A_StopSound(7)
  DeselectLoop:
    DBFG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    DBFG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_JumpIfInventory ("FDDoom2BFGCharge", 1, "ChargeStart")
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Cell", 40, "ChargeStart")
	Goto Dryfire
	
  ChargeStart:
    TNT1 A 0 A_PlaySound("doom2bfg9000/charge", 7, 1, 0, 0.8)
    TNT1 A 0 A_AlertMonsters (random(256,400))
    TNT1 A 0 A_GunFlash ("FlashCharge", GFF_NOEXTCHANGE)
	DBFG B 9
  ChargeRefire:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_JumpIfInventory ("FDDoom2BFGCharge", 1, 2)
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Cell", 40, 1)
	Goto Dryfire
    TNT1 A 0 A_GunFlash ("FlashCharge2", GFF_NOEXTCHANGE)
	DBFG B 18
    TNT1 A 0 A_GunFlash ("Flash")
	DBFG C 3 Bright
	TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("doom2bfg9000/fire", CHAN_WEAPON, 1, 0, 0.8)
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_TakeInventory ("FDDoom2BFGCharge", 1, TIF_NOTAKEINFINITE)
	Goto PLASMADIETIME
    TNT1 A 0 A_TakeInventory ("FDDoom2Cell", 40, TIF_NOTAKEINFINITE)
	Goto PLASMADIETIME
  PLASMADIETIME:
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla")
	TNT1 A 0 A_FireCustomMissile ("FDDoom2BFGBall", 0, 0)
  FiringComplete:
	DBFG D 1 Bright A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	DBFG D 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	DBFG D 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	DBFG D 1 Bright A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	DBFG E 1 Bright A_SetPitch(pitch + (0.7 * CallACS("FD_recoil")/65536.0))
	DBFG E 1 Bright A_SetPitch(pitch + (0.43 * CallACS("FD_recoil")/65536.0))
	DBFG E 1 Bright A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	DBFG F 1 Bright A_SetPitch(pitch + (0.29 * CallACS("FD_recoil")/65536.0))
	DBFG F 1 Bright A_SetPitch(pitch + (0.28 * CallACS("FD_recoil")/65536.0))
	DBFG F 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_ReFire ("RefireG")
	DBFG G 3
	TNT1 A 0 A_ReFire ("RefireH")
	DBFG H 3
	TNT1 A 0 A_ReFire ("RefireI")
	DBFG I 3
	DBFG A 11 A_ReFire
    Goto Ready
  RefireG:
    TNT1 A 0 A_PlaySound("doom2bfg9000/charge", 7, 1, 0, 0.8)
	DBFG GHI 3
	Goto ChargeRefire
  RefireH:
    TNT1 A 0 A_PlaySound("doom2bfg9000/charge", 7, 1, 0, 0.8)
	DBFG HIB 3
	Goto ChargeRefire
  RefireI:
    TNT1 A 0 A_PlaySound("doom2bfg9000/charge", 7, 1, 0, 0.8)
	DBFG IBB 3
	Goto ChargeRefire
  FiringVanilla:
	TNT1 A 0 A_FireCustomMissile ("FDDoom2BFGBallVanilla", 0, 0)
	Goto FiringComplete
	
  Flash:
	TNT1 A 3 A_Light2
	TNT1 A 4 A_Light (5)
	TNT1 A 3 A_Light2
	TNT1 A 3 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  FlashCharge:
	DBFG JJKKKJJKKKJJKKKJJKKKJJKKKJJKKK 1 Bright A_Light0
    TNT1 A 0 A_Light0
    Goto LightDone
  FlashCharge2:
	DBFG JJJKKKJJKKKJJKKKJJKKK 1 Bright A_Light0
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}

ACTOR FDDoom2BFGBall
{
  Radius 13
  Height 8
  Speed 25
  Damage (800)
  DamageType "BFG"
  Projectile
  +THRUSPECIES
  +FORCEXYBILLBOARD
  +EXPLODEONWATER
  +SKYEXPLODE
  RenderStyle Add
  Alpha 0.98
  SeeSound "doom2bfg9000/fire"
  Decal "BFGLightning"
  Species "Player"

	var int user_spriteframe;
  States
  {
  Spawn:
    TNT1 A 0 NoDelay A_PlaySound ("doom2bfg9000/loop", 7, 0.5, 1, 1.4)
	TNT1 A 0 A_SetUserVar("user_spriteframe", 0)
  SpawnLoop:
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "SpawnLoopFrame")
    TNT1 A 0 A_Jump (256, 1,2,3,4)
	TNT1 AAA 0 A_SpawnParticle (ffff6b, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,8), random(1,3), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.01, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3,4,5)
	TNT1 AAAA 0 A_SpawnParticle (439337, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,8), random(1,3), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.01, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3,4,5)
	TNT1 AAAA 0 A_SpawnParticle (5fcf57, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,8), random(1,3), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.01, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3,4)
	TNT1 AAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,8), random(1,3), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.01, 0.98, -1)
  SpawnLoopFrame:
	TNT1 A 0 A_JumpIf (user_spriteframe >= 6, "SpawnFrameB")
    Goto SpawnFrameA
  SpawnFrameA:
    D2BB A 1 Bright Light("FDBFGBall") A_SpawnItemEx ("FDDoom2BFGBallTrail", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME)
	Goto SpawnLoopEnd
  SpawnFrameB:
    D2BB B 1 Bright Light("FDBFGBall") A_SpawnItemEx ("FDDoom2BFGBallTrail", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME)
	Goto SpawnLoopEnd
  SpawnLoopEnd:
	TNT1 A 0 A_SetUserVar("user_spriteframe", user_spriteframe+1)
	TNT1 A 0 A_JumpIf (user_spriteframe >= 12, "Spawn")
	Goto SpawnLoop
	
  Death:
    TNT1 A 0
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "QuakeTime")
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAAAAAAAAA 0 A_SpawnParticle (ffff6b, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,70), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (439337, SPF_FULLBRIGHT | SPF_RELATIVE, random(35,105), random(1,20), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (5fcf57, SPF_FULLBRIGHT | SPF_RELATIVE, random(35,105), random(1,20), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,70), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
	Goto QuakeTime
  QuakeTime:
	TNT1 A 0 A_JumpIf(CallACS("FD_quakes") == 0, "DeathAnimation")
    TNT1 A 0 A_Quake (1, 40, 0, 1536, "")
    TNT1 A 0 A_Quake (7, 24, 0, 768, "")
	Goto DeathAnimation
  DeathAnimation:
    TNT1 A 0 A_PlaySound ("doom2bfg9000/explosion", 7, 1, 0, 0.4)
    D2BB C 8 Bright Light("FDGreenExplosion1")
    D2BB D 8 Bright Light("FDGreenExplosion2")
    TNT1 A 0 A_BFGSpray ("FDTNTBFGRay")
    D2BB E 8 Bright Light("FDGreenExplosion3")
    D2BB F 8 Bright Light("FDGreenExplosion4")
    D2BB G 8 Bright Light("FDGreenExplosion5")
    D2BB H 8 Bright Light("FDGreenExplosion6")
    Stop
  }
}
