 
ACTOR FDHellboundShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDHellboundPlayer"
  Inventory.Icon "HBSHOTGU"
  Weapon.UpSound ""
  Weapon.AmmoType "FDHellboundShells"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 3
  Obituary "%o blew a new cakehole for %o."
  Decal "BulletChip"
  Tag "Borstal Shotgun"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDHellboundShells", 1, "ReadyAmmo")
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShotgunShots", 1, "ReadyAmmo")
    HSHT A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    HSHT A 1 A_WeaponReady
    Goto Ready
	
  Select:
    HSHT A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
	TNT1 A 0 A_TakeInventory("FDHellboundShotgunOverloaded", 1)
    HSHT A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    HSHT A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShotgunShots", 0, "GonnaReloadNothingery")
	TNT1 A 0 A_TakeInventory("FDHellboundShotgunOverloaded", 1)
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShells", 1, 2)
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShotgunShots", 1, "GonnaReload")
	Goto Dryfire
	TNT1 A 0 A_GiveInventory("FDHellboundShotgunShots",1)
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
	TNT1 A 0 A_PlaySound("hellboundshotgun/fire", CHAN_WEAPON, 0.8, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundshotgun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_FireBullets (4, 0.5, 10, 5, "FDBulletPuff", 0)
	TNT1 A 0 A_TakeInventory("FDHellboundShells", 1, TIF_NOTAKEINFINITE)
	HSHT N 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
	HSHT N 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
	HSHT O 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
	HSHT O 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	HSHT P 1 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
	HSHT Q 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	HSHT R 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	HSHT S 1 A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
	HSHT T 1 A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	HSHT UV 1
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShotgunShots", 0, "GonnaReloadNothingery")
	TNT1 A 0 A_ReFire
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShells", 1, 1)
	Goto GonnaReload
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShotgunShots", 1, "GonnaReloadAmmoery")
	Goto Ready
  GonnaReload:
	HSHT ABCDE 1 A_WeaponReady
	Goto KeepReloading
  GonnaReloadNothingery:
	TNT1 A 0 A_GiveInventory("FDHellboundShotgunOverloaded", 1)
	HSHT ABCDE 1
	Goto KeepReloading
  GonnaReloadAmmoery:
	HSHT A 1 A_Refire
	HSHT B 1 A_Refire ("RefireB")
	HSHT C 1 A_Refire ("RefireC")
	HSHT D 1 A_Refire ("RefireD")
	HSHT E 1 A_Refire ("RefireE")
	Goto KeepReloading
	
  KeepReloading:
	TNT1 A 0 A_PlaySound("hellboundshotgun/pump", 5, 0.9, 0, 1.25)
	HSHT F 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	HSHT G 1 A_SetPitch(pitch + (0.5 * CallACS("FD_recoil")/65536.0))
	HSHT H 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	HSHT IJKLM 1
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDHellboundShellCasingSpawner", 0,0,frandom(-7.6,-7.2),-12, FPF_NOAUTOAIM)
	TNT1 A 0 A_TakeInventory("FDHellboundShotgunShots", 1)
	HSHT L 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	HSHT K 1 A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	HSHT J 1 A_SetPitch(pitch - (0.3 * CallACS("FD_recoil")/65536.0))
	HSHT IHGF 1
	HSHT A 0 A_JumpIfInventory ("FDHellboundShotgunOverloaded", 1, 2)
	HSHT A 0 A_ReFire("ReloadEnd")
	HSHT A 0 A_JumpIfInventory ("FDHellboundShotgunShots", 1, "KeepReloading")
	Goto ReloadEnd
  ReloadEnd:
	TNT1 A 0 A_TakeInventory("FDHellboundShotgunOverloaded", 1)
	HSHT EDCB 1
	HSHT A 0 A_Refire // Gun fires at a speed of 37 tics
	Goto Ready
	
  RefireE:
	HSHT D 1
  RefireD:
	HSHT C 1
  RefireC:
	HSHT B 1
  RefireB:
	HSHT A 1
	Goto Fire
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

Actor FDHellboundShotgunShots : FDCounter
{
inventory.maxamount 3
}

Actor FDHellboundShotgunOverloaded : FDLevelLimit {}
