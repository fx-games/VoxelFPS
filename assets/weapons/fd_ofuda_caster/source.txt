 
ACTOR FDJPCPPlasmaRifle : FDWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPPLASMA"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPCell"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 6
  Obituary "%o was sealed away by %k."
  Decal ""
  Tag "Ofuda Caster"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, "ReadyAmmo")
  ReadyNoAmmo:
    JPLA A 1 A_WeaponReady
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, 1) // If you run across ammo while empty, the gun will autoload a new amulet
    Goto ReadyNoAmmo
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    TNT1 A 0 A_PlaySound("jpcpplasmarifle/idle", 7, 0.3, 1, 1.4)
	JPLA BC 1
    Goto Ready
  ReadyAmmo:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    TNT1 A 0 A_PlaySound("jpcpplasmarifle/idle", 7, 0.3, 1, 1.4)
    JPLA C 1 A_WeaponReady
    Wait
	
  Select:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, 3)
    JPLA A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
    JPLA C 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_StopSound(7)
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, 3)
    JPLA A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
    JPLA C 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JPLA A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound ("jpcpplasmarifle/fire", CHAN_WEAPON, 1, 0, 0.8)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire")
	TNT1 A 0 A_FireCustomMissile ("FDJPCPOfuda")
  FireDone:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, 2)
    TNT1 A 0 A_StopSound(7) // Doesn't hum if there's no amulets loaded in it
	JPLA D 1 Bright A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Jump (256, "FireAnimation1", "FireAnimation2")
  FireAnimation1:
	JPLA E 1 Bright A_SetPitch(pitch + (0.06 * CallACS("FD_recoil")/65536.0))
	JPLA F 1 Bright A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	JPLA G 1 Bright A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
	Goto FireEnd
  FireAnimation2:
	JPLA H 1 Bright A_SetPitch(pitch + (0.06 * CallACS("FD_recoil")/65536.0))
	JPLA I 1 Bright A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	JPLA J 1 Bright A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
	Goto FireEnd
  FireEnd:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 1, 1)
	Goto Ready
	JPLA ABC 1
	JPLA CCCCCCCCCC 1 A_Refire
	Goto Ready
  FancyFire:
	TNT1 A 0 A_FireCustomMissile ("FDJPCPFancyOfuda")
	Goto FireDone
	
  Flash:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

actor FDJPCPSealEffectCheck : FDCustomFunction
{
   States
   {
   Pickup:
      TNT1 A 0 A_CheckFlag("BOSS", "NoEffect")
      TNT1 A 0 A_JumpIf(((health*1.0) / (CallACS("FDGetMaxHealth")*1.0)) > 0.3, "NoEffect")

      Stop
   NoEffect:
//      TNT1 A 0 A_Log ("2 stronk")
      TNT1 A 0
      Stop
   }
}
