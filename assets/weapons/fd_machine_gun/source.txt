 
ACTOR FDPlutChaingun : FDWeapon
{
  Inventory.RestrictedTo "FDPlutPlayer"
  Inventory.Icon "PLCHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDPlutBullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%o was hosed by %k."
  Decal "BulletChip"
  Tag "Machine Gun"
  States
  {
  Ready:
    PCHG A 1 A_WeaponReady
    Loop
  ReadyNoBob:
    PCHG AAAAAAAAAA 1 A_WeaponReady (WRF_NOBOB)
    Goto Ready
	
  Select:
    PCHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    PCHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    PCHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDPlutBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("plutchaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("plutchaingun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (4.2, 1.2, 1, randompick(9,12), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    PCHG B 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(4.6,4.2),-4, FPF_NOAUTOAIM)
    PCHG C 1 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
    PCHG C 1 A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("plutchaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("plutchaingun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (4.2, 1.2, 1, randompick(9,12), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    PCHG D 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(4.6,4.2),-4, FPF_NOAUTOAIM)
    PCHG E 1 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
    PCHG E 1 A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}
