// Fires extra projectiles to the sides

ACTOR FDBTSXShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDBTSXPlayer"
  Inventory.Icon "BXSHOTGU"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXShells"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 3
  Obituary "%o found they couldn't hide from %k's blasts."
  Decal ""
  Tag "Arc Blaster"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXShells", 1, "ReadyAmmo")
    XSHT A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    XSHT A 1 A_WeaponReady
    Loop
	
  Select:
    XSHT A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    XSHT A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    XSHT A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXShells", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("btsxshotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("btsxshotgun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXShotgunProjectile", 0, 1, 0, 6)
    TNT1 A 0 A_FireCustomMissile ("FDBTSXShotgunProjectile", 0, 0, 2, 6)
    TNT1 A 0 A_FireCustomMissile ("FDBTSXShotgunProjectile", 0, 0, -2, 6)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 1)
	Goto FireFinish
    TNT1 A 0 A_FireCustomMissile ("FDBTSXShotgunProjectile", 15, 0, 1, 6)
    TNT1 A 0 A_FireCustomMissile ("FDBTSXShotgunProjectile", -15, 0, -1, 6)
	Goto FireFinish
  FireFinish:
    XSHT B 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    XSHT B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    XSHT C 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    XSHT C 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    XSHT D 1 Bright A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    XSHT D 1 Bright A_SetPitch (pitch + (1.05 * CallACS("FD_recoil")/65536.0))
    XSHT A 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
  GonnaReload:
    XSHT E 1
    XSHT F 1
	XSHT G 2
	XSHT H 3
    TNT1 A 0 A_PlaySound("btsxshotgun/pumpdown", 5, 1, 0, 1.25)
	XSHT I 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	XSHT J 1 A_SetPitch(pitch + (0.23 * CallACS("FD_recoil")/65536.0))
	XSHT J 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	XSHT J 1 A_SetPitch(pitch - (0.06 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_PlaySound("btsxshotgun/pumpup", 6, 1, 0, 1.25)
	XSHT K 1 A_SetPitch(pitch - (0.28 * CallACS("FD_recoil")/65536.0))
	XSHT K 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	XSHT LMHGFE 2
	XSHT A 5
	XSHT AAAAAAA 1 A_Refire // Gun fires at a speed of 37 tics
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 2 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
	
  FancyFire:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyShotgunProjectile", 0, 1)
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyShotgunProjectile", 0, 0, 2)
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyShotgunProjectile", 0, 0, -2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 1)
	Goto FireFinish
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyShotgunProjectile", 15, 0, 1, 6)
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyShotgunProjectile", -15, 0, -1, 6)
	Goto FireFinish
  }
}
