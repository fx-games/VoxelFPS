 
ACTOR FDAliensChaingun : FDWeapon
{
  Inventory.RestrictedTo "FDAliensPlayer"
  Inventory.Icon "AACHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAliensBullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%o was rattled apart by %k."
  Decal "BulletChip"
  Tag "Ancient AK"
  States
  {
  Ready:
    ACHG A 1 A_WeaponReady
    Loop
  ReadyNoBob:
    ACHG AAAAAAAAAA 1 A_WeaponReady (WRF_NOBOB)
    Goto Ready
	
  Select:
    ACHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    ACHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    TNT1 A 0 A_ClearRefire
    ACHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDAliensAKJammed", 1, "Hold")
//    TNT1 A 0 A_Jump (15, "TING")
  Hold:
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,4)
    TNT1 A 0 A_PlaySound("alienschaingun/fire", CHAN_WEAPON, 0.7, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienschaingun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_JumpIfInventory ("FDAliensAKJammed", 1, "TINGShot")
    TNT1 A 0 A_FireBullets (4.6, 3.0, 1, randompick(12,13), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  Fire1Complete:
    ACHG B 1 Bright A_SetPitch(pitch - (0.45 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(4.6,4.2),-7, FPF_NOAUTOAIM)
    ACHG B 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    ACHG C 1 A_SetPitch(pitch + (0.35 * CallACS("FD_recoil")/65536.0))
    ACHG A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	Goto Fire2
	
  TING: // No longer used, used to have a chance of jamming, but then firing a super stronk shot when cleared
    TNT1 A 0 A_PlaySound("alienschaingun/tink", 6, 1, 0, 1.5)
    TNT1 A 0 A_GiveInventory ("FDAliensAKJammed", 1)
  TINGLoop:
    ACHG A 1
    TNT1 A 0 A_Refire ("TINGLoop")
	Goto Ready
  TINGShot:
    TNT1 A 0 A_TakeInventory ("FDAliensAKJammed", 1)
    TNT1 A 0 A_FireBullets (0, 0, 1, (60), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	Goto Fire1Complete
	
  Fire2:
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,4)
    TNT1 A 0 A_PlaySound("alienschaingun/fire", CHAN_WEAPON, 0.7, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienschaingun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (4.6, 3.0, 1, randompick(12,13), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    ACHG D 1 Bright A_SetPitch(pitch - (0.45 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(4.6,4.2),-4, FPF_NOAUTOAIM)
    ACHG D 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    ACHG E 1 A_SetPitch(pitch + (0.35 * CallACS("FD_recoil")/65536.0))
    ACHG A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_Refire ("Hold")
    TNT1 A 0 A_ClearRefire
    Goto ReadyNoBob
	
  Flash:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}

Actor FDAliensAKJammed : FDBoolean {}
