 
ACTOR FDDoom2Chaingun : FDWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2CHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDDoom2Bullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%o was chained to death by %k."
  Decal "BulletChip"
  Tag "Chaingun"
  States
  {
  Ready:
    DCHG A 1 A_WeaponReady
    Loop
  ReadyNoBob:
    DCHG AAAAAAAAAA 1 A_WeaponReady (WRF_NOBOB)
    Goto Ready
	
  Select:
    DCHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    DCHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    DCHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Bullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("doom2chaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("doom2chaingun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla")
    TNT1 A 0 A_FireBullets (5.6, 1.2, 1, randompick(18,19), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FiringComplete:
    DCHG A 1 A_SetPitch(pitch - (0.45 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(4.6,4.2),-7, FPF_NOAUTOAIM)
    DCHG B 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    DCHG C 1 A_SetPitch(pitch + (0.35 * CallACS("FD_recoil")/65536.0))
    DCHG D 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash ("Flash2")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("doom2chaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("doom2chaingun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla2")
    TNT1 A 0 A_FireBullets (5.6, 1.2, 1, randompick(18,19), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FiringComplete2:
    DCHG A 1 A_SetPitch(pitch - (0.45 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(4.6,4.2),-4, FPF_NOAUTOAIM)
    DCHG B 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    DCHG C 1 A_SetPitch(pitch + (0.35 * CallACS("FD_recoil")/65536.0))
    DCHG D 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire
    Goto ReadyNoBob
  FiringVanilla:
    TNT1 A 0 A_FireBullets (5.6, 0, 1, 5*random(1,3), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	Goto FiringComplete
  FiringVanilla2:
    TNT1 A 0 A_FireBullets (5.6, 0, 1, 5*random(1,3), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	Goto FiringComplete2
	
  Flash:
    DCHG E 1 Bright A_Light2
    DCHG F 1 Bright A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  Flash2:
    DCHG G 1 Bright A_Light2
    DCHG H 1 Bright A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}
