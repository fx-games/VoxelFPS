 
ACTOR FDWhitemarePlasmaRifle : FDWeapon
{
  Inventory.RestrictedTo "FDWhitemarePlayer"
  Inventory.Icon "WMPLASMA"
  Weapon.UpSound ""
  Weapon.AmmoType "FDWhitemareCell"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 6
  Obituary "%o was covered in a stream of flames by %k."
  Decal "Scorch"
  Tag "Flamethrower"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReady")
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/idle", 7, 1, 1, 1.4)
    WPLA A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReadyNoSound")
    WPLA A 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatSelect")
    WPLA A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatDeselect")
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)
    TNT1 A 0 A_StopSound (6)
    TNT1 A 0 A_StopSound (7)
  DeselectLoop:
    WPLA A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  DryfireFiring:
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireend", CHAN_WEAPON, 1, 0, 1.2)
    TNT1 A 0 A_StopSound (CHAN_WEAPON)
    TNT1 A 0 A_StopSound (5)
    TNT1 A 0 A_StopSound (6)
    TNT1 A 0 A_StopSound (7)
  Dryfire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatDryfire")
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/dryfire", 5, 1, 0, 1.25)
  DryfireLoop:
    WPLA A 6
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "DryfireLoop")
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatFire")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerEfficiency", 1, "FireAmmoCheckDone")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareCell", 1, "FireAmmoConvert")
	Goto Dryfire
  FireAmmoCheckDone:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatFireAmmoCheckDone")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fire", 7, 1, 0, 1.2)
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fire2", 5, 1, 0, 1.2)
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireloop", CHAN_WEAPON, 1, 1, 1.1)
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireloop2", 6, 1, 1, 0.7)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlame", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameFancy", 0, 0)
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerEfficiency", 1)
	WPLA B 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
	Goto FireHeatLevelDone
  FireHeatLevelDone:
	TNT1 A 0 A_Refire ("FireLoop")
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireend", CHAN_WEAPON, 1, 0, 1.2)
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)
    TNT1 A 0 A_StopSound (6)
    TNT1 A 0 A_StopSound (7)
	WPLA FG 2 Bright
	WPLA A 2
	Goto Ready
	
  FireLoop:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatFireLoop")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerEfficiency", 1, "FireLoopAmmoCheckDone")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareCell", 1, "FireLoopAmmoConvert")
	Goto DryfireFiring
  FireLoopAmmoCheckDone:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatFireLoopAmmoCheckDone")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerEfficiency", 1)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerFrame", 2, "FireLoopFrameE")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerFrame", 1, "FireLoopFrameD")
	Goto FireLoopFrameC
  FireLoopFrameC:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlame", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameFancy", 0, 0)
	WPLA C 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerFrame", 1)
	Goto FireLoopAnimationDone
  FireLoopFrameD:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlow", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlowFancy", 0, 0)
	WPLA D 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerFrame", 1)
	Goto FireLoopAnimationDone
  FireLoopFrameE:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlow", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlowFancy", 0, 0)

    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)
	Goto FireLoopAnimationDone
  FireLoopAnimationDone:
  FireLoopHeatLevelDone:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerEfficiency", 1, "FireLoop")
	TNT1 A 0 A_Refire ("FireLoop")
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireend", CHAN_WEAPON, 1, 0, 1.2)
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)
    TNT1 A 0 A_StopSound (6)
    TNT1 A 0 A_StopSound (7)
	WPLA FG 2 Bright
	WPLA A 2
	Goto Ready
	
  FireAmmoConvert: // Convert one flamethrower ammo into 8 flames
    TNT1 A 0 A_TakeInventory ("FDWhitemareCell", 1, TIF_NOTAKEINFINITE)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerEfficiency", 4)
	Goto FireAmmoCheckDone
  FireLoopAmmoConvert:
    TNT1 A 0 A_TakeInventory ("FDWhitemareCell", 1, TIF_NOTAKEINFINITE)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerEfficiency", 4)
	Goto FireLoopAmmoCheckDone
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light (3)
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone

// ============= THERMOCOAT ACTIVE =============================================

  ThermoCoatReady:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ThermoCoatReadyNoSound")
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/idle", 7, 1, 1, 1.4)
    WPLW A 1 A_WeaponReady
    Loop
  ThermoCoatReadyNoSound:
    WPLW A 1 A_WeaponReady
    Loop
	
  ThermoCoatSelect:
    WPLW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  ThermoCoatDeselect:
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)
    TNT1 A 0 A_StopSound (6)
    TNT1 A 0 A_StopSound (7)

    WPLW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  ThermoCoatDryfireFiring:

    TNT1 A 0 A_StopSound (CHAN_WEAPON)
    TNT1 A 0 A_StopSound (5)
    TNT1 A 0 A_StopSound (6)
    TNT1 A 0 A_StopSound (7)
  ThermoCoatDryfire:
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/dryfire", 5, 1, 0, 1.25)
  ThermoCoatDryfireLoop:
    WPLW A 6
    TNT1 A 0 A_JumpIfInventory ("FDSynthFireFire", 1, "ThermoCoatDryfireLoop")
    Goto ThermoCoatReady
	
  ThermoCoatFire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerEfficiency", 1, "ThermoCoatFireAmmoCheckDone")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareCell", 1, "ThermoCoatFireAmmoConvert")
	Goto ThermoCoatDryfire
  ThermoCoatFireAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fire", 7, 1, 0, 1.2)
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fire2", 5, 1, 0, 1.2)
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireloop", CHAN_WEAPON, 1, 1, 1.1)
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireloop2", 6, 1, 1, 0.7)

	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlame", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameFancy", 0, 0)
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerEfficiency", 1)
	WPLW B 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
	Goto ThermoCoatFireHeatLevelDone
  ThermoCoatFireHeatLevelDone:

    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)

    TNT1 A 0 A_StopSound (7)
	WPLW FG 2 Bright
	WPLW A 2
	Goto ThermoCoatReady
	
  ThermoCoatFireLoop:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerEfficiency", 1, "ThermoCoatFireLoopAmmoCheckDone")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareCell", 1, "ThermoCoatFireLoopAmmoConvert")
	Goto ThermoCoatDryfireFiring
  ThermoCoatFireLoopAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerEfficiency", 1)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerFrame", 2, "ThermoCoatFireLoopFrameE")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerFrame", 1, "ThermoCoatFireLoopFrameD")
	Goto ThermoCoatFireLoopFrameC
  ThermoCoatFireLoopFrameC:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlame", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameFancy", 0, 0)
	WPLW C 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerFrame", 1)
	Goto ThermoCoatFireLoopAnimationDone
  ThermoCoatFireLoopFrameD:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlow", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlowFancy", 0, 0)
	WPLW D 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerFrame", 1)
	Goto ThermoCoatFireLoopAnimationDone
  ThermoCoatFireLoopFrameE:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, 3)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlow", 0, 0)
	TNT1 A 0 A_Jump(256, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlamethrowerFlameNoGlowFancy", 0, 0)
	WPLW E 1 Bright A_SpawnItemEx ("FDTinyRecoilHax", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)
	Goto ThermoCoatFireLoopAnimationDone
  ThermoCoatFireLoopAnimationDone:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFlamethrowerEfficiency", 1, "ThermoCoatFireLoop")
	TNT1 A 0 A_Refire ("ThermoCoatFireLoop")
    TNT1 A 0 A_PlaySound("whitemareplasmarifle/fireend", CHAN_WEAPON, 1, 0, 1.2)
    TNT1 A 0 A_TakeInventory ("FDWhitemareFlamethrowerFrame", 0)

    TNT1 A 0 A_StopSound (7)

	Goto ThermoCoatFireAmmoCheckDone
  ThermoCoatFireLoopAmmoConvert:

    TNT1 A 0 A_GiveInventory ("FDWhitemareFlamethrowerEfficiency", 4)

  }
