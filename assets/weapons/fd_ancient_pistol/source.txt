 
ACTOR FDAliensPistol : FDWeapon
{
  Inventory.RestrictedTo "FDAliensPlayer"
  Inventory.Icon "AAPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAliensPistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%k managed to get a hit on %o and blew their head off."
  Decal "BulletChip"
  Tag "Ancient Pistol"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDAliensPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReadyUseAmmo")
    APIS A 1 A_WeaponReady
    Loop
  ReadyFull:
    APIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Goto Ready
	
  ReadyUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, 1)
	Goto ReadyFull
    APIS A 1 A_WeaponReady
    Goto Ready
	
  Select:
    APIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    APIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    APIS A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready

// aaaagh
// next person who asks for yet another pistol cvar gets fucking shanked

  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto FireNoAmmoCheckDone
  FireNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire")
    Goto FireNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, "FireNoAmmoCheckDone")
    Goto Dryfire
	
  FireNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0.2, 0.2, -1, randompick(17,20), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0.2, 0.2, -1, randompick(8,10), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDAliensBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireComplete
	
  FireNoMagConsumptionInaccurate: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireNoMagConsumptionInaccurateBalanced")
    TNT1 A 0 A_FireBullets (2.9, 2.9, -1, randompick(17,20), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagConsumptionInaccurateBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (2.9, 2.9, -1, randompick(8,10), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
	
  FireNoAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("alienspistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienspistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoMagConsumption") // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (0.2, 0.2, -1, randompick(17,20), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	Goto FireComplete
  Inaccurate:
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("alienspistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienspistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoMagConsumptionInaccurate") // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (2.9, 2.9, -1, randompick(17,20), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	Goto FireComplete
  FireComplete:
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD9mmCasingSpawner", 0,0,frandom(0.5,0.6),-1, FPF_NOAUTOAIM)
    APIS B 1 Bright A_SetPitch (pitch - (0.6 * CallACS("FD_recoil")/65536.0))
    APIS C 1 Bright A_SetPitch (pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    APIS D 1 A_SetPitch (pitch + (0.31 * CallACS("FD_recoil")/65536.0))
    APIS E 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    APIS F 1 A_SetPitch (pitch + (0.07 * CallACS("FD_recoil")/65536.0))
    APIS G 1 A_SetPitch (pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    APIS AAA 1
    TNT1 A 0 A_JumpIfNoAmmo ("Ready")
	Goto CrimeForReloading
	
  AltFire:
    TNT1 A 0 A_JumpIfInventory ("FDAliensPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 0, "Reloadery")
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, "Reloadery")
	Goto DryFire // Normally we'd return to Ready, but we disable altfire if full anyway, so we can go to dryfire without problems
  Reloadery:
    APIS HI 2
    TNT1 A 0 A_PlaySound("alienspistol/unload", 6, 1, 0, 1.25)
    APIS JK 2
    APIS L 16
    APIS M 2
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReloadUseAmmo")
    TNT1 A 0 A_GiveInventory ("FDAliensPistolAmmo", 13)
	Goto ReloadFinish
  ReloadUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDAliensPistolAmmo", 0, "ReloadFinish")
    TNT1 A 0 A_JumpIfInventory ("FDAliensBullets", 1, 1)
	Goto ReloadFinish
    TNT1 A 0 A_GiveInventory ("FDAliensPistolAmmo", 1)
    TNT1 A 0 A_TakeInventory ("FDAliensBullets", 1)
	Goto ReloadUseAmmo
  ReloadFinish:
    TNT1 A 0 A_PlaySound("alienspistol/load", 5, 1, 0, 1.25)
    APIS NOPQRSAA 2
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
		
	// no don't look it's crime
  CrimeForReloading:
    TNT1 A 0 A_WeaponReady (WRF_NOPRIMARY | WRF_NOBOB)// | WRF_NOSWITCH)
    APIS A 1 A_Refire ("Inaccurate")
    TNT1 A 0 A_WeaponReady (WRF_NOPRIMARY | WRF_NOBOB)// | WRF_NOSWITCH)
    APIS A 1 A_Refire ("Inaccurate")
    TNT1 A 0 A_WeaponReady (WRF_NOPRIMARY | WRF_NOBOB)// | WRF_NOSWITCH)
    APIS A 1 A_Refire ("Inaccurate")
    TNT1 A 0 A_WeaponReady (WRF_NOPRIMARY | WRF_NOBOB)// | WRF_NOSWITCH)
    APIS A 1 A_Refire ("Inaccurate")
    APIS AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 1 A_WeaponReady (WRF_NOBOB)// | WRF_NOSWITCH) // long enough that the shot will be accurate again
	Goto Ready
  }
}
