 
ACTOR FDPlutPistol : FDWeapon
{
  Inventory.RestrictedTo "FDPlutPlayer"
  Inventory.Icon "PLPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDPlutPistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%o lost a duel to %k."
  Decal "BulletChip"
  Tag "Ol' Faithful"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReadyUseAmmo")
    PPIS A 1 A_WeaponReady
    Loop
  ReadyFull:
    PPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Goto Ready
	
  ReadyUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDPlutBullets", 1, 1)
	Goto ReadyFull // Has no ammo to be reloaded, so disable altfire
    PPIS A 1 A_WeaponReady
    Goto Ready
	
  Select:
    PPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    PPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    PPIS A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready

// aaaagh
// next person who asks for yet another pistol cvar gets fucking shanked

  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto FireNoAmmoCheckDone
  FireNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire")
    Goto FireNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDPlutBullets", 1, "FireNoAmmoCheckDone")
    Goto Dryfire
	
  FireNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(22,24), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDPlutBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireComplete
	
  FireNoAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("plutpistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("plutpistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoMagConsumption") // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(22,24), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FireComplete:
    PPIS B 1 Bright A_SetPitch (pitch - (0.6 * CallACS("FD_recoil")/65536.0))
    PPIS B 1 Bright A_SetPitch (pitch - (0.06 * CallACS("FD_recoil")/65536.0))
    PPIS C 1 A_SetPitch (pitch - (0.02 * CallACS("FD_recoil")/65536.0))
    PPIS C 1
    PPIS D 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    PPIS D 1 A_SetPitch (pitch + (0.09 * CallACS("FD_recoil")/65536.0))
    PPIS E 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    PPIS E 1 A_SetPitch (pitch + (0.30 * CallACS("FD_recoil")/65536.0))
    PPIS F 1 A_SetPitch (pitch + (0.09 * CallACS("FD_recoil")/65536.0))
    PPIS F 1 A_SetPitch (pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    Goto Ready
	
  AltFire:
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 0, "Reloadery")
    TNT1 A 0 A_JumpIfInventory ("FDPlutBullets", 1, "Reloadery")
	Goto DryFire // Normally we'd return to Ready, but we disable altfire if full anyway, so we can go to dryfire without problems
  Reloadery:
    TNT1 A 0 A_PlaySound("plutpistol/reload", 6, 1, 0, 1.25)
    PPIS GHIJKL 2
  CasingSpawn:
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, "CasingSpawnDone")
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 6, "CasingSpawnDone")
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 5, 10)
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 4, 8)
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 3, 6)
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 2, 3)
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 1, 2)
	TNT1 AAAAAA 0 A_FireCustomMissile ("FDRevolverCasingSpawner", 0,0,frandom(-3.6,-3.2),-3, FPF_NOAUTOAIM)
	Goto CasingSpawnDone
  CasingSpawnDone:
    PPIS MN 2
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReloadUseAmmo")
    TNT1 A 0 A_GiveInventory ("FDPlutPistolAmmo", 6)
	Goto ReloadFinish
  ReloadUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDPlutPistolAmmo", 0, "ReloadFinish")
    TNT1 A 0 A_JumpIfInventory ("FDPlutBullets", 1, 1)
	Goto ReloadFinish
    TNT1 A 0 A_GiveInventory ("FDPlutPistolAmmo", 1)
    TNT1 A 0 A_TakeInventory ("FDPlutBullets", 1)
	Goto ReloadUseAmmo
  ReloadFinish:
    PPIS OPQRSTU 2
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}
