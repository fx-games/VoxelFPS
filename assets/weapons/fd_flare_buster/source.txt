 
ACTOR FDWhitemarePistol : FDWeapon
{
  Inventory.RestrictedTo "FDWhitemarePlayer"
  Inventory.Icon "WMPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDWhitemarePistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%k set %o on fire and it was hilarious."
  Tag "Flare Buster"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReady")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
    WPIS A 1 A_WeaponReady
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    TNT1 A 0 A_PlaySound("whitemare/steam3", 6, 0.3, 1, 1.8)
    TNT1 A 0 A_FireCustomMissile ("FDGunSteamSpawner", frandom(-0.1,0.1), 0, 0, 6, FPF_NOAUTOAIM)
    TNT1 A 0 A_GiveInventory ("FDWhitemarePistolRecharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolRecharge", 0, "AutoRecharge")
    Loop
  ReadyFull:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReadyFull")
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_TakeInventory ("FDWhitemarePistolRecharge", 100)
  ReadyFullLoop:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReadyFullLoop")
    WPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Loop
	
  AutoRecharge:
    TNT1 A 0 A_TakeInventory ("FDWhitemarePistolRecharge", 100)
    TNT1 A 0 A_GiveInventory ("FDWhitemarePistolAmmo", 1)
    Goto Ready
	
  Select:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatSelect")
    WPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
	
  Deselect:
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatDeselect")
    WPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatFire")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, 2) // If reloading is disabled, then don't fuggin reload
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
    TNT1 A 0 A_FireCustomMissile ("FDGunSteamSpawner", frandom(-0.1,0.1), 0, 0, 6)
    TNT1 A 0 A_PlaySound("whitemarepistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("whitemarepistol/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlareshot", 0, 0, 0, 6)
    TNT1 A 0 A_FireBullets (0, 0, 1, 0, "FDHarmlessPuff", FBF_NORANDOM) // Used to trigger wall switches
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FiringComplete") // If reloading is disabled, don't use ammo
	TNT1 A 0 A_TakeInventory ("FDWhitemarePistolAmmo", 1, TIF_NOTAKEINFINITE)
	Goto FiringComplete
  FiringComplete:
	// -0.6
    WPIS B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    WPIS B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    WPIS C 1 Bright A_SetPitch (pitch - (0.08 * CallACS("FD_recoil")/65536.0))
    WPIS C 1 Bright A_SetPitch (pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	// 0.6
    WPIS D 1 Bright A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    WPIS D 1 Bright A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    WPIS E 1 Bright A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    WPIS E 1 Bright A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    WPIS A 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    WPIS A 1 A_SetPitch (pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
	Goto Ready
	
	
  AltFire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatAltFire")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull") // If reloading is disabled, then don't fuggin reload
  Reloadery:
    TNT1 A 0 A_TakeInventory ("FDWhitemarePistolAmmo", 6)
    WPIS FGHI 2
    TNT1 A 0 A_PlaySound("whitemarepistol/open", 6, 1, 0, 1.25)
    TNT1 A 0 A_PlaySound("whitemarepistol/whirr1", 5, 0.6, 0, 1.65)
    WPIS JK 2
  ReloadLoop:
    TNT1 A 0 A_PlaySound("whitemare/steam3", CHAN_WEAPON, 0.3, 1, 1.8)
    WPIS LMLM 1 A_FireCustomMissile ("FDGunSteamSpawner", 8+frandom(-0.1,0.1), 0, 7, 13, FPF_NOAUTOAIM)
    TNT1 A 0 A_GiveInventory ("FDWhitemarePistolAmmo", 1)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolAmmo", 0, "ReloadEnd")
	Goto ReloadLoop
  ReloadEnd:
    TNT1 A 0 A_PlaySound("whitemarepistol/whirr2", 5, 0.6, 0, 1.65)
    WPIS NOP 2
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    TNT1 A 0 A_PlaySound("whitemarepistol/close", 6, 1, 0, 1.25)
    WPIS IHGFA 2
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone

// ============= THERMOCOAT ACTIVE =============================================
	
  ThermoCoatReady:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolAmmo", 0, "ThermoCoatReadyFull")
    WPIW A 1 A_WeaponReady
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    TNT1 A 0 A_PlaySound("whitemare/steam3", 6, 0.3, 1, 1.8)
    TNT1 A 0 A_FireCustomMissile ("FDGunSteamSpawner", frandom(-0.1,0.1), 0, 0, 6, FPF_NOAUTOAIM)
    TNT1 A 0 A_GiveInventory ("FDWhitemarePistolRecharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolRecharge", 0, "ThermoCoatAutoRecharge")
    Loop
  ThermoCoatReadyFull:
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_TakeInventory ("FDWhitemarePistolRecharge", 100)
  ThermoCoatReadyFullLoop:
    WPIW A 1 A_WeaponReady (WRF_NOSECONDARY)
    Loop
	
  ThermoCoatAutoRecharge:
    TNT1 A 0 A_TakeInventory ("FDWhitemarePistolRecharge", 100)
    TNT1 A 0 A_GiveInventory ("FDWhitemarePistolAmmo", 1)
    Goto ThermoCoatReady
	
  ThermoCoatSelect:
    WPIW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  ThermoCoatDeselect:
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    WPIW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  ThermoCoatFire:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, 2) // If reloading is disabled, then don't fuggin reload
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
    TNT1 A 0 A_FireCustomMissile ("FDGunSteamSpawner", frandom(-0.1,0.1), 0, 0, 6)
    TNT1 A 0 A_PlaySound("whitemarepistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("whitemarepistol/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireCustomMissile ("FDWhitemareFlareshot", 0, 0, 0, 6)
    TNT1 A 0 A_FireBullets (0, 0, 1, 0, "FDHarmlessPuff", FBF_NORANDOM) // Used to trigger wall switches
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ThermoCoatFiringComplete") // If reloading is disabled, don't use ammo
	TNT1 A 0 A_TakeInventory ("FDWhitemarePistolAmmo", 1, TIF_NOTAKEINFINITE)
	Goto ThermoCoatFiringComplete
  ThermoCoatFiringComplete:
    WPIW B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    WPIW B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    WPIW C 1 Bright A_SetPitch (pitch - (0.08 * CallACS("FD_recoil")/65536.0))
    WPIW C 1 Bright A_SetPitch (pitch - (0.02 * CallACS("FD_recoil")/65536.0))
    WPIW D 1 Bright A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    WPIW D 1 Bright A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    WPIW E 1 Bright A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    WPIW E 1 Bright A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    WPIW A 1 A_SetPitch (pitch + (0.048 * CallACS("FD_recoil")/65536.0))
    WPIW A 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfNoAmmo ("ThermoCoatAltFire")
	Goto ThermoCoatReady
	
  ThermoCoatAltFire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolAmmo", 0, "ThermoCoatReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ThermoCoatReadyFull") // If reloading is disabled, then don't fuggin reload
  ThermoCoatReloadery:
    TNT1 A 0 A_TakeInventory ("FDWhitemarePistolAmmo", 6)
    WPIW FGHI 2
    TNT1 A 0 A_PlaySound("whitemarepistol/open", 6, 1, 0, 1.25)
    TNT1 A 0 A_PlaySound("whitemarepistol/whirr1", 5, 0.6, 0, 1.65)
    WPIW JK 2
  ThermoCoatReloadLoop:
    TNT1 A 0 A_PlaySound("whitemare/steam3", CHAN_WEAPON, 0.3, 1, 1.8)
    WPIW LMLM 1 A_FireCustomMissile ("FDGunSteamSpawner", 8+frandom(-0.1,0.1), 0, 7, 13, FPF_NOAUTOAIM)
    TNT1 A 0 A_GiveInventory ("FDWhitemarePistolAmmo", 1)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemarePistolAmmo", 0, "ThermoCoatReloadEnd")
	Goto ThermoCoatReloadLoop
  ThermoCoatReloadEnd:
    TNT1 A 0 A_PlaySound("whitemarepistol/whirr2", 5, 0.6, 0, 1.65)
    WPIW NOP 2
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    TNT1 A 0 A_PlaySound("whitemarepistol/close", 6, 1, 0, 1.25)
    WPIW IHGFA 2
    Goto ThermoCoatReady
  }
}

ACTOR FDWhitemareFlareshot
{
  Radius 7
  Height 5
  Speed 30
  Projectile
  RenderStyle Add
  Alpha 0.7
  Scale 0.9
  Damage (5*random(1,3))
  DamageType "Fire"
  Decal "SmallFireScorch"
  Species "Player"
  +HITTRACER
  +THRUSPECIES
  +FORCEXYBILLBOARD
  +EXPLODEONWATER

	var int user_spriteframe;
  States
  {
  Spawn:
    TNT1 A 0
  SpawnLoop:
    TNT1 A 0 A_PlaySound("whitemarepistol/projectileloop", 7, 0.8, 1, 1.1)
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFlareGlow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFlareFireballFlare", frandom(-0.7,0.7),frandom(-0.7,0.7),frandom(-0.7,0.7), 0,0,frandom(-0.1,-0.05), 0, SXF_NOCHECKPOSITION)
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "SpriteFrame")
	TNT1 A 0 A_SpawnParticle (e06c0c, 					SPF_RELATIVE, random(3,7), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
	TNT1 AA 0 A_SpawnParticle (fca018, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,6), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
	TNT1 AA 0 A_SpawnParticle (fce080, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,5), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
	TNT1 A 0 A_SpawnParticle (ffffff,  SPF_FULLBRIGHT | SPF_RELATIVE, random(3,4), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, "SpriteFrame")
	Goto SpriteFrame
  SpriteFrame:
	TNT1 A 0 A_JumpIf (user_spriteframe == 1, "SpriteFrameB")
	Goto SpriteFrameA
  SpriteFrameA:
    WPIX A 1 Bright Light("FDWhitemarePistolFlare1")
	TNT1 A 0 A_SetUserVar("user_spriteframe", 1)
    Goto SpriteFrameEnd
  SpriteFrameB:
    WPIX B 1 Bright Light("FDWhitemarePistolFlare2")
	TNT1 A 0 A_SetUserVar("user_spriteframe", 0)
    Goto SpriteFrameEnd
  SpriteFrameEnd:
    TNT1 A 0 A_Jump (230, 2)
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireballSmoke", frandom(-0.7,0.7),frandom(-0.7,0.7),frandom(-0.7,0.7), 0,0,frandom(0.1,0.15), 0, SXF_NOCHECKPOSITION)
    TNT1 A 0
    Goto SpawnLoop
	
  XDeath:
  Crash:
	TNT1 A 0
	Goto IgniteDOT
	
	// This sequence is used to set enemies alight.
  IgniteDOT:
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, "GenerateDOTTime", AAPTR_TRACER)
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 1, "DOT1")
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 2, "DOT2")
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 3, "DOT3")
  DOT0:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOT", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT1:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoSmoke", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT2:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTToaster", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT3:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoVisuals", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  GenerateDOTTime:
	TNT1 AAAAAAAAAA 0 A_GiveInventory ("FDWhitemareFireDOTActive", 1, AAPTR_TRACER) // 10 seconds guaranteed burn
	Goto DeathParticles
	
  Death:
//    TNT1 A 0 A_SpawnItemEx ("FDWhitemarePistolFlareIlluminator", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION)
    TNT1 A 0 A_Jump (256, "DeathParticles")
	Goto DeathParticles
  DeathParticles:
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "DeathAnimation")
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAA 0 A_SpawnParticle (e06c0c, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAA 0 A_SpawnParticle (fca018, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAAAA 0 A_SpawnParticle (fce080, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:
	TNT1 A 0 A_SetScale (scalex * randompick(-1, 1), scaley) // randomly flip the sprite horizontally
    TNT1 A 0 A_PlaySound ("weapons/plasmax", 7, 1, 0, 1.2)
    WPIX K 3 Bright Light("FDWhitemarePistolFlareImpact1")
    WPIX L 3 Bright Light("FDWhitemarePistolFlareImpact2")
    WPIX M 3 Bright Light("FDWhitemarePistolFlareImpact3")
    WPIX N 3 Bright Light("FDWhitemarePistolFlareImpact4")
    WPIX O 3 Bright Light("FDWhitemarePistolFlareImpact5")
    WPIX P 3 Bright Light("FDWhitemarePistolFlareImpact6")
    Stop
  }
}
