 
ACTOR FDAlienVendettaShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVSHOTGU"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAlienVendettaShells"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 3
  Obituary "%o was unable to stand before %k's onslaught."
  Decal "BulletChip"
  Tag "Drum Shotgun"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShells", 1, "ReadyAmmo")
    VSHT A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    VSHT A 1 A_WeaponReady
    Loop
	
  Select:
    VSHT A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    VSHT A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    VSHT A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShells", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,3)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShotgunState", 1, 4)
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaShotgunState", 1)
    TNT1 A 0 A_PlaySound("alienvendettashotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienvendettashotgun/fire", 7, 0.05, 0, 0.20)
	Goto FireSoundDone
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaShotgunState", 1)
    TNT1 A 0 A_PlaySound("alienvendettashotgun/fire", 6, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienvendettashotgun/fire", CHAN_BODY, 0.05, 0, 0.20)
	Goto FireSoundDone
  FireSoundDone:
    TNT1 A 0 A_FireBullets (5.6, 5.8, 9, 5, "FDBulletPuff", FBF_USEAMMO)
    VSHT B 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    VSHT B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    VSHT C 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    VSHT C 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    VSHT D 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    VSHT D 1 A_SetPitch (pitch + (1.05 * CallACS("FD_recoil")/65536.0))
    VSHT E 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
    VSHT A 2
  GonnaReload:
    TNT1 A 0 A_PlaySound("alienvendettashotgun/pumpin", 5, 1, 0, 1.25)
	VSHT A 1 A_SetPitch(pitch + (0.09 * CallACS("FD_recoil")/65536.0))
	VSHT F 1 A_SetPitch(pitch + (0.23 * CallACS("FD_recoil")/65536.0))
	VSHT F 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDTNTShellCasingSpawner", 0,0,-6,-6, FPF_NOAUTOAIM)
	VSHT G 1 A_SetPitch(pitch - (0.06 * CallACS("FD_recoil")/65536.0))
	VSHT G 1 A_SetPitch(pitch - (0.12 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_PlaySound("alienvendettashotgun/pumpout", 5, 1, 0, 1.25)
	VSHT H 1 A_SetPitch(pitch - (0.10 * CallACS("FD_recoil")/65536.0))
	VSHT H 1 A_SetPitch(pitch - (0.07 * CallACS("FD_recoil")/65536.0))
	VSHT I 1 A_SetPitch(pitch - (0.04 * CallACS("FD_recoil")/65536.0))
	VSHT I 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
	VSHT A 3
	VSHT AAAAAAA 1 A_Refire // Gun fires at a speed of 37 tics
    Goto Ready
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

Actor FDAlienVendettaShotgunState : FDBoolean {}
