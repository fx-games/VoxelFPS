 
ACTOR FDTNTRocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDTNTPlayer"
  Inventory.Icon "TNROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDTNTRocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was blasted apart by %k."
  Decal ""
  Tag "Missile Launcher"
  States
  {
  Ready:
    TROK A 1 A_WeaponReady
    Loop
	
  Select:
    TROK A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    TROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDTNTRocket", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_GunFlash
    TROK BBCC 1 Bright
    TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("tntrocketlauncher/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("tntrocketlauncher/fire", 7, 1, 0, 0.4)
	TNT1 A 0 A_FireCustomMissile ("FDTNTMissile", frandom(-1.0,1.0), 1)
    TROK D 1 Bright A_SetPitch (pitch - (1.8 * CallACS("FD_recoil")/65536.0))
    TROK D 1 Bright A_SetPitch (pitch - (0.15 * CallACS("FD_recoil")/65536.0))
    TROK E 1 Bright A_SetPitch (pitch - (0.05 * CallACS("FD_recoil")/65536.0))
    TROK E 1 Bright A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    TROK F 1 A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TROK F 1 A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TROK A 1 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
    TROK A 1 A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TROK A 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    TROK A 1 A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    TROK AAAAAAAAAA 1 A_Refire
    Goto Ready
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 2 A_Light (3)
    TNT1 A 2 A_Light (4)
    TNT1 A 2 A_Light2
    TNT1 A 2 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

ACTOR FDTNTMissile : Rocket
{
  Radius 6
  Height 4
  Scale 0.4
  Speed 6
  Damage (30*random(1,3))
  Damagetype "Explosive"
  Species "Player"
  -ROCKETTRAIL
  -DEHEXPLOSION
  +THRUSPECIES
  +THRUGHOST
  +EXPLODEONWATER
  Decal "Scorch"
  SeeSound ""
  DeathSound ""
  States
  {
  Spawn:
    TMIS B 12
    TNT1 A 0 A_PlaySound ("tntrocketlauncher/jet", 7, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound ("tntrocketlauncher/loop", CHAN_WEAPON, 1, 1, 0.5)
    TNT1 A 0 A_ScaleVelocity (6)
  GoMissileGoGoGo:
    TMIS A 1 Bright Light("FDRocketPulse") A_SpawnItemEx ("FDTNTMissileTrail", -8,0,0, -5,0,0, 0, SXF_NOCHECKPOSITION)
    Loop
  Death:
    TNT1 A 0 A_SetTranslucent (0.95, 1)
    TNT1 A 0 A_SetScale (1.0)
    TNT1 A 0 A_Explode (128,128)
    TNT1 A 0 A_PlaySound ("tntrocketlauncher/explosion", 7, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("tntrocketlauncher/explosion", CHAN_WEAPON, 0.05, 0, 0.05)
    TNT1 A 0 A_Stop
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "QuakeTime")
    TNT1 A 0 A_Jump (256, 1,5,10)
	TNT1 AAAAAAAAAAAA 0 A_SpawnParticle (730000, SPF_FULLBRIGHT | SPF_RELATIVE, random(30,70), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,5,10)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (ef0000, SPF_FULLBRIGHT | SPF_RELATIVE, random(25,60), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,5,10)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (f37317, SPF_FULLBRIGHT | SPF_RELATIVE, random(20,50), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,5,10)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (ffa35b, SPF_FULLBRIGHT | SPF_RELATIVE, random(15,40), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,3,6)
	TNT1 AAAAAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,4), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, "QuakeTime")
	Goto QuakeTime
  QuakeTime:
	TNT1 A 0 A_JumpIf(CallACS("FD_quakes") == 0, "DeathAnimation")
    TNT1 A 0 A_Quake (1, 12, 0, 1024, "")
    TNT1 A 0 A_Quake (3, 8, 0, 512, "")
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:
    MEXP A 4 Bright Light("FDExplosion1")
    MEXP B 4 Bright Light("FDExplosion2")
    MEXP C 3 Bright Light("FDExplosion3")
    MEXP D 3 Bright Light("FDExplosion4")
    MEXP E 2 Bright Light("FDExplosion5")
    MEXP F 2 Bright Light("FDExplosion6")
    Stop
  }
}
