 
ACTOR FDJPCPBFG9000 : FDBFGWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPBFG900"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPBFGCharge"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 7
  Obituary "%o was crushed by the power of light and darkness, courtesy of %k."
  Decal ""
  Tag "Onmyou Devastator"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("jpcpbfg9000/idle", 7, 0.3, 1, 1.4)
    JBFG A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    JBFG A 1 A_WeaponReady
    Loop
	
  Select:
    JBFG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    "####" "#" 0 A_StopSound(7)
  DeselectLoop:
    JBFG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JBFG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBFGCharge", 1, "ChargeStart")
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDJPCPCell", 50, "ChargeStart")
	Goto Dryfire
	
  ChargeStart:
    TNT1 A 0 A_PlaySound("jpcpbfg9000/charge", CHAN_WEAPON, 1, 0, 0.8)
    TNT1 A 0 A_PlaySound("jpcpbfg9000/kanji", 7, 0.7, 0, 1.4)
    TNT1 A 0 A_AlertMonsters (random(256,400))
    TNT1 A 0 A_GunFlash ("FlashChargeup", GFF_NOEXTCHANGE)
	JBFG BCCCD 3 Bright
    TNT1 A 0 A_PlaySound("jpcpbfg9000/orb", 6, 0.7, 0, 1.4)
	JBFG EFGHIJ 3 Bright
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_TakeInventory ("FDJPCPBFGCharge", 1, TIF_NOTAKEINFINITE)
	Goto RELEASEBALANCE
    TNT1 A 0 A_TakeInventory ("FDJPCPCell", 50, TIF_NOTAKEINFINITE)
	Goto RELEASEBALANCE
  RELEASEBALANCE:
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_FireCustomMissile ("FDJPCPBFGBall", 0, 0, 0, 6)
	JBFG K 1 Bright A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
	JBFG K 1 Bright A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	JBFG K 1 Bright A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	JBFG L 1 Bright A_SetPitch(pitch - (2.38 * CallACS("FD_recoil")/65536.0))
	JBFG L 1 Bright A_SetPitch(pitch - (0.03 * CallACS("FD_recoil")/65536.0))
	JBFG L 1 Bright A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0)) // 2.5
	JBFG M 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
	JBFG M 1 A_SetPitch(pitch + (1.35 * CallACS("FD_recoil")/65536.0))
	JBFG M 1 A_SetPitch(pitch + (1.05 * CallACS("FD_recoil")/65536.0))
	JBFG N 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	JBFG N 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	JBFG N 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0)) // 2.88
	JBFG A 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	JBFG A 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	JBFG A 1 A_SetPitch(pitch - (0.08 * CallACS("FD_recoil")/65536.0)) // 2.5
    Goto Ready
	
  Flash:
	TNT1 A 3 A_Light (10)
	TNT1 A 3 A_Light (3)
	TNT1 A 3 A_Light2
	TNT1 A 3 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  FlashChargeup:
	TNT1 A 6 A_Light1
	TNT1 A 6 A_Light2
	TNT1 A 3 A_Light (5)
	TNT1 A 12 A_Light (random(1,3))
	TNT1 A 3 A_Light (5)
	TNT1 A 3 A_Light (8)
	TNT1 A 3 A_Light0
    Goto LightDone
  }
}

/*
// The BFG went through three entire iterations to get to where it is now.
//
// The first iteration placed an orb on the end of the gun, which would block shots and change modes constantly.
// Releasing the trigger would launch it in the mode it was in.
// Doing that was pretty much unfeasible.
//
// The second iteration placed a stationary orb that would change modes and block shots.
// The BFG activated repulsors, and running into the orb would launch it in the direction you are looking in the mode it was in.
// While I actually got that working, it wasn't fun to use, was really gimmicky and obtuse for no real reason.
//
// After more discussion, we came up with the final iteration, a big boomy bfg ball of bang.
// Keep it simple, stupid.


// This is the code from the second version. I decided not to delete it because it's fucking madness and it pleases me.

ACTOR FDJPCPYinYangStationary
{
  Radius 24
  Height 16
  Health 100000
  Speed 0
  PainChance 256
  Scale 0.6
  RenderStyle Translucent
  Alpha 1.0
//  Species "Player"
//  +THRUSPECIES
  +NODAMAGE
  +SHOOTABLE
  +SOLID
  +ISMONSTER
  +NOTAUTOAIMED
  +NOGRAVITY
  +DONTFALL
  +NOTRIGGER
  +FRIENDLY
  +QUICKTORETALIATE
  +CANNOTPUSH
  +NOTELEPORT
  +NOTELESTOMP
  +DONTBLAST
  +NOBLOCKMONST
  +NOINFIGHTING
  +NOTIMEFREEZE
  +NOFEAR
  +SEEINVISIBLE
  +DONTTHRUST
  +NORADIUSDMG
  +DONTMORPH
  +DONTSQUASH
  +NOTELEOTHER
  +NOTELEFRAG
  +DONTDRAIN
  +NOBLOODDECALS
  +DONTGIB
  +DONTSPLASH
  +FORCEXYBILLBOARD
  -COUNTKILL
  
  var int user_yinyangmode;
  var int user_existancetimer;
  States
  {
  Spawn:
    TNT1 A 0 NoDelay A_PlaySound ("jpcpyinyang/idle", 7, 0.5, 1, 1.4)
    TNT1 A 0 A_GiveInventory ("FDJPCPYinYangExists", 1, AAPTR_TARGET)
	Goto ModeSwitch
	
  ModeSwitch:
    "####" A 0 A_JumpIf (user_yinyangmode == 1, "YinMode")
    "####" A 0 A_JumpIf (user_yinyangmode == 3, "YangMode")
  YinYangMode:
    JYYB A 0 A_PlaySound ("jpcpyinyang/mode", 6, 1, 0, 1.1)
    Goto MainLoop
  YinMode:
    JYIN A 0 A_PlaySound ("jpcpyin/mode", 6, 1, 0, 1.1)
    Goto MainLoop
  YangMode:
    JYAN A 0 A_PlaySound ("jpcpyang/mode", 6, 1, 0, 1.1)
    Goto MainLoop
	
  MainLoop:
    "####" AAAA 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" BBBB 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" CCCC 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" DDDD 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" EEEE 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" FFFF 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" GGGG 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
    "####" HHHH 1 Bright A_JumpIfInventory ("FDJPCPYangBang", 1, "LAUNCHSEQUENCE")
  LoopEnd:
    "####" A 0 A_SetUserVar("user_existancetimer", user_existancetimer + 1)
    "####" A 0 A_JumpIf (user_existancetimer >= 20, "IsOfTheDed")
    "####" A 0 A_JumpIf (user_yinyangmode == 3, "Reset")
    "####" A 0 A_SetUserVar("user_yinyangmode", user_yinyangmode + 1)
    Goto ModeSwitch
  Reset:
    "####" A 0 A_SetUserVar("user_yinyangmode", 0)
    Goto ModeSwitch
	
  Pain: // Decreases the time it'll remain active, but counterattacks enemies
    "####" A 0 A_SetUserVar("user_existancetimer", user_existancetimer + 1)
    "####" A 0 A_JumpIf (user_existancetimer >= 20, "IsOfTheDed")
    "####" A 0 A_JumpIf (user_yinyangmode == 1, "PainYin")
    "####" A 0 A_JumpIf (user_yinyangmode == 3, "PainYang")
  PainYinYang:
    "####" A 0 A_PlaySound ("jpcpyinyang/mode", 6, 1, 0, 1.1)
    Goto MainLoop
  PainYin:
    "####" A 0 A_PlaySound ("jpcpyin/mode", 6, 1, 0, 1.1)
    Goto MainLoop
  PainYang:
    "####" A 0 A_PlaySound ("jpcpyang/mode", 6, 1, 0, 1.1)
    Goto MainLoop
	
  LAUNCHSEQUENCE: // Launches in the direction you are looking when you hit it
    "####" A 0 A_TakeInventory ("FDJPCPYinYangExists", 1, 0, AAPTR_MASTER)
    "####" A 0 A_ChangeFlag ("SOLID", 0)
    "####" A 0 A_ChangeFlag ("SHOOTABLE", 0)
    "####" A 0 A_ChangeFlag ("MISSILE", 1)
    "####" A 0 A_ChangeFlag ("ISMONSTER", 0)
	"####" A 0 A_SetAngle ((CallACS("FDGetMasterAngle") / 65536.0) * 360.0)
    "####" A 0 A_RearrangePointers (AAPTR_MASTER, AAPTR_DEFAULT, AAPTR_NULL)
    "####" A 0 A_JumpIf (user_yinyangmode == 1, "YinChosen")
