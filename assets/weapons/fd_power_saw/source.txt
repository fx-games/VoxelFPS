 
ACTOR FDPlutChainsaw : FDWeapon
{
  Inventory.RestrictedTo "FDPlutPlayer"
  Inventory.Icon "PLCHAINS"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Obituary "%o was opened like a crate by %k."
  Decal ""
  +WEAPON.MELEEWEAPON
  Tag "Power Saw"
  States
  {
  Ready:
    TNT1 A 0 A_PlaySound("plutchainsaw/idleloop", 7, 0.8, 1, 1.5)
    TNT1 A 0 A_AlertMonsters (random(256,368))
    PSAW A 4 A_WeaponReady
    TNT1 A 0 A_AlertMonsters (random(256,368))
    PSAW B 4 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0 A_PlaySound("plutchainsaw/idlestart", 5, 0.8, 0, 1.5)
  SelectLoop:
    PSAW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_PlaySound("plutchainsaw/idlestop", 7, 0.8, 0, 1.5)
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
  DeselectLoop:
    PSAW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("plutchainsaw/start", CHAN_WEAPON, 0.7, 0, 0.9)
    PSAW ABABAB 2
    TNT1 A 0 A_Refire
    Goto FireEnd
  Hold:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("plutchainsaw/loop", CHAN_WEAPON, 0.7, 1, 0.9)
    TNT1 A 0 A_Saw ("", "", 2*random(2,5), "FDPlutChainsawPuff", SF_NOPULLIN | SF_NOTURN | SF_NORANDOM, 84, 3.2, 0.05)
    PSAW C 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    PSAW D 1 A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Saw ("", "", 2*random(2,5), "FDPlutChainsawPuff", SF_NOPULLIN | SF_NOTURN | SF_NORANDOM, 84, 3.2, 0.05)
    PSAW E 1 A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    PSAW F 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire
    Goto FireEnd
	
  FireEnd:
    TNT1 A 0 A_PlaySound("plutchainsaw/stop", CHAN_WEAPON, 0.7, 0, 0.9)
    PSAW AB 4 A_AlertMonsters (random(256,368))
    Goto Ready
  }
}
