 
ACTOR FDWhitemareShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDWhitemarePlayer"
  Inventory.Icon "WMSHOTGU"
  Weapon.UpSound ""
  Weapon.AmmoType "FDWhitemareShells"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 3
  Obituary "%o was scorched by %k's firey shotgun."
  Decal "BulletChip"
  Tag "Drake Knight Shotgun"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReady")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareShells", 1, "ReadyAmmo")
    WSHT A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatReadyAmmo")
    WSHT A 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatSelect")
    WSHT A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatDeselect")
    WSHT A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatDryfire")
    WSHT A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, "ThermoCoatFire")
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareShells", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("whitemareshotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("whitemareshotgun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareShotgunFireball", 0, 1)
	TNT1 AAAAA 0 A_FireCustomMissile ("FDWhitemareShotgunFireball", frandom(-5.6,5.6), 0, 0, 0, 0, frandom(-0.8,0.8))
	Goto FiringComplete
  FiringComplete:
    WSHT B 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    WSHT B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    WSHT C 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    WSHT C 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    WSHT D 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    WSHT D 1 A_SetPitch (pitch + (1.05 * CallACS("FD_recoil")/65536.0))
    WSHT E 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
  GonnaReload:
    WSHT F 1
    WSHT G 1
	WSHT H 2
	WSHT I 3
    TNT1 A 0 A_PlaySound("whitemareshotgun/pump", 5, 1, 0, 1.25)
	WSHT J 3
	WSHT K 1
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareShellCasingSpawner", 0,0,-14,-6, FPF_NOAUTOAIM)
    TNT1 A 0 A_PlaySound("whitemaresupershotgun/hiss", 6, 0.5, 0, 1.25)
	WSHT L 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	WSHT M 1 A_SetPitch(pitch + (0.23 * CallACS("FD_recoil")/65536.0))
	WSHT M 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	WSHT M 1 A_SetPitch(pitch - (0.06 * CallACS("FD_recoil")/65536.0))
	WSHT L 1 A_SetPitch(pitch - (0.28 * CallACS("FD_recoil")/65536.0))
	WSHT K 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	WSHT JI 1
	WSHT HGFN 2
	WSHT A 3
	WSHT AAAAAAA 1 A_Refire // Gun fires at a speed of 37 tics
    Goto Ready
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
	
// ============= THERMOCOAT ACTIVE =============================================
	
  ThermoCoatReady:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareShells", 1, "ThermoCoatReadyAmmo")
    WSHW A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ThermoCoatReadyAmmo:
    WSHW A 1 A_WeaponReady
    Loop
	
  ThermoCoatSelect:
    WSHW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  ThermoCoatDeselect:
    WSHW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  ThermoCoatDryfire:
    WSHW A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  ThermoCoatFire:
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareShells", 1, 1)
	Goto ThermoCoatDryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("whitemareshotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("whitemareshotgun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareShotgunFireball", 0, 1)
	TNT1 AAAAA 0 A_FireCustomMissile ("FDWhitemareShotgunFireball", frandom(-5.6,5.6), 0, 0, 0, 0, frandom(-0.8,0.8))
	Goto ThermoCoatFiringComplete
  ThermoCoatFiringComplete:
    WSHW B 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    WSHW B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    WSHW C 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    WSHW C 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    WSHW D 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    WSHW D 1 A_SetPitch (pitch + (1.05 * CallACS("FD_recoil")/65536.0))
    WSHW E 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
  ThermoCoatGonnaReload:
    WSHW F 1
    WSHW G 1
	WSHW H 2
	WSHW I 3
    TNT1 A 0 A_PlaySound("whitemareshotgun/pump", 5, 1, 0, 1.25)
	WSHW J 3
	WSHW K 1
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareShellCasingSpawner", 0,0,-14,-6, FPF_NOAUTOAIM)
    TNT1 A 0 A_PlaySound("whitemaresupershotgun/hiss", 6, 0.5, 0, 1.25)
	WSHW L 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	WSHW M 1 A_SetPitch(pitch + (0.23 * CallACS("FD_recoil")/65536.0))
	WSHW M 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	WSHW M 1 A_SetPitch(pitch - (0.06 * CallACS("FD_recoil")/65536.0))
	WSHW L 1 A_SetPitch(pitch - (0.28 * CallACS("FD_recoil")/65536.0))
	WSHW K 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	WSHW JI 1
	WSHW HGFN 2
	WSHW A 3
	WSHW AAAAAAA 1 A_Refire // Gun fires at a speed of 37 tics
    Goto ThermoCoatReady
  }
}

ACTOR FDWhitemareShotgunFireball
{
  Radius 7
  Height 5
  Speed 30
  Projectile
  RenderStyle Add
  Alpha 0.98
  Damage (8*random(1,3))
  DamageType "Fire"
  Decal "BigFireScorch"
  Species "Player"
  +THRUSPECIES
  +FORCEXYBILLBOARD
  +EXPLODEONWATER
  +HITTRACER

	var int user_spriteframe;
  States
  {
  Spawn:
    TNT1 A 0
  SpawnLoop:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireballFlare", frandom(-0.7,0.7),frandom(-0.7,0.7),frandom(-0.7,0.7), 0,0,frandom(-0.1,-0.05), 0, SXF_NOCHECKPOSITION)
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "SpriteFrame")
	TNT1 A 0 A_SpawnParticle (430000, 					SPF_RELATIVE, random(3,7), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
	TNT1 AA 0 A_SpawnParticle (bf0000, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,6), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
	TNT1 AA 0 A_SpawnParticle (eb6f0f, SPF_FULLBRIGHT | SPF_RELATIVE, random(3,5), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
	TNT1 A 0 A_SpawnParticle (ffff00,  SPF_FULLBRIGHT | SPF_RELATIVE, random(3,4), random(1,3), 0, frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-0.2,0.2), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, "SpriteFrame")
	Goto SpriteFrame
  SpriteFrame:
	TNT1 A 0 A_JumpIf (user_spriteframe >= 6, "SpriteFrameD")
	TNT1 A 0 A_JumpIf (user_spriteframe >= 4, "SpriteFrameC")
	TNT1 A 0 A_JumpIf (user_spriteframe >= 2, "SpriteFrameB")
  SpriteFrameA:
    WFRB A 1 Bright Light("FDWhitemareShotgunFire")
    Goto SpriteFrameEnd
  SpriteFrameB:
    WFRB B 1 Bright Light("FDWhitemareShotgunFire")
    Goto SpriteFrameEnd
  SpriteFrameC:
    WFRB C 1 Bright Light("FDWhitemareShotgunFire")
    Goto SpriteFrameEnd
  SpriteFrameD:
    WFRB D 1 Bright Light("FDWhitemareShotgunFire")
    Goto SpriteFrameEnd
  SpriteFrameEnd:
    TNT1 A 0 A_Jump (230, 2)
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireballSmoke", frandom(-0.7,0.7),frandom(-0.7,0.7),frandom(-0.7,0.7), 0,0,frandom(0.1,0.15), 0, SXF_NOCHECKPOSITION)
	TNT1 A 0 A_SetUserVar("user_spriteframe", user_spriteframe + 1)
	TNT1 A 0 A_JumpIf (user_spriteframe >= 8, "ResetFrame")
    TNT1 A 0 A_Jump (256, "SpawnLoop")
    Goto SpawnLoop
	
  ResetFrame:
	TNT1 A 0 A_SetUserVar("user_spriteframe", 0)
    TNT1 A 0 A_Jump (256, "SpawnLoop")
    Goto SpawnLoop
	
  XDeath:
  Crash:
	TNT1 A 0
	Goto IgniteDOT
	
	// This sequence is used to set enemies alight.
  IgniteDOT:
	TNT1 A 0 A_Jump (180, "DeathParticles") // High chance to not ignite enemies.
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, "GenerateDOTTime", AAPTR_TRACER)
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 1, "DOT1")
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 2, "DOT2")
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 3, "DOT3")
  DOT0:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOT", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT1:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoSmoke", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT2:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTToaster", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT3:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoVisuals", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  GenerateDOTTime:
	TNT1 AAAA 0 A_GiveInventory ("FDWhitemareFireDOTActive", 1, AAPTR_TRACER) // 4 seconds chance-based burn
	Goto Death
	
  Death:
    TNT1 A 0 A_Jump (256, "DeathParticles")
	Goto DeathParticles
  DeathParticles:
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "DeathAnimation")
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAA 0 A_SpawnParticle (430000, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAA 0 A_SpawnParticle (bf0000, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAAAAAAA 0 A_SpawnParticle (eb6f0f, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAA 0 A_SpawnParticle (ffff00, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:
	TNT1 A 0 A_SetScale (scalex * randompick(-1, 1), scaley) // randomly flip the sprite horizontally
    TNT1 A 0 A_PlaySound ("weapons/plasmax", 7, 1, 0, 1.2)
    WFRB K 3 Bright Light("FDWhitemareShotgunFireImpact1")
    WFRB L 3 Bright Light("FDWhitemareShotgunFireImpact2")
    WFRB M 3 Bright Light("FDWhitemareShotgunFireImpact3")
    WFRB N 3 Bright Light("FDWhitemareShotgunFireImpact4")
    WFRB O 3 Bright Light("FDWhitemareShotgunFireImpact5")
    WFRB P 3 Bright Light("FDWhitemareShotgunFireImpact6")
    Stop
  }
}

// cast from the Blazeblade from a murderslash
ACTOR FDWhitemareFistFireball : FDWhitemareShotgunFireball
{
  Damage (4*random(1,3))
  Speed 18
  ReactionTime 1
  States
  {
  ResetFrame:
	TNT1 A 0 A_SetUserVar("user_spriteframe", 0)
    TNT1 A 0 A_Countdown
    TNT1 A 0 A_Jump (256, "SpawnLoop")
    Goto SpawnLoop
	
	// This sequence is used to set enemies alight.
  IgniteDOT:
	TNT1 A 0 A_Jump (100, "DeathParticles") // Decent chance to ignite enemies.
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, "GenerateDOTTime", AAPTR_TRACER)
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 1, "DOT1")
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 2, "DOT2")
	TNT1 A 0 A_JumpIf(CallACS("FD_enemyeffects") == 3, "DOT3")
  DOT0:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOT", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT1:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoSmoke", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT2:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTToaster", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  DOT3:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoVisuals", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  GenerateDOTTime:
	TNT1 AAA 0 A_GiveInventory ("FDWhitemareFireDOTActive", 1, AAPTR_TRACER) // 3 seconds chance-based burn
	Goto Death
  }
}
