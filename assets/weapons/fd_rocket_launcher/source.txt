 
ACTOR FDDoom2RocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2ROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDDoom2Rocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was blown to smitheroons by %k."
  Decal ""
  Tag "Rocket Launcher"
  States
  {
  Ready:
    DROK A 1 A_WeaponReady
    Loop
	
  Select:
    DROK A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    DROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    DROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Rocket", 1, 1)
	Goto Dryfire
	DROK A 2
    DROK A 4 A_GunFlash ("Flash")
	TNT1 A 0 A_AlertMonsters
	DROK B 2
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("doom2rocketlauncher/fire1", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("doom2rocketlauncher/fire2", 7, 1, 0, 0.4)
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla")
	TNT1 A 0 A_FireCustomMissile ("FDDoom2RocketProjectile", frandom(-1.0,1.0), 1)
  FiringComplete:
	DROK C 1 A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	DROK C 1 A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0)) // 3.25
	DROK C 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	DROK D 1 A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	DROK D 1 A_SetPitch(pitch + (0.6 * CallACS("FD_recoil")/65536.0))
	DROK D 1 A_SetPitch(pitch + (0.39 * CallACS("FD_recoil")/65536.0))
	DROK E 1 A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	DROK E 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	DROK F 1 A_SetPitch(pitch + (0.19 * CallACS("FD_recoil")/65536.0))
	DROK F 1 A_SetPitch(pitch + (0.13 * CallACS("FD_recoil")/65536.0))
	DROK G 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	DROK A 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0)) // 3.25
    DROK AAAAAAAAAA 1 A_Refire
    Goto Ready
  FiringVanilla:
	TNT1 A 0 A_FireCustomMissile ("FDDoom2RocketProjectileVanilla", 0, 1)
	Goto FiringComplete
	
  Flash:
	DROK H 2 Bright A_Light1
	DROK I 2 Bright
	DROK J 2 Bright A_Light2
	DROK K 3 Bright
	DROK L 3 Bright
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}

    TNT1 A 0 A_Jump (256, 1,5,10)

	TNT1 AAAAAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,4), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, "QuakeTime")
	Goto QuakeTime
  QuakeTime:
	TNT1 A 0 A_JumpIf(CallACS("FD_quakes") == 0, "DeathAnimation")
    TNT1 A 0 A_Quake (1, 12, 0, 1024, "")
    TNT1 A 0 A_Quake (5, 8, 0, 512, "")
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:
    TNT1 A 0 A_SetScale (1.0)
    TNT1 A 0 A_SetTranslucent (0.98, 1)
    TNT1 A 0 A_PlaySound("doom2rocketlauncher/explosion", CHAN_WEAPON, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("doom2rocketlauncher/explosion", 7, 0.10, 0, 0.01)
    MEXP A 4 Bright Light("FDExplosion1")
    MEXP B 4 Bright Light("FDExplosion2")
    MEXP C 3 Bright Light("FDExplosion3")
    MEXP D 3 Bright Light("FDExplosion4")
    MEXP E 2 Bright Light("FDExplosion5")
    MEXP F 2 Bright Light("FDExplosion6")
    Stop
  }
}

ACTOR FDDoom2RocketSmokeTrail
{
  Radius 4
  Height 4
  RenderStyle Translucent
  Alpha 0.4
  +NOGRAVITY
  +NOINTERACTION
  +FORCEXYBILLBOARD
  States
  {
  Spawn:
	TNT1 A 3
    TNT1 A 0 A_SetScale (scalex - (scalex * 2), scaley)
	TNT1 A 0 A_SetScale (frandom(0.9,1.5))
	TNT1 A 0 A_SetScale (scalex + frandom(0.03,0.1))
	SSMO A 1 A_FadeTo (0, frandom(0.01,0.03), 1)
	TNT1 A 0 A_SetScale (scalex + frandom(0.03,0.1))
	SSMO B 1 A_FadeTo (0, frandom(0.01,0.03), 1)
	TNT1 A 0 A_SetScale (scalex + frandom(0.03,0.1))
	SSMO C 1 A_FadeTo (0, frandom(0.01,0.03), 1)
	TNT1 A 0 A_SetScale (scalex + frandom(0.03,0.1))
	SSMO D 1 A_FadeTo (0, frandom(0.01,0.03), 1)
	TNT1 A 0 A_SetScale (scalex + frandom(0.03,0.1))
	SSMO E 1 A_FadeTo (0, frandom(0.01,0.03), 1)
  SpawnLoop:
	TNT1 A 0 A_SetScale (scalex + frandom(0.03,0.1))
	SSMO E 1 A_FadeTo (0, frandom(0.05,0.08), 1)
    Loop
  }
}
