 
ACTOR FDJPCPSuperShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPSUPERS"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPShells"
  Weapon.AmmoUse 2
  Weapon.SlotNumber 3
  Obituary "%o got resonated by the neutron triple threat of %k."
  Decal ""
  Tag "Neutron Cascader"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPShells", 2, "ReadyAmmo")
    JSSG A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    JSSG A 1 A_WeaponReady
    Loop
	
  Select:
    JSSG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    JSSG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JSSG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPShells", 2, 1)
	Goto Dryfire
    JSSG B 1
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("jpcpsupershotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronCore", 0, 1, 0, 7)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronWave", 0, 0, 0, 7)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronCoreSuper", 2, 0, 2, 4)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronWave", 2, 0, 2, 4)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronCoreSuper", -2, 0, -2, 4)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronWave", -2, 0, -2, 4)
    JSSG C 1 Bright A_SetPitch (pitch - (1.6 * CallACS("FD_recoil")/65536.0))
    JSSG D 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    JSSG E 1 A_SetPitch (pitch - (0.5 * CallACS("FD_recoil")/65536.0))
    JSSG F 1 A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    JSSG G 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    JSSG G 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    JSSG H 1 A_SetPitch (pitch + (0.75 * CallACS("FD_recoil")/65536.0))
    JSSG I 1 A_SetPitch (pitch + (1.5 * CallACS("FD_recoil")/65536.0))
    JSSG A 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDJPCPShells", 2, "GonnaReload")
    Goto Ready // 10 tics
  GonnaReload:
    TNT1 A 0 A_PlaySound("jpcpsupershotgun/open", 5, 1, 0, 1.25)
    TNT1 A 0 A_PlaySound("jpcpsupershotgun/charge", 6, 1, 0, 1.25)
	JSS2 A 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	JSS2 A 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	JSS2 B 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	JSS2 BC 1
	JSS2 C 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	JSS2 D 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	JSS2 D 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
    JSS2 D 2
	
    JSS2 E 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 E 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 E 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 E 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 E 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 F 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 F 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 F 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 F 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 G 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 G 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 G 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 H 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 H 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 I 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    JSS2 I 1 A_SetPitch(pitch + (0.06 * CallACS("FD_recoil")/65536.0))
    JSS2 I 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    JSS2 I 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JSS2 I 1
    JSS2 JK 2
    TNT1 A 0 A_PlaySound("jpcpsupershotgun/close", 5, 1, 0, 1.25)
	JSS2 L 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	JSS2 L 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	JSS2 M 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
    JSS2 M 1
    JSS2 N 2
    JSS2 O 1
	JSS2 O 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	JSS2 P 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	JSS2 P 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    JSS2 QRA 2 // 49 tics
    JSS2 AAAAA 1 A_Refire // Gun fires at a speed of 59 tics
    Goto Ready
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}
