 
ACTOR FDJPCPRocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPRocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was exposed to the spectrum by %k."
  Decal ""
  Tag "Prism Cannon"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/idle", 7, 0.4, 1, 1.4)
    JROK A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    JROK A 1 A_WeaponReady
    Loop
	
  Select:
    JROK A 1 A_Raise
	"####" "#" 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    "####" "#" 0 A_PlaySound("jpcprocketlauncher/idle", 7, 0.4, 1, 1.4)
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    "####" "#" 0 A_StopSound(7)
    JROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPRocket", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/fire", 7, 1, 0, 0.4)
	JROK ABCDEF 2
	TNT1 A 0 A_GunFlash ("Flash")
	JROK G 2 Bright
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_FireCustomMissile ("FDJPCPPrismProjectile", 0, 1, 0, 2)
	// -3.25
	JROK H 1 Bright A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	JROK H 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	// 3.25
	JROK H 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	JROK I 1 Bright A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	JROK I 1 Bright A_SetPitch(pitch + (0.6 * CallACS("FD_recoil")/65536.0))
	JROK I 1 Bright A_SetPitch(pitch + (0.39 * CallACS("FD_recoil")/65536.0))
	JROK J 1 Bright A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	JROK J 1 Bright A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	JROK K 1 Bright A_SetPitch(pitch + (0.19 * CallACS("FD_recoil")/65536.0))
	JROK K 1 Bright A_SetPitch(pitch + (0.13 * CallACS("FD_recoil")/65536.0))
	JROK A 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	JROK A 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JROK AAAAAAAAAA 1 A_Refire
    Goto Ready
	
  Flash:
	TNT1 A 2 A_Light1
	TNT1 A 3 A_Light2
	TNT1 A 3 A_Light (5)
	TNT1 A 3 A_Light2
	TNT1 A 2 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}

ACTOR FDJPCPPrismProjectile
{
  Radius 4
  Height 4
  Speed 25
  Damage (randompick(100,110,120))
  DamageType "Purifying"
  Species "Player"
  +THRUSPECIES
  +THRUGHOST
  +FORCEXYBILLBOARD
  +BLOODLESSIMPACT
  Decal "BigScorch"
  Projectile
  RenderStyle Add
  Alpha 0.99
  Scale 1.2
  
  var int user_damagetimer;
  var int user_fadetimer;
  var int user_animationframe;
  States
  {
  Spawn:
    TNT1 A 0
    TNT1 A 1 Light("FDPrismCannonGlow")
	TNT1 A 0 A_Jump(256, "SpawnTrail")
	Goto SpawnTrail
  SpawnTrail:
	TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail",  (0.01*momx)/-35.0, -(0.01*momy)/-35.0, 2+(0.01*momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail2", (8 *momx)/-35.0, -(8 *momy)/-35.0, 2+(8 *momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail3", (16 *momx)/-35.0, -(16 *momy)/-35.0, 2+(16 *momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail4", (24 *momx)/-35.0, -(24 *momy)/-35.0, 2+(24 *momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail5", (32 *momx)/-35.0, -(32 *momy)/-35.0, 2+(32 *momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail6", (40 *momx)/-35.0, -(40 *momy)/-35.0, 2+(40 *momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
    TNT1 A 0 A_SpawnItemEx("FDJPCPPrismProjectileTrail7", (48 *momx)/-35.0, -(48 *momy)/-35.0, 2+(48 *momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION)
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "Spawn")
    TNT1 A 0 A_Jump (256, 1,3,6)
	TNT1 AAAAAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE | SPF_RELANG, random(8,16), random(2,3), 0, frandom(-16.0,16.0),frandom(-4.2,4.2),frandom(-4.2,4.2), 0,0,0, 0,0,0, 0.5, -1)
	TNT1 A 0 A_Jump(256, "Spawn")
	Goto Spawn
	
  Death:
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/explosion", CHAN_WEAPON, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/explosion", 7, 0.10, 0, 0.01)
    TNT1 A 0 A_Explode (32, 128, 0, 1, 128)
	TNT1 A 0 A_ChangeFlag ("PAINLESS", 1)
	TNT1 A 0 A_ChangeFlag ("NODAMAGETHRUST", 1)
    TNT1 A 0 A_SetUserVar("user_fadetimer", 20)
    TNT1 A 0 A_SetUserVar("user_damagetimer", 6)
	
  DeathLoop:
    TNT1 A 0 A_RadiusGive ("FDJPCPPrismSlightBlinding", 128, RGF_PLAYERS, 1)
    TNT1 A 0 A_RadiusGive ("FDJPCPPrismBlinding", 96, RGF_PLAYERS, 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "QuakeTime")
    TNT1 A 0 A_Jump (256, 1,3,6,9)
	TNT1 AAAAAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(70,140), random(1,3), frandom(0,360), 0,0,0, frandom(0.0,4.0),frandom(-0.2,0.2),frandom(-4.0,4.0), frandom(-0.2,0.2),frandom(-0.2,0.2),frandom(-0.2,0.2), 0.98, -1)
    TNT1 A 0 A_Jump (256, "QuakeTime")
	Goto QuakeTime
  QuakeTime:
	TNT1 A 0 A_JumpIf(CallACS("FD_quakes") == 0, "FrameChoose")
    TNT1 A 0 A_Quake (1, 2, 0, 128, "")
    TNT1 A 0 A_Jump (256, "FrameChoose")
	Goto FrameChoose
	
  FrameChoose:
    TNT1 A 0 A_JumpIf (user_damagetimer == 1, 3)
    TNT1 A 0 A_SetScale (frandom (1.2,1.3))
    TNT1 A 0 A_Jump (256, 2)
    TNT1 A 0 A_SetScale (frandom (1.8,1.9))
    TNT1 A 0 A_JumpIf (user_animationframe == 1, "FrameChooseB")
    TNT1 A 0 A_JumpIf (user_animationframe == 2, "FrameChooseC")
    TNT1 A 0 A_JumpIf (user_animationframe == 3, "FrameChooseD")
    TNT1 A 0 A_JumpIf (user_animationframe == 4, "FrameChooseE")
    TNT1 A 0 A_JumpIf (user_animationframe == 5, "FrameChooseF")
    TNT1 A 0 A_Jump (256, "FrameChooseA")
	Goto FrameChooseA
  FrameChooseA:
//	TNT1 A 0 A_ChangeFlag ("PAINLESS", 1)
    JRKP A 1 Bright Light("FDPrismGlow") A_SpawnItemEx ("FDJPCPPrismAfterglow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME | SXF_TRANSFERSCALE)
    TNT1 A 0 A_SetUserVar("user_animationframe", 1)
    TNT1 A 0 A_Jump (256, "IncrementTheCrap")
	Goto IncrementTheCrap
  FrameChooseB:
    JRKP B 1 Bright Light("FDPrismGlow") A_SpawnItemEx ("FDJPCPPrismAfterglow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME | SXF_TRANSFERSCALE)
    TNT1 A 0 A_SetUserVar("user_animationframe", 2)
    TNT1 A 0 A_Jump (256, "IncrementTheCrap")
	Goto IncrementTheCrap
  FrameChooseC:
    JRKP C 1 Bright Light("FDPrismGlow") A_SpawnItemEx ("FDJPCPPrismAfterglow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME | SXF_TRANSFERSCALE)
    TNT1 A 0 A_SetUserVar("user_animationframe", 3)
    TNT1 A 0 A_Jump (256, "IncrementTheCrap")
	Goto IncrementTheCrap
  FrameChooseD:
    JRKP D 1 Bright Light("FDPrismGlow") A_SpawnItemEx ("FDJPCPPrismAfterglow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME | SXF_TRANSFERSCALE)
    TNT1 A 0 A_SetUserVar("user_animationframe", 4)
    TNT1 A 0 A_Jump (256, "IncrementTheCrap")
	Goto IncrementTheCrap
  FrameChooseE:
    JRKP E 1 Bright Light("FDPrismGlow") A_SpawnItemEx ("FDJPCPPrismAfterglow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME | SXF_TRANSFERSCALE)
    TNT1 A 0 A_SetUserVar("user_animationframe", 5)
    TNT1 A 0 A_Jump (256, "IncrementTheCrap")
	Goto IncrementTheCrap
  FrameChooseF:
//	TNT1 A 0 A_ChangeFlag ("PAINLESS", 0)
    JRKP F 1 Bright Light("FDPrismGlow") A_SpawnItemEx ("FDJPCPPrismAfterglow", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERSPRITEFRAME | SXF_TRANSFERSCALE)
    TNT1 A 0 A_SetUserVar("user_animationframe", 0)
    TNT1 A 0 A_Jump (256, "IncrementTheCrap")
	Goto IncrementTheCrap
	
  IncrementTheCrap:
    TNT1 A 0 A_SetUserVar("user_damagetimer", user_damagetimer - 1)
    TNT1 A 0 A_JumpIf (user_damagetimer <= 0, "DamageDealer")
    TNT1 A 0 A_Jump (256, "DeathLoop")
	Goto DeathLoop
  DamageDealer:
    TNT1 A 0 A_Explode (8, 128, 0, 1, 128)
    TNT1 A 0 A_SetUserVar("user_fadetimer", user_fadetimer - 1)
    TNT1 A 0 A_SetUserVar("user_damagetimer", 6)
    TNT1 A 0 A_JumpIf (user_fadetimer <= 0, "GottaGoBye")
    TNT1 A 0 A_Jump (256, "DeathLoop")
	Goto DeathLoop
	
  GottaGoBye:
	TNT1 A 0 A_ChangeFlag ("FORCEPAIN", 1)
	TNT1 A 0 A_ChangeFlag ("PAINLESS", 0)
	TNT1 A 0 A_ChangeFlag ("NODAMAGETHRUST", 0)
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/finalexplosion", CHAN_WEAPON, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("jpcprocketlauncher/finalexplosion", 7, 0.10, 0, 0.01)
    TNT1 A 0 A_Explode (10, 128, 0, 1)
    TNT1 A 0 A_RadiusThrust (5000, 128, RTF_AFFECTSOURCE | RTF_NOIMPACTDAMAGE)
    TNT1 A 0 A_SetScale (2.2)
	TNT1 A 0 A_JumpIf(CallACS("FD_quakes") == 0, 3)
    TNT1 A 0 A_Quake (1, 20, 0, 512, "")
    TNT1 A 0 A_Quake (3, 15, 0, 256, "")
    JRK2 A 3 Bright Light("FDPrismBlast1")
    JRK2 B 3 Bright Light("FDPrismBlast2")
    JRK2 C 3 Bright Light("FDPrismBlast3")
    JRK2 D 3 Bright Light("FDPrismBlast4")
    JRK2 E 3 Bright Light("FDPrismBlast5")
	Stop
  }
}
