 
ACTOR FDHellboundChaingun : FDWeapon
{
  Inventory.RestrictedTo "FDHellboundPlayer"
  Inventory.Icon "HBCHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDHellboundBullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%o received a couple of unexpected holes, courtesy of %o's rifle."
  Decal "BulletChip"
  Tag "Zaiger Rifle"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDLevelBeginAnimation", 1, "GonnaReload")
    HCHG A 1 A_WeaponReady
    Loop
  ReadyNoBob:
    HCHG AAAAAAAAAA 1 A_WeaponReady (WRF_NOBOB)
    Goto Ready
	
  Select:
    HCHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
//	TNT1 A 0 A_TakeInventory("FDHellboundZaigerShots", 0)
    HCHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    HCHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, 1)
	Goto Dryfire
//    TNT1 A 0 A_JumpIfInventory ("FDHellboundZaigerShots", 0, "GonnaReload")
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_PlaySound("hellboundchaingun/Fire", 1, 0.65, 0, 0.7)
	TNT1 A 0 A_FireBullets (4.5, 2.0, 1, randompick(22,26), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
//	TNT1 A 0 A_GiveInventory("FDHellboundZaigerShots", 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,5)
	HCHG N 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	HCHG O 1 Bright A_SetPitch (pitch - (0.2  * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(2.6,2.2),-4, FPF_NOAUTOAIM)
	HCHG P 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	HCHG Q 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
  Fire2:
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, 1)
	Goto Dryfire
//    TNT1 A 0 A_JumpIfInventory ("FDHellboundZaigerShots", 0, "GonnaReload")
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_PlaySound("hellboundchaingun/Fire", 1, 0.65, 0, 0.7)
	TNT1 A 0 A_FireBullets (4.5, 2.0, 1, randompick(22,26), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
//	TNT1 A 0 A_GiveInventory("FDHellboundZaigerShots", 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,5)
	HCHG N 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	HCHG O 1 Bright A_SetPitch (pitch - (0.2  * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD556mmSmallCasingSpawner", 0,0,frandom(2.6,2.2),-4, FPF_NOAUTOAIM)
	HCHG P 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	HCHG Q 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_ReFire
	Goto ReadyNoBob
		
  GonnaReload:
	TNT1 A 0 A_PlaySound("hellboundchaingun/out",5,1.0,0,1.2)
	HCHG ABCDEF 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	HCHG G 10
	HCHG HIJ 1
//	TNT1 A 0 A_TakeInventory("FDHellboundZaigerShots", 0)
	HCHG K 1 A_PlaySound("hellboundchaingun/in",5,1.0,0,1.2)
	HCHG L 7
	HCHG DCAAAA 1 A_SetPitch (pitch - (0.15 * CallACS("FD_recoil")/65536.0))
	Goto Ready
	
  Flash:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}
