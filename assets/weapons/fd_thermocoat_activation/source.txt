
ACTOR FDWhitemareChainsaw : FDWeapon
{
  Inventory.RestrictedTo "FDWhitemarePlayer"
  Inventory.Icon "WMCHAINS"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Weapon.SlotPriority 0
  Obituary "%o tried to hug %k at the wrong time."
  Decal ""
  Tag "ThermoCoat Activation"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "ReadyActive")
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatLevel", 1, 1)
	Goto ReadyNoHeat
    WSAW A 1 A_WeaponReady
    Loop
  ReadyNoHeat:
    WSAW A 1 A_WeaponReady (WRF_NOFIRE)
    Goto Ready
  ReadyActive:
    TNT1 A 1 A_WeaponReady (WRF_NOFIRE)
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "ReadyActive")
    "####" "######" 0 A_Lower
    Goto Select
	
  Select:
    WSAW A 1 A_Raise
    "####" "##" 0 A_Raise
    Loop
	
  Deselect:
    WSAW A 1 A_Lower
    "####" "##" 0 A_Lower
    Loop
  DeselectAfter:
    "####" "#" 0 A_Lower
    Loop
	
  Fire:
	TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatLevel", 1, 1)
	Goto ReadyNoHeat
    TNT1 A 0 A_PlaySound("plutfist/swing", 7, 1.0, 0, 1.5)
	 // -0.47
	WSAW B 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
	WSAW B 1 A_SetPitch(pitch - (0.03 * CallACS("FD_recoil")/65536.0))
	WSAW C 1 A_SetPitch(pitch - (0.08 * CallACS("FD_recoil")/65536.0))
	WSAW C 1 A_SetPitch(pitch - (0.12 * CallACS("FD_recoil")/65536.0))
	WSAW D 1 A_SetPitch(pitch - (0.10 * CallACS("FD_recoil")/65536.0))
	WSAW D 1 A_SetPitch(pitch - (0.09 * CallACS("FD_recoil")/65536.0))
	WSAW E 1 A_SetPitch(pitch - (0.03 * CallACS("FD_recoil")/65536.0))
	WSAW E 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
	WSAW E 2
	// 0.27
	WSAW F 1 A_SetPitch(pitch + (0.21 * CallACS("FD_recoil")/65536.0))
	WSAW F 1 A_SetPitch(pitch + (0.06 * CallACS("FD_recoil")/65536.0))
	
    TNT1 A 0 A_PlaySound("btsxchainsaw/on", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("btsxchainsaw/beep", 6, 1.0, 0, 1.5)
	 // -0.11
	WSAW G 1 A_SetPitch(pitch - (0.10 * CallACS("FD_recoil")/65536.0))
	WSAW G 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_SpawnItemEx ("FDWhitemareHeatOverloadFireSpawner", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETTARGET)
	 // 0.38
	WSAW H 1 A_SetPitch(pitch + (0.18 * CallACS("FD_recoil")/65536.0))
	WSAW H 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
	WSAW I 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	WSAW I 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))

actor FDWhitemareHeatOverloadFireSpawner
{
    Radius 6
    Height 6
    +NOINTERACTION
	+NOCLIP

	var int user_flametimer;
	var int user_flameangle;
    States
    {
    Spawn:
		TNT1 A 1
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatLevel", 1, 1, AAPTR_TARGET)
		Stop
		TNT1 A 0 A_GiveInventory ("FDWhitemareHeatActive", 1, AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTSoundSpamPrevention", 1, 3, AAPTR_TARGET) // Stops even more goddamn burn sound spam on enemy death
		TNT1 A 0 A_GiveInventory ("FDWhitemareFireDOTSoundSpamPrevention", 1, AAPTR_TARGET)
		TNT1 A 0 A_PlaySound("whitemare/firedotstart", CHAN_WEAPON, 1.0, 0, 0.9)
		TNT1 A 0 A_PlaySound("whitemare/firedotloop", 7, 1.0, 1, 1.0)
		TNT1 A 0 A_SetUserVar("user_flametimer", 10)
		TNT1 A 0 A_Jump (256, "BurnLoop")
		Goto BurnLoop
		
    BurnLoop:
		TNT1 A 1 Light("FDWhitemareFireDOT") A_GiveInventory ("FDWhitemareHeatOverloadDOTEffects", 1, AAPTR_TARGET)
		TNT1 A 0 A_TakeInventory ("FDWhitemareHeatLevel", 1, 0, AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatLevel", 1, "BurnLoopDamageCheck", AAPTR_TARGET)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
    BurnLoopDamageCheck:
		TNT1 A 0 A_SetUserVar("user_flametimer", user_flametimer-1)
		TNT1 A 0 A_JumpIf (user_flametimer <= 0, "BurnLoopDamage")
		TNT1 A 0 A_Jump (256, "BurnLoopWarpCheck")
		Goto BurnLoopWarpCheck
    BurnLoopWarpCheck:
		TNT1 A 0 A_Warp (AAPTR_TARGET, 0,0,0, 0, WARPF_NOCHECKPOSITION, "BurnLoop", 0.5)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
		
    BurnLoopDamage:
		TNT1 A 0 A_SetUserVar("user_flameangle", 0)
		TNT1 A 0 A_SpawnItemEx ("FDWhitemareHeatOverloadDamage", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
		TNT1 A 0 A_Warp (AAPTR_TARGET, 0,0,0, 0, WARPF_NOCHECKPOSITION, "BurnLoopDamageAmount", 0.5)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
    BurnLoopDamageAmount:
		TNT1 A 0 A_SpawnItemEx ("FDWhitemareHeatOverloadFire", 20,0,0, 12,0,0, user_flameangle, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
		TNT1 A 0 A_SetUserVar("user_flameangle", user_flameangle+15)
		TNT1 A 0 A_JumpIf (user_flameangle >= 359, "BurnLoopDamageEnd")
		TNT1 A 0 A_Jump (256, "BurnLoopDamageAmount")
		Goto BurnLoopDamageAmount
    BurnLoopDamageEnd:
		TNT1 A 0 A_SetUserVar("user_flameangle", 0)
		TNT1 A 0 A_SetUserVar("user_flametimer", 10)
		TNT1 A 0 A_Jump (256, "BurnLoop")
		Goto BurnLoop
		
    BurnEnd:
		TNT1 A 0 A_TakeInventory ("FDWhitemareHeatActive", 1, 0, AAPTR_TARGET)
		TNT1 A 0 A_TakeInventory ("FDWhitemareHeatLevel", 0, 0, AAPTR_TARGET) // Remove all heat level
		TNT1 A 0 A_StopSound(7)
		TNT1 A 10
		Stop
    }
}

actor FDWhitemareHeatOverloadFire

	Goto GenerateDOTTime
  DOT1:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoSmoke", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime

	Goto GenerateDOTTime
  DOT3:
    TNT1 A 0 A_SpawnItemEx ("FDWhitemareFireDOTNoVisuals", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_TRANSFERPOINTERS)
	Goto GenerateDOTTime
  GenerateDOTTime:
	TNT1 AA 0 A_GiveInventory ("FDWhitemareFireDOTActive", 1, AAPTR_TRACER) // 2 seconds guaranteed burn

	Stop
    }

 // The ThermoCoat is fun.
 // Anyway, the main DOT mechanic for Whitemare is here.
 
actor FDWhitemareFireDOT
{
    Radius 6
    Height 6
    +NOINTERACTION
	+NOCLIP

	var int user_dottimer;
	var int user_maxdamage;
    States
    {
    Spawn:
		TNT1 A 1
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, 1, AAPTR_TRACER) // Stops multiple burn sounds when you kill with a shotgun
		Stop
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTSoundSpamPrevention", 1, 3, AAPTR_TRACER) // Stops even more goddamn burn sound spam on enemy death
		TNT1 A 0 A_GiveInventory ("FDWhitemareFireDOTSoundSpamPrevention", 1, AAPTR_TRACER)
		TNT1 A 0 A_PlaySound("whitemare/firedotstart", CHAN_WEAPON, 1.0, 0, 0.9)
		TNT1 A 0 A_CheckFlag ("BOSS", 3, AAPTR_TRACER)
		TNT1 A 0 A_PlaySound("whitemare/firedotloop", 7, 1.0, 1, 1.0)
		TNT1 A 0 A_Jump (256, 2)
		TNT1 A 0 A_PlaySound("whitemare/firedotloop2", 7, 1.0, 1, 0.75)
		TNT1 A 0 A_SetUserVar("user_dottimer", 32)
		TNT1 A 0 A_Jump (256, "BurnLoop")
		Goto BurnLoop
    BurnLoop:
		TNT1 A 0 A_JumpIfHealthLower (1, "BurnEnd", AAPTR_TRACER)
		TNT1 A 0 A_JumpIf (user_maxdamage == 1, "BurnLoopBlue")
		TNT1 A 1 Light("FDWhitemareFireDOT") A_GiveInventory ("FDWhitemareFireDOTEffects", 1, AAPTR_TRACER)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, "BurnLoopDamageCheck", AAPTR_TRACER)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
    BurnLoopBlue: // Generates blue flames at maximum power
		TNT1 A 1 Light("FDWhitemareFireDOTBlue") A_GiveInventory ("FDWhitemareFireDOTBlueEffects", 1, AAPTR_TRACER)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, "BurnLoopDamageCheck", AAPTR_TRACER)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
    BurnLoopDamageCheck:
		TNT1 A 0 A_SetUserVar("user_dottimer", user_dottimer-1)
		TNT1 A 0 A_JumpIf (user_dottimer <= 0, "BurnLoopDamage")
		TNT1 A 0 A_Jump (256, "BurnLoopWarpCheck")
		Goto BurnLoopWarpCheck
    BurnLoopWarpCheck:
		TNT1 A 0 A_Warp (AAPTR_TRACER, 0,0,0, 0, WARPF_NOCHECKPOSITION, "BurnLoop", 0.5)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
		
    BurnLoopDamage:
		TNT1 A 0 A_Warp (AAPTR_TRACER, 0,0,0, 0, WARPF_NOCHECKPOSITION, "BurnLoopDamageAmount", 0.5)
		TNT1 A 0 A_Jump (256, "BurnEnd")
		Goto BurnEnd
    BurnLoopDamageAmount:
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTLevel", 3500, "BurnLoopDamage5", AAPTR_TRACER) // At least 100 seconds of burn
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTLevel", 2625, "BurnLoopDamage4", AAPTR_TRACER) // At least 75 seconds of burn
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTLevel", 1750, "BurnLoopDamage3", AAPTR_TRACER) // At least 50 seconds of burn
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTLevel", 875, "BurnLoopDamage2", AAPTR_TRACER) // At least 25 seconds of burn
		TNT1 A 0 A_Jump (256, "BurnLoopDamage1") // At least 0 seconds of burn
		Goto BurnLoopDamage1
    BurnLoopDamage1:
//		TNT1 A 0 A_Log ("5 Damage")
		TNT1 A 0 A_CustomMissile ("FDWhitemareFireDOTDamage1", 0,0, 0, CMF_AIMDIRECTION | CMF_TRACKOWNER, 0, AAPTR_TRACER)
		TNT1 A 0 A_SetUserVar("user_maxdamage", 0) // Resets fire color back to normal
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "BurnLoopDamageEnd", AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, 2, AAPTR_TARGET)
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
		TNT1 A 0 A_GiveInventory ("FDWhitemareHeatLevel", 5, AAPTR_TARGET) // Adds 5 to current Player Heat
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
    BurnLoopDamage2:
//		TNT1 A 0 A_Log ("10 Damage")
		TNT1 A 0 A_CustomMissile ("FDWhitemareFireDOTDamage2", 0,0, 0, CMF_AIMDIRECTION | CMF_TRACKOWNER, 0, AAPTR_TRACER)
		TNT1 A 0 A_SetUserVar("user_maxdamage", 0) // Resets fire color back to normal
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "BurnLoopDamageEnd", AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, 2, AAPTR_TARGET)
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
		TNT1 A 0 A_GiveInventory ("FDWhitemareHeatLevel", 10, AAPTR_TARGET) // Adds 10 to current Player Heat
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
    BurnLoopDamage3:
//		TNT1 A 0 A_Log ("20 Damage")
		TNT1 A 0 A_CustomMissile ("FDWhitemareFireDOTDamage3", 0,0, 0, CMF_AIMDIRECTION | CMF_TRACKOWNER, 0, AAPTR_TRACER)
		TNT1 A 0 A_SetUserVar("user_maxdamage", 0) // Resets fire color back to normal
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "BurnLoopDamageEnd", AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, 2, AAPTR_TARGET)
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
		TNT1 A 0 A_GiveInventory ("FDWhitemareHeatLevel", 20, AAPTR_TARGET) // Adds 20 to current Player Heat
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
    BurnLoopDamage4:
//		TNT1 A 0 A_Log ("40 Damage")
		TNT1 A 0 A_CustomMissile ("FDWhitemareFireDOTDamage4", 0,0, 0, CMF_AIMDIRECTION | CMF_TRACKOWNER, 0, AAPTR_TRACER)
		TNT1 A 0 A_SetUserVar("user_maxdamage", 0) // Resets fire color back to normal
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "BurnLoopDamageEnd", AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, 2, AAPTR_TARGET)
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
		TNT1 A 0 A_GiveInventory ("FDWhitemareHeatLevel", 40, AAPTR_TARGET) // Adds 40 to current Player Heat
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
    BurnLoopDamage5:
//		TNT1 A 0 A_Log ("80 Damage")
		TNT1 A 0 A_CustomMissile ("FDWhitemareFireDOTDamage5", 0,0, 0, CMF_AIMDIRECTION | CMF_TRACKOWNER, 0, AAPTR_TRACER)
		TNT1 A 0 A_SetUserVar("user_maxdamage", 1) // Turns the fire blue to indicate it's max level damage
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareHeatActive", 1, "BurnLoopDamageEnd", AAPTR_TARGET)
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareChainsaw", 1, 2, AAPTR_TARGET)
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
		TNT1 A 0 A_GiveInventory ("FDWhitemareHeatLevel", 80, AAPTR_TARGET) // Adds 80 to current Player Heat
		TNT1 A 0 A_Jump (256, "BurnLoopDamageEnd")
		Goto BurnLoopDamageEnd
    BurnLoopDamageEnd:
		TNT1 A 0 A_SetUserVar("user_dottimer", 32)
		TNT1 A 0 A_Jump (256, "BurnLoop")
		Goto BurnLoop
		
    BurnEnd: // non-damaging fire when either the powerup runs out, or the enemy dies
		TNT1 A 0 A_SetUserVar("user_dottimer", random(120,280)) // repurpose this for death fire
		TNT1 A 0 A_TakeInventory ("FDWhitemareFireDOTActive", 1, 0, AAPTR_TRACER) // sometimes powerup remains when an enemy is dead
		TNT1 A 0 A_StopSound(7)
		TNT1 A 0 A_Jump (256, "BurnEndLoop")
		Goto BurnEndLoop
    BurnEndLoop:
		TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireDOTActive", 1, "BurnCease", AAPTR_TRACER) // If a new fire is applied, stop the smouldering
		TNT1 A 0 A_JumpIf (user_maxdamage == 1, "BurnEndLoopBlue")
		TNT1 A 1 Light("FDWhitemareFireDOTEnd") A_GiveInventory ("FDWhitemareFireDOTEndEffects", 1, AAPTR_TRACER)
		TNT1 A 0 A_Jump (256, "BurnEndLoopEnd")
		Goto BurnEndLoopEnd
    BurnEndLoopBlue:
		TNT1 A 1 Light("FDWhitemareFireDOTBlueEnd") A_GiveInventory ("FDWhitemareFireDOTBlueEndEffects", 1, AAPTR_TRACER)
		TNT1 A 0 A_Jump (256, "BurnEndLoopEnd")
		Goto BurnEndLoopEnd
    BurnEndLoopEnd:
		TNT1 A 0 A_SetUserVar("user_dottimer", user_dottimer-1)
		TNT1 A 0 A_JumpIf (user_dottimer <= 0, "BurnCease")
		TNT1 A 0 A_Warp (AAPTR_TRACER, 0,0,0, 0, WARPF_NOCHECKPOSITION, "BurnEndLoop", 0.5)
		TNT1 A 0 A_Jump (256, "BurnCease")
		Goto BurnCease
		
    BurnCease:
		TNT1 A 0
		Stop
    }
}
