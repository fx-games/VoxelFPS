actor BTConnector : Weapon
{
Inventory.PickupMessage "Picked up the BT51 'Connector'."
weapon.ammogive1 0
weapon.ammogive2 14
//weapon.ammotype "Clip" //Added these in case someone wants to disable reloading altogether.
//weapon.ammogive 13 //Backdash the three AmmoType sections and the +AMMO_CHECKBOTH flags below.
weapon.ammotype1 "BTLoaded"
weapon.ammotype2 "Clip"
weapon.ammouse1 1
weapon.ammouse2 1
Weapon.KickBack 200
Weapon.SlotNumber 2
Obituary "%o was silenced by %k's Blacktail Connector."
Tag "Blacktail 51 'Connector'"
+AMMO_CHECKBOTH
+NOAUTOFIRE
+WEAPON.NOALERT
states
{
//Remove all A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); to disable the red dot.
//Left the original sprites in the BlacktailCovert folder.
Spawn:
SOCP A -1
stop
Ready:
SOCO A 1 { A_WeaponReady(WRF_ALLOWRELOAD); A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ);} //Remove WRF_ALLOWRELOAD flag to remove manual reloading.
SOCO A 0 A_JumpIfInventory("ScopedIn",1,"ScopedReady")
Loop
Select:
SOCO EE 0 A_Raise
SOCO E 1 A_Raise
loop
Deselect:
SOCO EE 0 A_Lower
SOCO E 1 A_Lower
loop
Fire:
SOCO A 0 A_JumpIfInventory("ScopedIn",1,"ScopedFire")
NULL A 0 A_JumpIf(CountInv("BTLoaded") == 0,"Reload") //Backdash this to disable the reload check
SOCO A 0 { A_PlayWeaponSound("SOCOFIR"); A_FireBullets(1.5,1.5,1,7,"BTPuff");}
//NULL A 0 A_Quake(2, 4, 0, 48576, "None") //Remove the backdashes to enable visual recoil.
SOCF A 1 BRIGHT { A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
SOCO BB 1 { A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
NULL A 0 A_JumpIf(CountInv("BTLoaded") == 0,"NoRefire")
SOCO CDAAA 1 { A_WeaponReady(WRF_NOBOB); A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
SOCO A 0 A_Refire
Goto Ready
NoRefire:
SOCO CDAAA 1 { A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
SOCO A 0 A_Refire
Goto Ready

//All scope-related parts are here.
//To remove it just delete the ScopedIn inventory checks. and this whole section.

AltFire:
SOCO A 0 A_JumpIfInventory("ScopedIn",1,"Unscope")
Goto Scope
Scope:
NULL A 0 A_PlaySoundEX("SOCOZO1",CHAN_6)
SOCO A 0 A_SetBlend("000000",1.2,10)
SCOP A 0 A_GiveInventory("ScopedIn",1)
SCOP A 1 A_ZoomFactor(4.0)
SCOP C 1 A_ZoomFactor(6.0)
Goto ScopedReady
ScopedReady:
SCOP D 1 { A_ZoomFactor(10.0); A_WeaponReady(WRF_ALLOWRELOAD|WRF_NOSWITCH); A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ);}
Loop
ScopedFire:
NULL A 0 A_JumpIf(CountInv("BTLoaded") == 0,"ScopedReload") //Backdash this to disable the reload check
SCOP A 0 { A_PlayWeaponSound("SOCOFIR"); A_FireBullets(1.5,1.5,1,7,"BTPuff");}
//NULL A 0 A_Quake(2, 4, 0, 48576, "None") //Remove the backdashes to enable visual recoil.
SCOP A 1 BRIGHT { A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
SCOP B 1 { A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
SCOP CDDDDDD 1 { A_WeaponReady(WRF_NOBOB); A_FireBullets(0,0,1,0,"RedDot",FBF_NOFLASH|FBF_NORANDOMPUFFZ); }
SOCO A 0 A_Refire
Goto ScopedReady
Unscope:
NULL A 0 A_PlaySoundEX("SOCOZO2",CHAN_6)
SOCO A 0 A_TakeInventory("ScopedIn",99)
SCOP E 1 A_ZoomFactor(8.0)
SCOP F 1 A_ZoomFactor(4.0)
SCOP G 1 A_ZoomFactor(1.0)
SOCO A 0 A_SetBlend("000000",1.2,5)
Goto Ready
ScopedReload:
NULL A 0 A_JumpIfInventory("BTLoaded", 13, 2)
NULL A 0 A_JumpIfInventory("Clip", 1, "UnscopeReload")
SCOP D 1 A_WeaponReady
Goto ScopedReady
UnscopeReload:
NULL A 0 A_PlaySoundEX("SOCOZO2",CHAN_6)
SCOP E 1 A_ZoomFactor(8.0)
SCOP F 1 A_ZoomFactor(4.0)
SCOP G 1 A_ZoomFactor(1.0)
SOCO A 0 A_SetBlend("000000",1.2,5)
Goto ReloadWork

//

Reload:
SOCO A 0 A_JumpIfInventory("ScopedIn",1,"ScopedReload")
NULL A 0 A_JumpIfInventory("BTLoaded", 14, 2)
NULL A 0 A_JumpIfInventory("Clip", 1, "ReloadWork")
SOCO A 1 A_WeaponReady
Goto Ready
ReloadWork:
SOCO E 1
SOCR A 1
NULL A 0 A_PlaySoundEX("SOCOOPN",CHAN_6)
SOCR BC 2
SOCR C 4
SOCR D 1
NULL A 0 A_PlaySoundEX("SOCOCLS",CHAN_6)
SOCR EF 1
ReloadLoop:
NULL A 0 A_TakeInventory("Clip", 1, TIF_NOTAKEINFINITE)
NULL A 0 A_GiveInventory("BTLoaded", 1)
NULL A 0 A_JumpIfInventory("BTLoaded", 14, "ReloadFinish")
NULL A 0 A_JumpIfInventory("Clip", 1, "ReloadLoop")
Goto ReloadFinish
ReloadFinish:
SOCR G 2
SOCR HI 2
SOCR CB 2
SOCR A 1
SOCO E 1
SOCO A 0 A_JumpIfInventory("ScopedIn",1,"ScopeFinish")
NULL A 0 A_Refire("Fire")
Goto Ready
ScopeFinish:
SOCO A 2
SOCO A 2 A_ZoomFactor(2.0)
SOCO A 2 A_ZoomFactor(4.0)
NULL A 0 A_PlaySoundEX("SOCOZO1",CHAN_6)
SOCO A 0 A_SetBlend("000000",1.2,5)
SCOP A 0 A_GiveInventory("ScopedIn",1)
SCOP A 1 A_ZoomFactor(6.0)
SCOP C 1 A_ZoomFactor(8.0)
Goto ScopedReady
}
}
