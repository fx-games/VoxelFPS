 
ACTOR FDHellboundBFG9000 : FDBFGWeapon
{
  Inventory.RestrictedTo "FDHellboundPlayer"
  Inventory.Icon "HBBFG900"
  Weapon.UpSound ""
  Weapon.AmmoType "FDHellboundBFGCharge"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 7
  Obituary "%k felt a small explosion wasn't good enough for %o."
  Decal ""
  Tag "Totenheim"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("hellboundbfg9000/idle", 7, 0.4, 1, 1.4)
    HBFG A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    HBFG A 1 A_WeaponReady
    Loop
	
  Select:
    HBFG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    "####" "#" 0 A_StopSound(7)
  DeselectLoop:
    HBFG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    HBFG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBFGCharge", 1, "ChargeStart")
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDHellboundCell", 100, "ChargeStart")
	Goto Dryfire
	
  ChargeStart:
    TNT1 A 0 A_PlaySound("hellboundbfg9000/charge", 7, 1, 0, 0.8)
    TNT1 A 0 A_AlertMonsters (random(256,400))
	HBFG BCDEFGHI 2
	HBFG A 1
    TNT1 A 0 A_PlaySound("hellboundbfg9000/load", 5, 1, 0, 1.2)
	HBFG JKLMNOPR 2
	HBFG A 4
    TNT1 A 0 A_GunFlash ("Flash")
	HBFG S 3 Bright
	TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("hellboundbfg9000/fire", CHAN_WEAPON, 1, 0, 0.8)
	TNT1 A 0 A_PlaySound("hellboundbfg9000/firemain", 5, 1, 0, 1.2)
	TNT1 A 0 A_Quake(6, 20, 0, 512, "")
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_TakeInventory ("FDHellboundBFGCharge", 1, TIF_NOTAKEINFINITE)
	Goto NOOKULAAR
    TNT1 A 0 A_TakeInventory ("FDHellboundCell", 100, TIF_NOTAKEINFINITE)
	Goto NOOKULAAR
  NOOKULAAR:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 0, "NOOKULAARY")
	TNT1 A 0 A_FireCustomMissile ("FDHellboundTotenheimMissile", 0, 0)
	Goto THEBOOMBOIHASBEENDEPLOYED
  NOOKULAARY:
	TNT1 A 0 A_FireCustomMissile ("FDHellboundSimpleTotenheimMissile", 0, 0)
	Goto THEBOOMBOIHASBEENDEPLOYED
  THEBOOMBOIHASBEENDEPLOYED:
	// -3.25
	HBFG T 1 Bright A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	HBFG T 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	// 3.25
	HBFG T 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	HBFG U 1 Bright A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	HBFG U 1 Bright A_SetPitch(pitch + (0.7 * CallACS("FD_recoil")/65536.0))
	HBFG U 1 Bright A_SetPitch(pitch + (0.43 * CallACS("FD_recoil")/65536.0))
	HBFG V 1 Bright A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	HBFG V 1 Bright A_SetPitch(pitch + (0.29 * CallACS("FD_recoil")/65536.0))
	HBFG V 1 Bright A_SetPitch(pitch + (0.17 * CallACS("FD_recoil")/65536.0))
	HBFG W 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	HBFG W 2
	HBFG X 3
	HBFG A 3
	HBFG A 11 A_ReFire
    Goto Ready
	
  Flash:
	TNT1 A 3 A_Light2
	TNT1 A 3 A_Light (5)
	TNT1 A 3 A_Light2
	TNT1 A 3 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}
