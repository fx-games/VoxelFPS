ACTOR FDBTSXBFG9000 : FDBFGWeapon
{
  Inventory.RestrictedTo "FDBTSXPlayer"
  Inventory.Icon "BXBFG900"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXBFGCharge"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 7
  Obituary "%o lost the spark of life after being zapped by %k."
  Decal ""
  Tag "X-Spark Cannon"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("btsxbfg9000/idle", 7, 0.4, 1, 1.4)
    XBFG A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    XBFG A 1 A_WeaponReady
    Loop
	
  Select:
    XBFG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    "####" "#" 0 A_StopSound(7)
  DeselectLoop:
    XBFG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    XBFG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXBFGCharge", 1, "ChargeStart")
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 250, "ChargeStart")
	Goto Dryfire
  QuickFire:
    TNT1 A 0 A_PlaySound("btsxbfg9000/quickfire", 7, 1, 0, 0.8)
	Goto ChargeDone
	
  ChargeStart:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "QuickFire")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "QuickFire")
    TNT1 A 0 A_PlaySound("btsxbfg9000/charge", 7, 1, 0, 0.8)
    TNT1 A 0 A_AlertMonsters (random(256,400))
	XBFG A 15
    TNT1 A 0 A_GunFlash ("FlashCharge", GFF_NOEXTCHANGE)
	XBFG BCDEF 1 Bright
	XBFG A 12
    TNT1 A 0 A_GunFlash ("FlashCharge", GFF_NOEXTCHANGE)
	XBFG BCDEF 1 Bright
	XBFG A 8
    TNT1 A 0 A_GunFlash ("FlashCharge", GFF_NOEXTCHANGE)
	XBFG BCDEF 1 Bright
	XBFG A 5
    TNT1 A 0 A_GunFlash ("FlashCharge", GFF_NOEXTCHANGE)
	XBFG BCDEF 1 Bright
    TNT1 A 0 A_GunFlash ("FlashCharge", GFF_NOEXTCHANGE)
	XBFG BCDEF 1 Bright
  ChargeDone:
    TNT1 A 0 A_GunFlash ("Flash")
	XBFG G 2 Bright
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_TakeInventory ("FDBTSXBFGCharge", 1, TIF_NOTAKEINFINITE)
	Goto UNLEASHTHESPARK
    TNT1 A 0 A_TakeInventory ("FDBTSXCell", 250, TIF_NOTAKEINFINITE)
	Goto UNLEASHTHESPARK
  UNLEASHTHESPARK:
	TNT1 A 0 A_FireCustomMissile ("FDBTSXBFGBall", 0, 0, 0, 6)
	// -3.25
	XBFG H 1 Bright A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	XBFG H 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	// 3.25
	XBFG I 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	XBFG I 1 Bright A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	XBFG J 1 Bright A_SetPitch(pitch + (0.7 * CallACS("FD_recoil")/65536.0))
	XBFG J 1 Bright A_SetPitch(pitch + (0.43 * CallACS("FD_recoil")/65536.0))
	XBFG K 1 Bright A_SetPitch(pitch + (0.30 * CallACS("FD_recoil")/65536.0))
	XBFG K 1 Bright A_SetPitch(pitch + (0.29 * CallACS("FD_recoil")/65536.0))
	XBFG A 1 A_SetPitch(pitch + (0.18 * CallACS("FD_recoil")/65536.0))
	XBFG A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	XBFG AAAAAAAAAA 2 A_ReFire
    Goto Ready
	
  Flash:
	TNT1 A 2 A_Light2
	TNT1 A 2 A_Light (10)
	TNT1 A 2 A_Light (5)
	TNT1 A 2 A_Light (3)
	TNT1 A 2 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  FlashCharge:
	TNT1 A 1 A_Light1
	TNT1 A 1 A_Light2
	TNT1 A 1 A_Light1
	TNT1 A 1 A_Light2
	TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}
