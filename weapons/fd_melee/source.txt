 
ACTOR FDAlienVendettaFist : FDWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVFIST"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Obituary "%o was taken down by %k's lightning fast strikes."
  Decal ""
  +WEAPON.MELEEWEAPON
  Tag "Melee"
  States
  {
  Ready:
    TNT1 A 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 1 A_Raise
    "####" "##" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaFistComboCounter", 0)
  DeselectLoop:
    TNT1 A 1 A_Lower
    "####" "##" 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaFistComboCounter", 0, "LeftHook")
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaFistComboCounter", 1, "RightJab")
	Goto LeftJab
	
  LeftJab:
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaFistComboCounter", 1)
    TNT1 A 0 A_PlaySound("plutfist/swing", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("melee/whoosh", 7, 1.0, 0, 1.5)
	VFIS AB 1
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "BerserkHit")
    TNT1 A 0 A_CustomPunch (4*random(4,6), 1, CPF_NORANDOMPUFFZ, "FDAlienVendettaJabPuff", 80)
    Goto LeftJabFinish
  BerserkHit:
    TNT1 A 0 A_CustomPunch (4*random(20,30), 1, CPF_NORANDOMPUFFZ, "FDAlienVendettaJabPuff", 80)
    Goto LeftJabFinish
  LeftJabFinish:
	VFIS C 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
	VFIS D 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaJabHit", 1, "LeftJabHitFinish")
	VFIS D 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	VFIS E 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	VFIS E 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VFIS F 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1
    Goto Ready
  LeftJabHitFinish:
	VFIS C 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	VFIS D 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	VFIS D 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VFIS E 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	VFIS EFF 1
	TNT1 A 1
    Goto Ready

  RightJab:
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaFistComboCounter", 1)
    TNT1 A 0 A_PlaySound("plutfist/swing", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("melee/whoosh", 7, 1.0, 0, 1.5)
	VFIS GH 1
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "RightJabBerserkHit")
    TNT1 A 0 A_CustomPunch (4*random(4,6), 1, CPF_NORANDOMPUFFZ, "FDFistPuff", 80)
    Goto RightJabFinish
  RightJabBerserkHit:
    TNT1 A 0 A_CustomPunch (4*random(20,30), 1, CPF_NORANDOMPUFFZ, "FDFistPuff", 80)
    Goto RightJabFinish
  RightJabFinish:
	VFIS I 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
	VFIS J 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaJabHit", 1, "RightJabHitFinish")
	VFIS J 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	VFIS K 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	VFIS K 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VFIS L 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1
    Goto Ready
  RightJabHitFinish:
	VFIS I 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	VFIS J 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	VFIS J 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VFIS K 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	VFIS KLL 1
	TNT1 A 1
    Goto Ready
	
  LeftHook:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaFistComboCounter", 0)
    TNT1 A 0 A_PlaySound("plutfist/swing", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("melee/whoosh", 7, 1.0, 0, 1.5)
	VFIS MNO 1
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "LeftHookBerserkHit")
    TNT1 A 0 A_CustomPunch (12*random(4,6), 1, CPF_NORANDOMPUFFZ, "FDAlienVendettaHookPuff", 80)
    Goto LeftHookFinish
  LeftHookBerserkHit:
    TNT1 A 0 A_CustomPunch (12*random(20,30), 1, CPF_NORANDOMPUFFZ, "FDAlienVendettaBerserkHookPuff", 80)
    Goto LeftHookFinish
  LeftHookFinish:
	VFIS P 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
	VFIS Q 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	VFIS R 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaJabHit", 1, "LeftHookHitFinish")
	VFIS S 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	VFIS T 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VFIS U 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	TNT1 A 6
    Goto Ready
  LeftHookHitFinish:
	VFIS P 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	VFIS Q 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	VFIS R 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	VFIS SSTTUU 1
	TNT1 A 10
    Goto Ready
  }
}

Actor FDAlienVendettaFistComboCounter : FDCounter
{
inventory.maxamount 2
}
