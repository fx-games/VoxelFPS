 
ACTOR FDJPCPPistol : FDWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPPistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%k found the most silent way to eliminate %o."
  Decal "BulletChip"
  Tag "Silenced Pistol"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReadyUseAmmo")
    JPIS A 1 A_WeaponReady
    Loop
  ReadyFull:
    JPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Goto Ready
	
  ReadyUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBullets", 1, 1)
	Goto ReadyFull
    JPIS A 1 A_WeaponReady
    Goto Ready
	
  Select:
    JPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    JPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JPIS A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready

// aaaagh
// next person who asks for yet another pistol cvar gets fucking shanked

  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto FireNoAmmoCheckDone
  FireNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire")
    Goto FireNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBullets", 1, "FireNoAmmoCheckDone")
    Goto Dryfire
	
  FireNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0, 0, 1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0, 0, 1, randompick(5,6), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDJPCPBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireComplete
	
  FireNoAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters (random(100,128))
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("jpcppistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("jpcppistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoMagConsumption") // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (0, 0, 1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FireComplete:
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD9mmCasingSpawner", 0,0,frandom(0.5,0.6),-1, FPF_NOAUTOAIM)
    JPIS B 1 Bright A_SetPitch (pitch - (0.6 * CallACS("FD_recoil")/65536.0))
    JPIS B 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0)) // 0.7
    JPIS C 1 A_SetPitch (pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    JPIS C 1 A_SetPitch (pitch + (0.04 * CallACS("FD_recoil")/65536.0))
    JPIS D 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    JPIS D 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    JPIS E 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    JPIS E 1 A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    JPIS A 1 A_SetPitch (pitch + (0.04 * CallACS("FD_recoil")/65536.0))
    JPIS A 1 A_SetPitch (pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_WeaponReady (WRF_NOPRIMARY | WRF_NOBOB)
	Goto Ready
	
  AltFire:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 0, "Reloadery")
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBullets", 1, "Reloadery")
	Goto DryFire // Normally we'd return to Ready, but we disable altfire if full anyway, so we can go to dryfire without problems
  Reloadery:
    TNT1 A 0 A_PlaySound("jpcppistol/reloadopen", 6, 0.3, 0, 1.25)
    JPIS FGH 2
    JPIS IJKL 2
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReloadUseAmmo")
    TNT1 A 0 A_GiveInventory ("FDJPCPPistolAmmo", 10)
	Goto ReloadFinish
  ReloadUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDJPCPPistolAmmo", 0, "ReloadFinish")
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBullets", 1, 1)
	Goto ReloadFinish
    TNT1 A 0 A_GiveInventory ("FDJPCPPistolAmmo", 2) // Intentional because of the rarity of needles compared to vanilla bullets
    TNT1 A 0 A_TakeInventory ("FDJPCPBullets", 1)
	Goto ReloadUseAmmo
  ReloadFinish:
    TNT1 A 0 A_PlaySound("jpcppistol/reloadin", 5, 0.8, 0, 1.25)
    JPIS MN 2
    TNT1 A 0 A_PlaySound("jpcppistol/reloadclose", 6, 0.3, 0, 1.25)
    JPIS OPQRA 2
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light0
    TNT1 A 0 A_Light0
	Goto LightDone

  }
}
