 
ACTOR FDAlienVendettaBFG9000 : FDBFGWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVBFG900"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAlienVendettaBFGCharge"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 7
  Obituary "%o was marked for vengeance by %k."
  Decal ""
  Tag "The Vendetta Machine"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, "ReadyNoSound")
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/idle", 6, 0.7, 1, 1.4)
    VBFG A 1 A_WeaponReady
    Loop
  ReadyNoSound:
    VBFG A 1 A_WeaponReady
    Loop

  Select:
    VBFG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    "####" "#" 0 A_TakeInventory ("FDAlienVendettaBFGTargetActive", 1) // Breaks the lock-on
    "####" "#" 0 A_StopSound(6)
  DeselectLoop:
    VBFG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop

  Dryfire:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaBFGTargetActive", 1) // Breaks the lock-on
    VBFG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
  ScanStop:
    VBFG A 12 A_PlaySound("alienvendettabfg9000/scandeactivate", 7, 1, 0, 1.6)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, 2)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGCharge", 1, "StartAmmoCheckDone")
	Goto Dryfire
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGSplitCell", 1, "StartAmmoCheckDone")
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaCell", 1, "SplitCell")
	Goto Dryfire
	
  SplitCell:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaCell", 1, TIF_NOTAKEINFINITE)
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaBFGSplitCell", 3)
  StartAmmoCheckDone:
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/scanactivate", 6, 1, 0, 1.6)
    TNT1 A 0 A_AlertMonsters (random(64,128))
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/scanloop", 7, 1, 1, 1.6)
	Goto VendettaScan
  VendettaScan:
//    TNT1 A 0 A_RailAttack (0, 0, 0, "None", "Red", RGF_SILENT | RGF_NOPIERCING | RGF_FULLBRIGHT, 0, "FDHarmlessPuff", 0, 0, 1024, 1, 0.1, 0, "", -0.4) 
    VBFG B 1 A_FireBullets (0,0, -1, 0, "FDAlienVendettaBFGTargeter", FBF_NOFLASH | FBF_NORANDOMPUFFZ, 1024)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGTargetActive", 1, "VendettaActivation") // Lock-on detected
    VBFG C 1 A_FireBullets (0,0, -1, 0, "FDAlienVendettaBFGTargeter", FBF_NOFLASH | FBF_NORANDOMPUFFZ, 1024)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGTargetActive", 1, "VendettaActivation") // Lock-on detected
    VBFG D 1 A_FireBullets (0,0, -1, 0, "FDAlienVendettaBFGTargeter", FBF_NOFLASH | FBF_NORANDOMPUFFZ, 1024)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGTargetActive", 1, "VendettaActivation") // Lock-on detected
    VBFG E 1 A_FireBullets (0,0, -1, 0, "FDAlienVendettaBFGTargeter", FBF_NOFLASH | FBF_NORANDOMPUFFZ, 1024)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGTargetActive", 1, "VendettaActivation") // Lock-on detected
    TNT1 A 0 A_Refire ("VendettaScan")
	Goto ScanStop
	
  VendettaActivation:
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/scandeactivate", 6, 1, 0, 1.6)
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/lockon", 5, 1, 0, 0.8)
    TNT1 A 0 A_AlertMonsters (random(256,400))
	VBFG FG 3
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/activate", 7, 1, 0, 1.2)
	VBFG H 14
  VendettaFiringLoop:
	TNT1 A 0 A_JumpIf(CallACS("FD_bfgammosystem") == 4, "VendettaFiringLoopCellSystem")
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGCharge", 1, 1)
	Goto VendettaFiringEnd
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaBFGCharge", 1, TIF_NOTAKEINFINITE)
	Goto VENDETTATIME
	
  VendettaFiringLoopCellSystem:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaBFGSplitCell", 1, "VendettaFiringLoopCellSystemAmmo")
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaCell", 1, "VendettaFiringLoopCellSystemSplitCell")
	Goto VendettaFiringEnd
  VendettaFiringLoopCellSystemSplitCell:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaCell", 1, TIF_NOTAKEINFINITE)
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaBFGSplitCell", 5)
	Goto VENDETTATIME
  VendettaFiringLoopCellSystemAmmo:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaBFGSplitCell", 1, TIF_NOTAKEINFINITE)
	Goto VENDETTATIME
	
  VENDETTATIME:
	TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/fire", CHAN_WEAPON, 0.9, 0, 0.8)
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/fire2", 6, 1, 0, 0.8)
//	TNT1 A 0 A_FireCustomMissile ("FDAlienVendettaBFGBall", 0, 0)
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaBFGFiring", 1) // OH YES, THIS GUN DOES NOT ACTUALLY SHOOT
	// FEAR YOL'S ABSOLUTE CRIMES, FOR THEY ARE WITHOUT NUMBER
	// I AM KNOWN AS KING CRIMERA HAXFUCKER, AND YOU WILL FEAR ME
  FiringComplete:
	VBFG I 1 Bright A_SetPitch(pitch - (0.75 * CallACS("FD_recoil")/65536.0))
	VBFG J 1 Bright A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	VBFG K 1 Bright A_SetPitch(pitch + (0.65 * CallACS("FD_recoil")/65536.0))
	VBFG L 1 Bright A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	VBFG M 1
	VBFG H 1
	TNT1 A 0 A_ReFire ("VendettaFiringLoop")
  VendettaFiringEnd:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaBFGTargetActive", 1) // Breaks the lock-on
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/fireend", 5, 1, 0, 0.8)
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    TNT1 A 0 A_PlaySound("alienvendettabfg9000/idle", 6, 0.5, 1, 1.4)
	VBFG HGFGHGFGH 1
	VBFG A 3
    Goto Ready
	
  Flash:
	TNT1 A 1 A_Light2
	TNT1 A 1 A_Light (5)
	TNT1 A 1 A_Light2
	TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}
