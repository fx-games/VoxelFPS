 
ACTOR FDJPCPShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPSHOTGU"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPShells"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 3
  Obituary "%o was pulsed apart by %k."
  Decal ""
  Tag "Neutron Pulser"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPShells", 1, "ReadyAmmo")
    JSHT A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    JSHT A 1 A_WeaponReady
    Loop
	
  Select:
    JSHT A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    JSHT A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JSHT A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPShells", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("jpcpshotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronCore", 0, 1, 0, 6)
    TNT1 A 0 A_FireCustomMissile ("FDJPCPNeutronWave", 0, 0, 0, 6)
    JSHT B 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    JSHT C 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    JSHT D 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    JSHT E 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    JSHT A 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    JSHT A 1 A_SetPitch (pitch + (1.05 * CallACS("FD_recoil")/65536.0))
    JSHT A 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
  GonnaReload:
    JSHT F 1
    JSHT G 1
	JSHT H 2
	JSHT I 3
    TNT1 A 0 A_PlaySound("jpcpshotgun/pumpdown", 5, 1, 0, 1.25)
	JSHT J 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	JSHT K 1 A_SetPitch(pitch + (0.23 * CallACS("FD_recoil")/65536.0))
	JSHT K 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	JSHT K 1 A_SetPitch(pitch - (0.06 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_PlaySound("jpcpshotgun/pumpup", 6, 1, 0, 1.25)
	JSHT L 1 A_SetPitch(pitch - (0.28 * CallACS("FD_recoil")/65536.0))
	JSHT L 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	JSHT MNIHGF 2
	JSHT A 5
	JSHT AAAAAAA 1 A_Refire // Gun fires at a speed of 37 tics
    Goto Ready
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

ACTOR FDJPCPNeutronWave
{
  Radius 4
  Height 4
  Speed 26
  Damage (1)
  DamageType "Plasma"
  Species "Player"
  +THRUSPECIES
  +THRUGHOST
  +FORCEXYBILLBOARD
  +BLOODLESSIMPACT
  +RIPPER
  Decal ""
  Projectile
  RenderStyle Add
  Alpha 0.99
  Scale 1.5
  
  var int user_gonnafadenow;
  States
  {
  Spawn:
    TNT1 A 0
    TNT1 A 0 A_SetUserVar("user_gonnafadenow", 25)
  SpawnLoop:
    TNT1 A 1 Light("FDNeutronBlueGlow")
    TNT1 A 0 A_SetUserVar("user_gonnafadenow", user_gonnafadenow - 1)
    TNT1 A 0 A_JumpIf (user_gonnafadenow <= 0, "Dissipate")
	TNT1 A 0 A_Jump(256, "SpawnTrail")
	Goto SpawnTrail
  SpawnTrail:
	TNT1 A 0 A_SpawnItemEx("FDJPCPNeutronWaveTrail", 0,0,0, 0,0,0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION | SXF_ORIGINATOR | SXF_SETTRACER)
	TNT1 A 0 A_Jump(256, "SpawnLoop")
	Goto SpawnLoop
  Dissipate:
    TNT1 A 0 A_ChangeFlag ("THRUACTORS", 1)
    TNT1 A 0 A_ChangeFlag ("RIPPER", 0)
  DissipateLoop:
    TNT1 A 1
	Loop
	
  Death:
    TNT1 A 5
	Stop
  }
}
