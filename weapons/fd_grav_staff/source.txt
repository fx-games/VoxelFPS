 
ACTOR FDAlienVendettaChainsaw : FDWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVCHAINS"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Obituary "%k sent %o to train in 100x gravity."
  Decal ""
  +WEAPON.MELEEWEAPON
  Tag "Grav-Staff"
  States
  {
  Ready:
    TNT1 A 0 A_PlaySound("alienvendettachainsaw/idle", 7, 0.15, 1, 1.5)
    VSAW A 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0 A_PlaySound("alienvendettachainsaw/idle", 7, 0.15, 1, 1.5)
  SelectLoop:
    VSAW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_StopSound(7)
    TNT1 A 0 A_JumpIfInventory ("FDWeaponAimState", 1, "DeselectActive")
  DeselectLoop:
    VSAW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
  DeselectActive:
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    VSAW D 1 A_SetPitch(pitch - (0.04 * CallACS("FD_recoil")/65536.0))
    VSAW C 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    VSAW B 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	Goto DeselectLoop
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDWeaponAimState", 1, "Strike")
    VSAW B 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    VSAW C 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    VSAW D 1 A_SetPitch(pitch - (0.04 * CallACS("FD_recoil")/65536.0))
  Strike:
    TNT1 A 0 A_GiveInventory ("FDWeaponAimState", 1)
    TNT1 A 0 A_PlaySound("melee/whoosh", 6, 0.6, 0, 1.5)
    TNT1 A 0 A_PlaySound("alienvendettachainsaw/attack", 5, 1.0, 0, 1.2)
    VSAW F 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    VSAW F 1 A_SetPitch(pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    VSAW G 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
    VSAW G 1 A_SetPitch(pitch - (0.03 * CallACS("FD_recoil")/65536.0))
    VSAW H 1 A_SetPitch(pitch - (0.04 * CallACS("FD_recoil")/65536.0))
    VSAW H 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "BerserkHit")
    TNT1 A 0 A_CustomPunch (9*random(7,8), 1, CPF_NORANDOMPUFFZ, "FDAlienVendettaGravStaffPuff", 140)
    Goto ImpactCheck
  BerserkHit:
    TNT1 A 0 A_CustomPunch (9*random(21,24), 1, CPF_NORANDOMPUFFZ, "FDAlienVendettaGravStaffBerserkPuff", 140)
    Goto ImpactCheck
  ImpactCheck:
    TNT1 A 0 A_JumpIfInventory ("FDGravStaffSuccess", 1, "GravAttack")
    VSAW I 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
    VSAW I 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    VSAW J 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    VSAW J 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    VSAW K 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
    VSAW K 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    VSAW L 2
	Goto AttackEnd
  GravAttack:
    TNT1 A 0 A_AlertMonsters (random(128,192))
    TNT1 A 0 A_PlaySound("alienvendettachainsaw/gravity", CHAN_WEAPON, 0.8, 0, 0.9)
    TNT1 A 0 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
    VSAW M 1 Bright// A_FireCustomMissile ("FDAlienVendettaGravityRing1", 0, 0)
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    VSAW M 1 Bright// A_FireCustomMissile ("FDAlienVendettaGravityRing2", 0, 0)
    TNT1 A 0 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    VSAW N 1 Bright// A_FireCustomMissile ("FDAlienVendettaGravityRing3", 0, 0)
    TNT1 A 0 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    VSAW N 1 Bright// A_FireCustomMissile ("FDAlienVendettaGravityRing4", 0, 0)
    TNT1 A 0 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
    VSAW O 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    VSAW O 1
    VSAW P 2
	Goto AttackEnd
	
  AttackEnd:
    VSAW EEE 2
	TNT1 A 0 A_ReFire
    VSAW EEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEEE 1 A_WeaponReady (WRF_NOBOB)
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    VSAW D 1 A_SetPitch(pitch - (0.04 * CallACS("FD_recoil")/65536.0))
    VSAW C 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    VSAW B 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    Goto Ready
  }
}

ACTOR FDAlienVendettaGravStaffEffectCheckerSpawner : FDCustomFunction
{
  States
  {

ACTOR FDAlienVendettaGravStaffEffectChecker
{
	+NOGRAVITY
	+NONSHOOTABLE
	+THRUACTORS
	+NOTONAUTOMAP
    +NOTIMEFREEZE
	+NOCLIP
    +DONTSPLASH
    +NOTELEPORT
  States
  {
  Spawn:
    TNT1 A 1
  SpawnLoop:
	TNT1 A 0 A_JumpIfHealthLower (1, "Ded", AAPTR_MASTER)
	TNT1 A 1 Light("FDAVGravityGlow") A_GiveInventory ("FDAlienVendettaGravStaffEffectVisualSpawner", 1, AAPTR_MASTER)
	TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaGravStaffSpeedDebuff", 1, "WarpCheck", AAPTR_MASTER)
	TNT1 A 0 A_GiveInventory ("FDAlienVendettaGravStaffReset", 1, AAPTR_MASTER)
    Stop
  WarpCheck:
    TNT1 A 0 A_Warp(AAPTR_MASTER, 0,0,8, 0, WARPF_NOCHECKPOSITION, "SpawnLoop", 0.5)
	TNT1 A 0 A_GiveInventory ("FDAlienVendettaGravStaffReset", 1, AAPTR_MASTER)
    Stop
  Ded:
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaGravStaffReset", 1, AAPTR_MASTER)
    Stop
  }
}
