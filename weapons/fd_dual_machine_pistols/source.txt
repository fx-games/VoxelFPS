 
ACTOR FDTNTChaingun : FDWeapon
{
  Inventory.RestrictedTo "FDTNTPlayer"
  Inventory.Icon "TNCHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDTNTBullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%o was hosed by %k."
  Decal "BulletChip"
  Tag "Dual Machine Pistols"
  States
  {
  Ready:
    TCHG A 1 A_WeaponReady
    Loop
	
  Select:
    TCHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    TNT1 A 0 A_StopSound(7)
    TCHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    TCHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDTNTBullets", 2, "FireMain")
    TNT1 A 0 A_JumpIfInventory ("FDTNTBullets", 1, "FireEnd")
	Goto Dryfire
  FireMain:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-5,5)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,5)
    TNT1 A 0 A_PlaySound("tntchaingun/fire", CHAN_WEAPON, 1, 1, 0.7)
    TNT1 A 0 A_PlaySound("tntchaingun/fire", 7, 0.05, 1, 0.20)
    TNT1 A 0 A_JumpIfInventory ("FDTNTUziState", 1, "FireAnimation2")
	Goto FireAnimation1
  FireAnimation1:
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDPainlessBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDPainlessBulletPuff", FBF_NORANDOM)
    TNT1 A 0 A_GiveInventory ("FDTNTUziState", 1)
    TCHG B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner2", 0,0,frandom(-14.6,-14.2),-4, FPF_NOAUTOAIM)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner", 0,0,frandom(14.2,14.6),-4, FPF_NOAUTOAIM)
    TCHG C 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire
    Goto FireEnd
  FireAnimation2:
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDPainlessBulletPuff", FBF_NORANDOM)
    TNT1 A 0 A_TakeInventory ("FDTNTUziState", 1)
    TCHG D 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner2", 0,0,frandom(-14.6,-14.2),-4, FPF_NOAUTOAIM)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner", 0,0,frandom(14.2,14.6),-4, FPF_NOAUTOAIM)
    TCHG E 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Refire
    Goto FireEnd
	
  FireEnd:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-5,5)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,5)
    TNT1 A 0 A_PlaySound("tntchaingun/fireend", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("tntchaingun/fireend", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_JumpIfInventory ("FDTNTUziState", 1, "FireEndAnimation2")
	Goto FireEndAnimation1
  FireEndAnimation1:
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDPainlessBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDPainlessBulletPuff", FBF_NORANDOM)
    TNT1 A 0 A_GiveInventory ("FDTNTUziState", 1)
    TCHG B 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner2", 0,0,frandom(-14.6,-14.2),-4, FPF_NOAUTOAIM)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner", 0,0,frandom(14.2,14.6),-4, FPF_NOAUTOAIM)
    TCHG C 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    Goto FireEndRefire
  FireEndAnimation2:
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
    TNT1 A 0 A_FireBullets (5.5, 3.5, -1, randompick(3,4), "FDPainlessBulletPuff", FBF_NORANDOM)
    TNT1 A 0 A_TakeInventory ("FDTNTUziState", 1)
    TCHG D 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner2", 0,0,frandom(-14.6,-14.2),-4, FPF_NOAUTOAIM)
	TNT1 A 0 A_FireCustomMissile ("FDTNTCasingSpawner", 0,0,frandom(14.2,14.6),-4, FPF_NOAUTOAIM)
    TCHG E 1 Bright A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    Goto FireEndRefire
  FireEndRefire:
    TCHG A 2
	Goto Ready
	
  Flash:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

Actor FDTNTUziState	: FDBoolean {}
