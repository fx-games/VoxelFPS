 
ACTOR FDWhitemareRocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDWhitemarePlayer"
  Inventory.Icon "WMROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDWhitemareRocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was blasted by one of %k's burny boomers."
  Decal ""
  Tag "Firegrasp"
  States
  {
  Ready:
    WROK A 1 A_WeaponReady
    Loop
	
  Select:
    WROK A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_TakeInventory ("FDWhitemareFireballCharge", 0)
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    TNT1 A 0 A_StopSound(7)
  DeselectLoop:
    WROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    WROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_TakeInventory ("FDWhitemareFireballCharge", 0)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareRocket", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_PlaySound("whitemarerocketlauncher/charge", CHAN_WEAPON, 1, 0, 1.0)
    TNT1 A 0 A_PlaySound("whitemarerocketlauncher/chargeloop", 7, 0.7, 1, 1.5)
	WROK B 1
    TNT1 A 0 A_TakeInventory ("FDWhitemareRocket", 1)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFireballCharge", 1)
	Goto Chargeup
	
  Chargeup:
	WROK C 1 A_SpawnItemEx("FDWhitemareFiregraspChargeball", cos(pitch)*32, 0, ((height-19)+2) - sin(pitch)*32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETTARGET) // super mab math
	TNT1 A 0 A_Refire ("Chargeup2")
	Goto ChargeFinish
  Chargeup2:
	WROK D 1 A_SpawnItemEx("FDWhitemareFiregraspChargeball", cos(pitch)*32, 0, ((height-19)+2) - sin(pitch)*32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETTARGET) // super mab math
	TNT1 A 0 A_Refire ("Chargeup3")
	Goto ChargeFinish
  Chargeup3:
	WROK E 1 A_SpawnItemEx("FDWhitemareFiregraspChargeball", cos(pitch)*32, 0, ((height-19)+2) - sin(pitch)*32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETTARGET) // super mab math
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareFireballCharge", 0, "ChargeUp")
	TNT1 A 0 A_JumpIf(CallACS("FD_InfiniteAmmoCheck") == 1, 2)
    TNT1 A 0 A_JumpIfInventory ("FDWhitemareRocket", 1, 1)
	Goto ChargeUp
    TNT1 A 0 A_TakeInventory ("FDWhitemareRocket", 1, TIF_NOTAKEINFINITE)
    TNT1 A 0 A_GiveInventory ("FDWhitemareFireballCharge", 1)
	TNT1 A 0 A_Refire ("Chargeup")
	Goto ChargeFinish
	
  ChargeFinish:
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("whitemarerocketlauncher/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("whitemarerocketlauncher/fire", 7, 1, 0, 0.4)
	TNT1 A 0 A_FireCustomMissile ("FDWhitemareFiregraspProjectile", frandom(-0.2,0.2), 0)
//    TNT1 A 0 A_JumpIfInventory ("FDGotChainsaw", 1, 1)
//	Goto FiringComplete
//    TNT1 A 0 A_GiveInventory ("FDWhitemareHeatLevel", 30)
	Goto FiringComplete
  FiringComplete:
	WROK F 1 Bright A_SetPitch(pitch - (2.75 * CallACS("FD_recoil")/65536.0))
	WROK F 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0)) // 3.25
	WROK G 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	WROK G 1 Bright A_SetPitch(pitch + (1.25 * CallACS("FD_recoil")/65536.0))
	WROK H 1 Bright A_SetPitch(pitch + (0.6 * CallACS("FD_recoil")/65536.0))
	WROK H 1 Bright A_SetPitch(pitch + (0.39 * CallACS("FD_recoil")/65536.0))
	WROK I 1 Bright A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	WROK I 1 Bright A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	WROK J 1 Bright A_SetPitch(pitch + (0.19 * CallACS("FD_recoil")/65536.0))
	WROK J 1 Bright A_SetPitch(pitch + (0.13 * CallACS("FD_recoil")/65536.0))
	WROK K 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	WROK K 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0)) // 3.25
    WROK LLAAAAAAAA 1 A_Refire
    Goto Ready
	
  Flash:
	TNT1 A 2 A_Light1
	TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}
