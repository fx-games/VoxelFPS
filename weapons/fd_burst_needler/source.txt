 
ACTOR FDJPCPChaingun : FDWeapon
{
  Inventory.RestrictedTo "FDJPCPPlayer"
  Inventory.Icon "JPCHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDJPCPBullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%k decided to treat %o with acupuncture."
  Decal "BulletChip"
  Tag "Burst Needler"
  States
  {
  Ready:
    JCHG A 1 A_WeaponReady
    Loop
	
  Select:
    JCHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedleTimeFunTime", 0)
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedlesTravelInBursts", 0)
  DeselectLoop:
    JCHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    JCHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("jpcpchaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("jpcpchaingun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_FireCustomMissile ("FDJPCPBurstNeedleProjectile", 0, 1, 0, 4)
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedlesTravelInBursts", 1)
    TNT1 A 0 A_JumpIfInventory ("FDJPCPNeedleTimeFunTime", 1, "FireAnimation2")
  FireAnimation1:
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleTimeFunTime", 1)
    JCHG B 1 Bright A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    JCHG C 1 A_SetPitch(pitch + (0.17 * CallACS("FD_recoil")/65536.0))
    JCHG A 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	Goto FireEnd
  FireAnimation2:
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedleTimeFunTime", 1)
    JCHG D 1 Bright A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    JCHG E 1 A_SetPitch(pitch + (0.17 * CallACS("FD_recoil")/65536.0))
    JCHG A 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	Goto FireEnd
	
  FireEnd:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPNeedlesTravelInBursts", 0, 1)
    Goto Fire
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedlesTravelInBursts", 0)
  FireBurstChecker:
    JCHG A 1
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleHolder", 1)
    TNT1 A 0 A_JumpIfInventory ("FDJPCPNeedleHolder", 0, "FireAutoBegin")
    TNT1 A 0 A_Refire ("FireBurstChecker")
	
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedleHolder", 0)
    JCHG AAAAAA 1 A_WeaponReady (WRF_NOBOB)
    Goto Ready
	
  FireAutoBegin:
    TNT1 A 0 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedleHolder", 0)
    JCHG A 1
	Goto FireAuto
  FireAutoContinue:
    JCHG A 2
  FireAuto:
    TNT1 A 0 A_JumpIfInventory ("FDJPCPBullets", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("jpcpchaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("jpcpchaingun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_FireCustomMissile ("FDJPCPNeedleProjectile", frandom(-1.0,1.0), 1, 0, 4, 0, frandom(-1.0,1.0))
    TNT1 A 0 A_JumpIfInventory ("FDJPCPNeedleTimeFunTime", 1, "FireAutoAnimation2")
  FireAutoAnimation1:
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleTimeFunTime", 1)
    JCHG B 1 Bright A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    JCHG C 1 A_SetPitch(pitch + (0.17 * CallACS("FD_recoil")/65536.0))
    JCHG A 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	Goto FireAutoEnd
  FireAutoAnimation2:
    TNT1 A 0 A_TakeInventory ("FDJPCPNeedleTimeFunTime", 1)
    JCHG D 1 Bright A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    JCHG E 1 A_SetPitch(pitch + (0.17 * CallACS("FD_recoil")/65536.0))
    JCHG A 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	Goto FireAutoEnd
  FireAutoEnd:
    JCHG A 1 A_Refire ("FireAutoContinue")
  EmitSteam:
    TNT1 A 0 A_PlaySound("jpcpchaingun/hiss", 6, 1, 0, 1.2)
    JCHG FGHIJA 3
    Goto Ready

  Flash:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
    Goto LightDone
  }
}

ACTOR FDJPCPNeedleProjectile : FastProjectile
{
  Radius 4
  Height 4
  Speed 85
  Damage (random(10,14))
  Decal "BulletChip"
  DamageType "Bullet"
  +THRUSPECIES
  +FORCEXYBILLBOARD
  +EXPLODEONWATER
  +BLOODSPLATTER
  +HITTRACER
  Projectile
  Scale 0.7
//  SeeSound "jpcpchaingun/fire"
  Species "Player"
  States
  {
  Spawn:
//    TNT1 A 0 A_Log ("spawn")
    JNDL A 1 A_SpawnItemEx("FDJPCPNeedleTrail",(0.5*momx)/-35.0, -(0.5*momy)/-35.0, 2+(0.5*momz)/-35.0, 0, 0, 0, 0, SXF_ABSOLUTEANGLE | SXF_NOCHECKPOSITION | SXF_TRANSFERSCALE)
    Loop
	
  XDeath:
//    TNT1 A 0 A_Log ("xdeath")
    TNT1 A 0 A_PlaySound("bullet/hitflesh", CHAN_WEAPON, (CallACS("FD_bulletvolume")/65536.0), 0, 1.8)
	Goto DebuffActivate
  Crash:
//    TNT1 A 0 A_Log ("crash")
    TNT1 A 0 A_PlaySound("bullet/ricochet", CHAN_WEAPON, (CallACS("FD_bulletvolume")/65536.0), 0, 1.2)
	Goto DebuffActivate
  DebuffActivate:
    TNT1 A 0 A_JumpIf(CheckClass("ExplosiveBarrel", AAPTR_TRACER, TRUE), "Particles")
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleDebuffGiver", 1, AAPTR_TRACER) // Applies a speed and damage resistance debuff
  Particles:
//    TNT1 A 0 A_Log ("particle")
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "XDeathAnimation")
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AA 0 A_SpawnParticle (ff0000, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAA 0 A_SpawnParticle (ff8f8f, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAA 0 A_SpawnParticle (ffc7c7, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, "XDeathAnimation")
	Goto XDeathAnimation
  XDeathAnimation:
//    TNT1 A 0 A_Log ("xdeathend")
    TNT1 A 0
	Stop
	
  Death:
//    TNT1 A 0 A_Log ("death")
    TNT1 A 0 A_PlaySound("bullet/ricochet", CHAN_WEAPON, (CallACS("FD_bulletvolume")/65536.0), 0, 1.2)
  DeathParticles:
//    TNT1 A 0 A_Log ("death particle")
	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "DeathAnimation")
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AA 0 A_SpawnParticle (f14400, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAA 0 A_SpawnParticle (f4ac3b, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAAAAA 0 A_SpawnParticle (f6f4b9, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, 1,2,3)
	TNT1 AAA 0 A_SpawnParticle (fffee8, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,7), frandom(90,270), 0,0,0, frandom(0.2,6.0),frandom(-0.2,0.2),frandom(-3.0,3.0), 0,0,-0.1, 0.98, -1)
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:
//    TNT1 A 0 A_Log ("death thonk")
	TNT1 A 0 A_JumpIf(CallACS("FD_bulletpuffs") == 0, "NormalPuff")
    TNT1 A 0 A_SetTranslucent (0.98, 1)
    TNT1 A 0 A_SetScale (0.05)
	TNT1 A 0 A_JumpIf(CallACS("FD_bulletpuffs") == 2, 1)
	Goto FadeLoop
	TNT1 A 0 A_SpawnItemEx("FDGunSmoke",0,0,0, 0,0,frandom(0.1,0.3), 0, SXF_NOCHECKPOSITION)
    TNT1 A 0 A_Jump (256, 1,2,3,4,5,6,7)
    TNT1 AAAAAAA 0 A_SpawnItemEx ("FDBulletPuffSpark", frandom(-2.0,2.0),frandom(-2.0,2.0),frandom(-2.0,2.0) ,frandom(-5.0,5.0),frandom(-5.0,5.0),frandom(-1.0,4.0), 0, SXF_NOCHECKPOSITION)
    Goto FadeLoop
  FadeLoop:
//    TNT1 A 0 A_Log ("death fade")
    TNT1 A 0 A_JumpIf (alpha <= 0.4, "FadeLoop2")
    OGLO A 1 Bright Light("FDBulletPuffLight") A_SetScale (scalex * 1.10)
    TNT1 A 0 A_FadeTo (0, 0.1, 1)
    Loop
  FadeLoop2:
//    TNT1 A 0 A_Log ("death fade 2")
    OGLO A 1 Bright A_SetScale (scalex * 1.10)
    TNT1 A 0 A_FadeTo (0, 0.1, 1)
    Loop
	
   NormalPuff: // why the fuck can fast projectiles hit multiple times?
//    TNT1 A 0 A_Log ("vanilla puff")
    TNT1 A 0 A_SetTranslucent (0.5, 0)
    TNT1 A 0 A_CheckCeiling ("NormalPuffNoThrust")
    TNT1 A 0 ThrustThingZ (0, 5, 0, 1) // This causes an infinite loop if it hits a ceiling, excellent cool thanks very good
   NormalPuffNoThrust:
    TNT1 A 0 A_SetScale (1.0)
	PUFF A 4 Bright
	PUFF BCD 4
	Stop
  }
}

ACTOR FDJPCPNeedleDebuffGiver : FDCustomFunction
{
  States
  {
  Pickup:
	TNT1 A 0 A_JumpIfHealthLower (1, "Nope")
	TNT1 A 0 A_CheckFlag ("INVULNERABLE", "Nope")
	TNT1 A 0 A_CheckFlag ("FRIENDLY", "Nope")
    TNT1 A 0 A_JumpIfInventory ("FDJPCPNeedleResistanceDebuff", 1, "AlreadyDebuffed")
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleResistanceDebuffGiver", 1)
	TNT1 A 0 A_SpawnItemEx("FDJPCPNeedleDebuffVisual", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETTRACER)
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleSpeedDebuffActivate", 1)
    Stop
  AlreadyDebuffed:
    TNT1 A 0 A_GiveInventory ("FDJPCPNeedleResistanceDebuffGiver", 1)
    Stop
  Nope:
	TNT1 A 0
	Stop
  }
}

actor FDJPCPNeedleDebuffVisual
{
    RenderStyle Add
    Alpha 0.45
    +FORCEXYBILLBOARD
    +NOCLIP
    States
    {
    Spawn:
		TNT1 A 0
		TNT1 A 0 A_JumpIfHealthLower (1, "Nope", AAPTR_TRACER)
		TNT1 A 0 A_PlaySound ("jpcpchaingun/debuff", 6, 0.9, 0, 1.0)
		TNT1 A 0 A_PlaySound ("jpcpchaingun/debuffloop", 7, 0.5, 1, 1.3)
    DebuffLoop:
		TNT1 A 0 A_Warp (AAPTR_TRACER, 0,0,0, 0, WARPF_NOCHECKPOSITION | WARPF_INTERPOLATE, "DebuffCheck", 0.5)
        Goto EffectDeactivate
    DebuffCheck:
		TNT1 A 0 A_JumpIfInventory ("FDJPCPNeedleResistanceDebuff", 1, 1, AAPTR_TRACER)
        Goto EffectDeactivate
		TNT1 A 0 A_JumpIfHealthLower (1, "EffectDeactivate", AAPTR_TRACER)
    Effects:
		TNT1 A 1 Light("FDNeutronRedGlow")
		TNT1 A 0 A_SpawnItemEx("FDJPCPNeedleDebuffVisualEffect", frandom(-20.0,20.0),frandom(-20.0,20.0),frandom(0.0,50.0), 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETTRACER)
		TNT1 A 0 A_SpawnItemEx("FDJPCPDebuffLine", frandom(-30.0,30.0),frandom(-30.0,30.0),frandom(40.0,60.0), 0,0,-2, 0, SXF_NOCHECKPOSITION | SXF_SETTRACER)
		TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "DebuffLoop")
		TNT1 A 0 A_SpawnParticle (ff0000, SPF_FULLBRIGHT | SPF_RELATIVE, random(40,50), random(3,4), frandom(0,360), random(-30,30),random(-30,30),frandom(40.0,60.0), frandom(-0.5,0.5),frandom(-0.5,0.5),-0.2, 0,0,0, 0.5, -1)
        Goto DebuffLoop
    EffectDeactivate:
		TNT1 A 0 A_GiveInventory ("FDJPCPNeedleSpeedDebuffReset", 1, AAPTR_TRACER)
		TNT1 A 0 A_StopSound (7)
        Stop
    Nope:
		TNT1 A 0
        Stop
    }
}
