 
// NANOCORE EFFECTS
// Increased impact damage

ACTOR FDBTSXPistol : FDWeapon
{
  Inventory.RestrictedTo "FDBTSXPlayer"
  Inventory.Icon "BXPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXPistolCharge"
  Weapon.AmmoUse 0
  Weapon.SlotNumber 2
  Obituary "%k was charged with shooting %o and was fined X dollars."
  Decal ""
  Tag "Charge Pistol"
  States
  {
  Ready:
    TNT1 A 0 A_TakeInventory ("FDBTSXPistolCharge", 0)
    XPIS A 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0 A_TakeInventory ("FDBTSXPistolCharge", 0)
    XPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_StopSound(5)
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_TakeInventory ("FDBTSXPistolCharge", 0)
  DeselectLoop:
    XPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Fire:
  FireLevel0:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire0")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "FireLevel0X")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "FireLevel0X")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXPistolProjectile0", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireNormalEnd
  FireLevel0X:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXPistolProjectileNano0", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireNormalEnd
  FireNormalEnd:
    TNT1 A 0 A_StopSound(5)
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_TakeInventory ("FDBTSXPistolCharge", 0)
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
    TNT1 A 0 A_PlaySound("btsxpistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("btsxpistol/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (0, 0, 1, 0, "FDHarmlessPuff", FBF_NORANDOM) // Used to trigger wall switches
    XPIS B 1 Bright A_SetPitch (pitch - (0.6 * CallACS("FD_recoil")/65536.0))
    XPIS B 1 Bright A_SetPitch (pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    XPIS C 1
    XPIS C 1 A_SetPitch (pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    XPIS D 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    XPIS D 1 A_SetPitch (pitch + (0.18 * CallACS("FD_recoil")/65536.0))
    XPIS E 1 A_SetPitch (pitch + (0.26 * CallACS("FD_recoil")/65536.0))
    XPIS E 1 A_SetPitch (pitch + (0.10 * CallACS("FD_recoil")/65536.0))
    XPIS A 1 A_SetPitch (pitch + (0.04 * CallACS("FD_recoil")/65536.0))
    XPIS A 1
	Goto Ready
  FireLevel1:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire1")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "FireLevel1X")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "FireLevel1X")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXPistolProjectile1", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FireLevel1X:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXPistolProjectileNano1", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FireLevel2:
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire2")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "FireLevel2X")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "FireLevel2X")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXPistolProjectile2", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FireLevel2X:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXPistolProjectileNano2", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FireChargedEnd:
    TNT1 A 0 A_StopSound(5)
    TNT1 A 0 A_StopSound(6)
    TNT1 A 0 A_TakeInventory ("FDBTSXPistolCharge", 0)
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash ("FlashCharge2")
    TNT1 A 0 A_PlaySound("btsxpistol/chargedfire", CHAN_WEAPON, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("btsxpistol/chargedfire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (0, 0, 1, 0, "FDHarmlessPuff", FBF_NORANDOM) // Used to trigger wall switches
    XPIS S 1 Bright A_SetPitch (pitch - (1.4 * CallACS("FD_recoil")/65536.0))
    XPIS S 1 Bright A_SetPitch (pitch - (0.01 * CallACS("FD_recoil")/65536.0))
    XPIS T 1
    XPIS T 1 A_SetPitch (pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    XPIS T 1 A_SetPitch (pitch + (0.17 * CallACS("FD_recoil")/65536.0))
    XPIS T 1 A_SetPitch (pitch + (0.38 * CallACS("FD_recoil")/65536.0))
    XPIS U 1 A_SetPitch (pitch + (0.66 * CallACS("FD_recoil")/65536.0))
    XPIS U 1 A_SetPitch (pitch + (0.18 * CallACS("FD_recoil")/65536.0))
    XPIS D 1 A_SetPitch (pitch + (0.01 * CallACS("FD_recoil")/65536.0))
    XPIS D 1
    XPIS E 2
    XPIS A 2
	Goto Ready
	
  AltFire: // Hold altfire to charge up a more powerful shot, fires on release
    TNT1 A 0 A_PlaySound("btsxpistol/charge", 5, 1, 0, 1.25)
    TNT1 A 0 A_SpawnItemEx ("FDBTSXPistolChargeLight", 1,0,32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    XPIS FG 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
  ChargeLoop00:
    XPIS H 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXPistolCharge", 50, "ChargeLoop1Start")
    TNT1 A 0 A_Refire ("ChargeLoop01")
    Goto FireLevel0
  ChargeLoop01:
    XPIS I 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXPistolCharge", 50, "ChargeLoop1Start")
    TNT1 A 0 A_Refire ("ChargeLoop02")
    Goto FireLevel0
  ChargeLoop02:
    XPIS J 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXPistolCharge", 50, "ChargeLoop1Start")
    TNT1 A 0 A_Refire ("ChargeLoop00")
    Goto FireLevel0
  ChargeLoop1Start:
    XPIS K 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
  ChargeLoop10:
    XPIS L 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXPistolCharge", 0, "ChargeLoop2Start")
    TNT1 A 0 A_Refire ("ChargeLoop11")
    Goto FireLevel1
  ChargeLoop11:
    XPIS M 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXPistolCharge", 0, "ChargeLoop2Start")
    TNT1 A 0 A_Refire ("ChargeLoop12")
    Goto FireLevel1
  ChargeLoop12:
    XPIS N 1 Bright A_GiveInventory ("FDBTSXPistolCharge", 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXPistolCharge", 0, "ChargeLoop2Start")
    TNT1 A 0 A_Refire ("ChargeLoop10")
    Goto FireLevel1
  ChargeLoop2Start:
    TNT1 A 0 A_PlaySound("btsxpistol/chargeloop", 6, 1, 1, 1.25)
    XPIS O 1 Bright
  ChargeLoop20:
    XPIS P 1 Bright
    TNT1 A 0 A_Refire ("ChargeLoop21")
    Goto FireLevel2
  ChargeLoop21:
    XPIS Q 1 Bright
    TNT1 A 0 A_Refire ("ChargeLoop22")
    Goto FireLevel2
  ChargeLoop22:
    XPIS R 1 Bright
    TNT1 A 0 A_Refire ("ChargeLoop20")
    Goto FireLevel2
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  FlashCharge1:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light (4)
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  FlashCharge2:
    TNT1 A 1 A_Light (4)
    TNT1 A 1 A_Light (8)
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
	
  FancyFire0:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "FancyFire0X")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "FancyFire0X")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPistolProjectile0", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireNormalEnd
  FancyFire0X:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPistolProjectileNano0", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireNormalEnd
  FancyFire1:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "FancyFire1X")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "FancyFire1X")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPistolProjectile1", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FancyFire1X:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPistolProjectileNano1", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FancyFire2:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "FancyFire2X")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "FancyFire2X")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPistolProjectile2", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  FancyFire2X:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPistolProjectileNano2", frandom(-0.1,0.1), 0, 0, 6)
	Goto FireChargedEnd
  }
}
