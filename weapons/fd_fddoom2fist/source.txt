 
ACTOR FDDoom2Fist : FDWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2FIST"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Obituary "%o was dusted off by %k."
  Decal ""
  +WEAPON.MELEEWEAPON
  Tag "Melee"
  States
  {
  Ready:
    TNT1 A 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 1 A_Raise
    "####" "##" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 1 A_Lower
    "####" "##" 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_PlaySound("plutfist/swing", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("melee/whoosh", 7, 1.0, 0, 1.5)
	DFIS LBCD 1
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla")
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "BerserkHit")
    TNT1 A 0 A_CustomPunch (5*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDFistPuff", 80)
    Goto FireFinish
  BerserkHit:
    TNT1 A 0 A_CustomPunch (5*random(25,50), 1, CPF_NORANDOMPUFFZ, "FDFistPuff", 80)
    Goto FireFinish
  FireFinish:
	DFIS F 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
	DFIS E 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	DFIS F 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	DFIS F 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	DFIS G 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	DFIS G 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	DFIS HI 2
	DFIS JKL 1
	TNT1 A 5 A_ReFire
    Goto Ready
  FiringVanilla:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "BerserkHitVanilla")
    TNT1 A 0 A_CustomPunch (2*random(1,10), 1, CPF_NORANDOMPUFFZ, "FDFistPuff", 64)
    Goto FireFinish
  BerserkHitVanilla:
    TNT1 A 0 A_CustomPunch (20*random(1,10), 1, CPF_NORANDOMPUFFZ, "FDFistPuff", 64)
    Goto FireFinish
  }
}
