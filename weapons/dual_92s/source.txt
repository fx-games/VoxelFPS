ACTOR Dual92 : Weapon
{
   Weapon.SelectionOrder 700
   Weapon.SlotNumber 2
   weapon.ammogive1 0
   weapon.ammogive2 30
   weapon.ammotype1 "Dual92Loaded"
   weapon.ammotype2 "Clip"
   weapon.ammouse1 1
   weapon.ammouse2 1
   Inventory.PickupMessage "You got the Dual 92s! Shed some blood!"
   Obituary "%o didn't wake up to a better tomorrow."
   Tag "Dual 92s"
   Scale 0.6
   +AMMO_CHECKBOTH
   +NOAUTOFIRE
   +WEAPON.NOALERT
   States
   {
   Ready:
     B92I A 1 A_WeaponReady(WRF_ALLOWRELOAD)
     Loop
   Deselect:
     B92I A 1 A_Lower
     B92I AA 0 A_Lower
     Loop
   Select:
     B92I A 1 A_Raise
     B92I AA 0 A_Raise
     Loop
   Fire:
     TNT1 A 0 A_JumpIf(CountInv("Dual92Loaded") == 0,"Reload")
	 TNT1 A 0 A_AlertMonsters
 	 TNT1 A 0 A_PlaySound("B92FIRE", CHAN_WEAPON)
	 TNT1 A 0 A_FireBullets(2.5, 2.5, -1, 5)
	 TNT1 A 0 A_JumpIfInventory("92FireSwap",1,"Fire2")
	 Goto Fire1
   Fire1:
     B92L A 1 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY|WRF_NOSWITCH)
	 TNT1 A 0 A_Quake(1, 2, 0, 48576, "None")
	 TNT1 A 0 A_GiveInventory("92FireSwap",1)
     B92L B 1 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY|WRF_NOSWITCH)
     B92L C 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY|WRF_NOSWITCH)
     B92L DEF 1 A_WeaponReady(WRF_NOSWITCH)
	 goto FireRest
   Fire2:
     B92R A 1 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY|WRF_NOSWITCH)
	 TNT1 A 0 A_Quake(1, 2, 0, 48576, "None")
	 TNT1 A 0 A_TakeInventory("92FireSwap",99)
     B92R B 1 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY|WRF_NOSWITCH)
     B92R C 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY|WRF_NOSWITCH)
     B92R DEF 1 A_WeaponReady(WRF_NOSWITCH)
	 goto FireRest
   FireRest:	 
	 B92I A 1 A_WeaponReady
	 B92I A 0 A_ReFire("Fire")
	 Goto Ready
Reload:
     TNT1 A 0 A_JumpIfInventory("Dual92Loaded", 30, 2)
     TNT1 A 0 A_JumpIfInventory("Clip", 1, "ReloadWork")
     B92I A 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSECONDARY)
     Goto Ready
ReloadWork:
	 D9RA A 0 A_TakeInventory("92FireSwap",1)
     D92R KA 1
     D92R BC 1
     TNT1 A 0 A_PlaySoundEX("B92REL1",CHAN_7)
	 TNT1 A 1
	 TNT1 A 0 A_PlaySoundEX("B92REL1",CHAN_6)
	 TNT1 A 5
	 R92L ABCD 1
	 R92L E 5
     TNT1 A 0 A_PlaySoundEX("B92REL2",CHAN_6)
     R92L FGHI 1
	 R92L JK 2
     R92L CBAL 1
	 TNT1 A 5
	 R92R ABCD 1
	 R92R E 5
	 TNT1 A 0 A_PlaySoundEX("B92REL2",CHAN_6)
     R92R FGHI 1
	 R92R JK 2
     R92R CBAL 1
     TNT1 A 2
ReloadLoop:
     TNT1 A 0 A_TakeInventory("Clip", 1)
	 TNT1 A 0 A_GiveInventory("Dual92Loaded",1)
     TNT1 A 0 A_JumpIfInventory("Dual92Loaded", 30, "ReloadFinish")
     TNT1 A 0 A_JumpIfInventory("Clip", 1, "ReloadLoop")
     Goto ReloadFinish
ReloadFinish:
     D92R DEF 1
	 TNT1 A 0 A_PlaySoundEX("B92REL3",CHAN_6)
	 D92R GH 1
	 D92R IJK 2
     TNT1 A 0 A_Refire("Fire")
     Goto Ready
   Spawn:
     B92G A -1
     Stop
   }
}

ACTOR Dual92Loaded : Ammo
{
   +IGNORESKILL
   Inventory.Amount 0
   Inventory.MaxAmount 30
   Ammo.BackpackAmount 0
   Ammo.BackpackMaxAmount 30
}

ACTOR 92FireSwap : Inventory { Inventory.MaxAmount 1 }
