ACTOR FDBTSXPlasmaRifle : FDWeapon
{
  Inventory.RestrictedTo "FDBTSXPlayer"
  Inventory.Icon "BXPLASMA"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXCell"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 6
  Obituary "%o was mowed down by %k."
  Decal "PlasmaScorch"
  Tag "Plasma Vulcan"
  States
  {
  Ready:
    XPLA A 1 A_WeaponReady
    Loop
	
  Select:
    XPLA A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
    TNT1 A 0 A_StopSound(7)
    XPLA A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    XPLA A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Dryfire
	
  Windup:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "Hold")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "Hold")
    TNT1 A 0 A_PlaySound("btsxplasmarifle/spinup", 6, 1, 0, 1.5)
  Windup0:
    XPLA BCD 3
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown0
  Windup1:
    XPLA ABCD 2
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown1
    TNT1 A 0 A_Refire ("Windup2")
	Goto Winddown1
  Windup2:
    XPLA ABCD 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown2
    TNT1 A 0 A_Refire ("Windup3")
	Goto Winddown2
  Windup3:
    XPLA ABCD 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown3
    TNT1 A 0 A_Refire ("Hold")
	Goto Winddown3
	
  Hold:
    TNT1 A 0 A_PlaySound("btsxplasmarifle/fireloop", CHAN_WEAPON, 1, 1, 0.7)
    TNT1 A 0 A_PlaySound("btsxplasmarifle/fireloop", 7, 0.05, 1, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyHold0")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXPlasmaProjectile", frandom(-7.2,7.2), 1, 0,0, 1, frandom(-4.6,4.6))
	XPLA I 1 Bright A_SetPitch(pitch - (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 2)
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
    TNT1 A 0 A_Refire ("Hold1")
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
  Hold1:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXPlasmaProjectile", frandom(-7.2,7.2), 1, 0,0, 1, frandom(-4.6,4.6))
	XPLA J 1 Bright A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown
    TNT1 A 0 A_Refire ("Hold2")
	Goto Winddown
  Hold2:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXPlasmaProjectile", frandom(-7.2,7.2), 1, 0,0, 1, frandom(-4.6,4.6))
	XPLA K 1 Bright A_SetPitch(pitch - (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 2)
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
    TNT1 A 0 A_Refire ("Hold3")
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
  Hold3:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXPlasmaProjectile", frandom(-7.2,7.2), 1, 0,0, 1, frandom(-4.6,4.6))
	XPLA L 1 Bright A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown
    TNT1 A 0 A_Refire ("Hold")
	Goto Winddown
	
  Winddown:
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_PlaySound("btsxplasmarifle/fireend", CHAN_WEAPON, 1, 0, 1.5)
    TNT1 A 0 A_PlaySound("btsxplasmarifle/fireend", 7, 1, 0, 1.5)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "Ready")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "Ready")
    TNT1 A 0 A_PlaySound("btsxplasmarifle/spindown", 6, 1, 0, 1.5)
  Winddown3:
    XPLA ABCD 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown2
    TNT1 A 0 A_Refire ("Hold")
  Winddown2:
    XPLA ABCD 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown1
    TNT1 A 0 A_Refire ("Winddown3")
  Winddown1:
    XPLA ABCD 2
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown0
    TNT1 A 0 A_Refire ("Winddown2")
  Winddown0:
    XPLA ABCD 3
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Ready
    TNT1 A 0 A_Refire ("Winddown1")
	Goto Ready
	
  Flash:
    TNT1 A 1 A_Light (random(1,4))
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
	
  FancyHold0:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPlasmaProjectile", frandom(-7.2,7.2), 1, 0,6, 1, frandom(-4.6,4.6))
	XPLA E 1 Bright A_SetPitch(pitch - (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 2)
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
    TNT1 A 0 A_Refire ("FancyHold1")
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
  FancyHold1:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPlasmaProjectile", frandom(-7.2,7.2), 1, 0,6, 1, frandom(-4.6,4.6))
	XPLA F 1 Bright A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown
    TNT1 A 0 A_Refire ("FancyHold2")
	Goto Winddown
  FancyHold2:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPlasmaProjectile", frandom(-7.2,7.2), 1, 0,6, 1, frandom(-4.6,4.6))
	XPLA G 1 Bright A_SetPitch(pitch - (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 2)
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
    TNT1 A 0 A_Refire ("FancyHold3")
    TNT1 A 0 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	Goto Winddown
  FancyHold3:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyPlasmaProjectile", frandom(-7.2,7.2), 1, 0,6, 1, frandom(-4.6,4.6))
	XPLA H 1 Bright A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXCell", 1, 1)
	Goto Winddown
    TNT1 A 0 A_Refire ("FancyHold0")
	Goto Winddown
  }
}
