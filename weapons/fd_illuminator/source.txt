ACTOR FDAliensIlluminatorBeam : FDTNTMaserBeam
{

	Decal "Scorch"
    DamageType "AlienEnergy"
	Scale 0.5
	+NOTIMEFREEZE
	-PAINLESS
    States
    {
        Spawn:
            TNT1 A 0
            TNT1 A 0 A_SpawnItemEx("FDAliensIlluminatorBeamEffect", 5,0,2, velx,vely,velz, 0, SXF_NOCHECKPOSITION | SXF_ABSOLUTEMOMENTUM | SXF_TRANSFERPOINTERS)
			TNT1 A 0 A_Jump (256, "Movement")
			Goto Movement
        Movement:
            TNT1 A 0 A_JumpIf(user_life[0]>=500,"Death")
            TNT1 A 0 A_Warp(AAPTR_DEFAULT,velx,vely,velz,0,WARPF_ABSOLUTEOFFSET,1)
            Goto DetectSolid
            TNT1 A 0 A_SetUserArray("user_life",0,user_life[0]+1)
            Loop
        Death:
			TNT1 A 0 A_ChangeFlag ("NOCLIP", 1)
			TNT1 A 0 A_Explode (5, 256, 0)
			TNT1 A 0 A_SetScale (frandom(2.8,3.2))
			TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "DeathLoop")
			TNT1 A 0 A_Jump (256, 1,2,3)
			TNT1 AAAAAAAA 0 A_SpawnParticle (fe1784, SPF_FULLBRIGHT | SPF_RELATIVE, random(30,70), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)
