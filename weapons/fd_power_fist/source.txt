 
// NANOCORE EFFECTS
// Always 100% charged, rapidfire

ACTOR FDBTSXFist : FDWeapon

  Inventory.Icon "BXFIST"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXFistCharge"
  Weapon.AmmoUse 0
  Weapon.SlotNumber 1
  Weapon.SlotPriority 1
  Obituary "%o was smacked into orbit by %k."
  Decal ""
  +WEAPON.MELEEWEAPON
  Tag "Power Fist"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "ReadyOverdrive")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "ReadyOverdrive")
    TNT1 A 0 A_TakeInventory ("FDBTSXFistCharge", 0)
    XFIS A 1 A_WeaponReady
    Loop
  ReadyOverdrive:
    TNT1 A 0 A_GiveInventory ("FDBTSXFistCharge", 100)
    TNT1 A 1 A_WeaponReady
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "ReadyOverdrive")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "ReadyOverdrive")
    TNT1 A 0 A_TakeInventory ("FDBTSXFistCharge", 0)
    XFIS B 1
    Goto Ready
	
  Select:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "SelectInstant")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "SelectInstant")
    XFIS A 1 A_Raise
    "####" "##" 0 A_Raise
    Loop
  SelectInstant:
    TNT1 A 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "DeselectInstant")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "DeselectInstant")
    XFIS A 1 A_Lower
    "####" "##" 0 A_Lower
    Loop
  DeselectInstant:
    TNT1 A 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "OverdriveHit")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "OverdriveHit")
	XFIS B 1
    TNT1 A 0 A_PlaySound("btsxfist/charge", 7, 1.0, 0, 1.5)
  Charge:
    TNT1 A 1 A_GiveInventory ("FDBTSXFistCharge", 3)
    TNT1 A 0 A_Refire ("Charge")
	Goto Swing
  Swing:
    TNT1 A 0 A_PlaySound("btsxfist/fire", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("btsxfist/swing", 7, 1.0, 0, 1.5)
	XFIS CD 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 0, "HitX")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 90, "Hit9")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 80, "Hit8")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 70, "Hit7")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 60, "Hit6")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 50, "Hit5")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 40, "Hit4")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 30, "Hit3")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 20, "Hit2")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistCharge", 10, "Hit1")
	Goto Hit0
  OverdriveHit:
    TNT1 A 0 A_PlaySound("btsxfist/fire", CHAN_WEAPON, 1.0, 0, 1.5)
    TNT1 A 0 A_PlaySound("btsxfist/swing", 7, 1.0, 0, 1.5)
	XFIS CD 1
	Goto HitX
	
  Hit0:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit0BerserkHit")
    TNT1 A 0 A_CustomPunch (4*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit0BerserkHit:
    TNT1 A 0 A_CustomPunch (4*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit1:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit1BerserkHit")
    TNT1 A 0 A_CustomPunch (5*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit1BerserkHit:
    TNT1 A 0 A_CustomPunch (5*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit2:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit2BerserkHit")
    TNT1 A 0 A_CustomPunch (6*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit2BerserkHit:
    TNT1 A 0 A_CustomPunch (6*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit3:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit3BerserkHit")
    TNT1 A 0 A_CustomPunch (7*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit3BerserkHit:
    TNT1 A 0 A_CustomPunch (7*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit4:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit4BerserkHit")
    TNT1 A 0 A_CustomPunch (8*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit4BerserkHit:
    TNT1 A 0 A_CustomPunch (8*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit5:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit5BerserkHit")
    TNT1 A 0 A_CustomPunch (9*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit5BerserkHit:
    TNT1 A 0 A_CustomPunch (9*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit6:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit6BerserkHit")
    TNT1 A 0 A_CustomPunch (10*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit6BerserkHit:
    TNT1 A 0 A_CustomPunch (10*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit7:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit7BerserkHit")
    TNT1 A 0 A_CustomPunch (11*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit7BerserkHit:
    TNT1 A 0 A_CustomPunch (11*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit8:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit8BerserkHit")
    TNT1 A 0 A_CustomPunch (12*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit8BerserkHit:
    TNT1 A 0 A_CustomPunch (12*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit9:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "Hit9BerserkHit")
    TNT1 A 0 A_CustomPunch (13*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  Hit9BerserkHit:
    TNT1 A 0 A_CustomPunch (13*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  HitX:
    TNT1 A 0 A_JumpIfInventory ("PowerStrength", 1, "HitXBerserkHit")
    TNT1 A 0 A_CustomPunch (15*random(5,10), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
  HitXBerserkHit:
    TNT1 A 0 A_CustomPunch (15*random(15,30), 1, CPF_NORANDOMPUFFZ, "FDBTSXFistPuff", 80)
    Goto FireFinish
	
  FireFinish:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 3)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 2)
    TNT1 A 0 A_TakeInventory ("FDBTSXFistCharge", 0)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXFistImpact", 1, "FireFinishHit")
	XFIS E 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
	XFIS E 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	XFIS F 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	XFIS F 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	XFIS G 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	XFIS G 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 4)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 3)
	XFIS H 2
	XFIS A 5 A_ReFire
    Goto Ready
	XFIS I 2
    Goto Ready
  FireFinishHit:
	XFIS D 1 A_SetPitch(pitch - (0.65 * CallACS("FD_recoil")/65536.0))
	XFIS D 1 A_SetPitch(pitch + (0.3 * CallACS("FD_recoil")/65536.0))
	XFIS D 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	XFIS E 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	XFIS E 1 A_SetPitch(pitch + (0.07 * CallACS("FD_recoil")/65536.0))
	XFIS F 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	XFIS F 1
	XFIS G 2
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, 4)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, 3)
	XFIS H 2
	XFIS A 5 A_ReFire
    Goto Ready
	XFIS I 2
    Goto Ready
  }
