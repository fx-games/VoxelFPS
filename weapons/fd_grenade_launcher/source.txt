 
ACTOR FDPlutRocketLauncher : FDWeapon
{
  Inventory.RestrictedTo "FDPlutPlayer"
  Inventory.Icon "PLROCKET"
  Weapon.UpSound ""
  Weapon.AmmoType "FDPlutRocket"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 5
  Obituary "%o was blasted apart by %k."
  Decal ""
  Tag "Grenade Launcher"
  States
  {
  Ready:
    PROK A 1 A_WeaponReady
    Loop
	
  Select:
    PROK A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    PROK A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    PROK A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDPlutRocket", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_GunFlash
    PROK BBCC 1 Bright
    TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("plutrocketlauncher/fire1", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("plutrocketlauncher/fire2", 7, 1, 0, 0.4)
	TNT1 A 0 A_FireCustomMissile ("FDPlutGrenade", frandom(-1.0,1.0), 1)
    PROK D 1 Bright A_SetPitch (pitch - (4.8 * CallACS("FD_recoil")/65536.0))
    PROK D 1 Bright A_SetPitch (pitch - (0.15 * CallACS("FD_recoil")/65536.0))
    PROK E 1 Bright A_SetPitch (pitch - (0.05 * CallACS("FD_recoil")/65536.0))
    PROK E 1 Bright A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGrenadeCasingSpawner", 0,0,frandom(4.6,4.2),-8, FPF_NOAUTOAIM)
    PROK F 1 A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    PROK F 1 A_SetPitch (pitch + (0.8 * CallACS("FD_recoil")/65536.0))
    PROK G 1 A_SetPitch (pitch + (0.9 * CallACS("FD_recoil")/65536.0))
    PROK G 1 A_SetPitch (pitch + (0.9 * CallACS("FD_recoil")/65536.0))
    PROK H 1 A_SetPitch (pitch + (0.8 * CallACS("FD_recoil")/65536.0))
    PROK H 1 A_SetPitch (pitch + (0.6 * CallACS("FD_recoil")/65536.0))
    PROK A 1 A_SetPitch (pitch + (0.4 * CallACS("FD_recoil")/65536.0))
    PROK A 1 A_SetPitch (pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    PROK A 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    PROK A 1 A_SetPitch (pitch + (0.03 * CallACS("FD_recoil")/65536.0))
    PROK A 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    PROK A 4
    PROK AAAAAAAAAA 1 A_Refire
    Goto Ready
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 2 A_Light (3)
    TNT1 A 2 A_Light (4)
    TNT1 A 2 A_Light2
    TNT1 A 2 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

ACTOR FDPlutGrenade : Rocket // Copied from DRLA because lazy HEY IT'S LIKE I'M A LAZY PERSON OR SOMETHING
{
  Radius 6
  Height 4
  Scale 0.7
  Speed 35
  Damage (10*random(6,14))
  Damagetype "Explosive"
  Gravity 0.65
  Species "Player"
  +THRUSPECIES
  +USEBOUNCESTATE
  +THRUGHOST
//  -NOGRAVITY
  +EXPLODEONWATER
  BounceType "Doom"
  BounceFactor 0.25
  WallBounceFactor 0.5
  BounceCount 5
  ReactionTime 139
  Decal "Scorch"
  SeeSound ""
  BounceSound "plutrocketlauncher/bounce"
  DeathSound ""
  States
  {
  Spawn:
    TNT1 A 0 A_CountDown
    GREX A 1 Light("FDRocket")
    Loop
  Bounce:

    TNT1 A 0 A_CountDown
    GREX A 1 Light("FDRocket")
    Goto Spawn
  Death:

    TNT1 A 0 A_ChangeFlag ("NOGRAVITY", 1)

	TNT1 A 0 A_JumpIf(CallACS("FD_particles") == 0, "QuakeTime")
    TNT1 A 0 A_Jump (256, 1,5,10)
	TNT1 AAAAAAAAAAAA 0 A_SpawnParticle (730000, SPF_FULLBRIGHT | SPF_RELATIVE, random(30,70), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)

    TNT1 A 0 A_Jump (256, 1,5,10)

    TNT1 A 0 A_Jump (256, 1,5,10)
	TNT1 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 0 A_SpawnParticle (ffa35b, SPF_FULLBRIGHT | SPF_RELATIVE, random(15,40), random(1,10), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,-0.1, 0.98, -1)

	TNT1 AAAAAAAAA 0 A_SpawnParticle (ffffff, SPF_FULLBRIGHT | SPF_RELATIVE, random(10,30), random(1,4), frandom(0,360), 0,0,0, frandom(0.2,12.0),frandom(-0.2,0.2),frandom(-6.0,6.0), 0,0,0, 0.98, -1)
    TNT1 A 0 A_Jump (256, "QuakeTime")
	Goto QuakeTime
  QuakeTime:

    TNT1 A 0 A_Quake (5, 8, 0, 512, "")
    TNT1 A 0 A_Jump (256, "DeathAnimation")
	Goto DeathAnimation
  DeathAnimation:

    MEXP C 3 Bright Light("FDExplosion3")

    GLXP C 2 Bright Light("FDGroundExplosion3")
    GLXP D 2 Bright Light("FDGroundExplosion4")

    GLXP H 2 Bright Light("FDGroundExplosion8")
    GLXP I 2 Bright Light("FDGroundExplosion9")
    GLXP J 2 Bright Light("FDGroundExplosion10")
    Stop
  }
