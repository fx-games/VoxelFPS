 
ACTOR FDDoom2Pistol : FDWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2PISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDDoom2PistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%k fired a pistol that didn't look like a weird dome sitting on their hand and killed %o."
  Decal "BulletChip"
  Tag "Pistol"
  States
  {
  Ready:
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    TNT1 A 0 A_JumpIfInventory ("FDDoom2PistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReadyUseAmmo")
    DPIS A 1 A_WeaponReady
    Loop
  ReadyFull:
    DPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Goto Ready
	
  ReadyUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Bullets", 1, 1)
	Goto ReadyFull
    DPIS A 1 A_WeaponReady
    Goto Ready
	
  Select:
    DPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    DPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    TNT1 A 0 A_JumpIfInventory ("FDWeaponAimState", 1, "Dryfire2")
    DPIS A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
  Dryfire2:
    DPIS C 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto CrimeForReloading

// aaaagh
// next person who asks for yet another pistol cvar gets fucking shanked

  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDWeaponAimState", 1, "Hold")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto FireNoAmmoCheckDone
  FireNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire")
    Goto FireNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Bullets", 1, "FireNoAmmoCheckDone")
    Goto Dryfire
	
  FireNoAmmoCheckDone:
    TNT1 A 0 A_PlaySound("tntpistol/aim", 5, 1, 0, 1.25)
    DPIS IBC 1
    TNT1 A 0 A_GiveInventory ("FDWeaponAimState", 1)
	Goto Hold
	
  Hold:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "HoldNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto HoldNoAmmoCheckDone
  HoldNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire2")
    Goto HoldNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Bullets", 1, "HoldNoAmmoCheckDone")
    Goto Dryfire2
	
  FireNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (2.5, 0, 1, randompick(14,16), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
  FireNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (2.5, 0, 1, randompick(7,8), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
    Goto FireComplete
	
  FireVanilla:
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, 2) // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (5.6, 0, 1, 5*random(1,3), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	Goto FireComplete
    TNT1 A 0 A_FireBullets (5.6, 0, 1, 5*random(1,3), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	Goto FireComplete
	
  FireNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDDoom2Bullets", 1, TIF_NOTAKEINFINITE)
    Goto FireComplete
	
  HoldNoAmmoCheckDone:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("doom2pistol/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("doom2pistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FireVanilla")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoMagConsumption") // No reload means no mag consumption
    TNT1 A 0 A_FireBullets (2.5, 0, 1, randompick(14,16), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FireComplete:
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FD9mmCasingSpawner", 0,0,frandom(0.5,0.6),-1, FPF_NOAUTOAIM)
    DPIS D 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    DPIS D 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    DPIS E 1 A_SetPitch (pitch - (0.08 * CallACS("FD_recoil")/65536.0))
    DPIS E 1 A_SetPitch (pitch - (0.02 * CallACS("FD_recoil")/65536.0))
    DPIS F 1 A_SetPitch (pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    DPIS F 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    DPIS G 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    DPIS G 1 A_SetPitch (pitch + (0.3 * CallACS("FD_recoil")/65536.0))
    DPIS H 1 A_SetPitch (pitch + (0.048 * CallACS("FD_recoil")/65536.0))
    DPIS H 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    DPIS C 2
    TNT1 A 0 A_JumpIfNoAmmo ("CeaseFire")
	Goto CrimeForReloading
  CeaseFire:
    DPIS CC 1 A_Refire("Hold")
  CeaseFire2:
    DPIS BB 1 A_Refire("CeaseFire")
  CeaseFire3:
    DPIS II 1 A_Refire("CeaseFire2")
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    Goto Ready
	
  AltFire:
    TNT1 A 0 A_JumpIfInventory ("FDDoom2PistolAmmo", 0, "ReadyReset")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 0, "Reloadery")
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Bullets", 1, "Reloadery")
	Goto DryFire // Normally we'd return to Ready, but we disable altfire if full anyway, so we can go to dryfire without problems
  Reloadery:
    TNT1 A 0 A_JumpIfInventory ("FDWeaponAimState", 1, "HoldToReload")
  Reloadery2:
    TNT1 A 0 A_PlaySound("doom2pistol/unload", 6, 1, 0, 1.25)
    DPIS I 10
    DPIS JK 2
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReloadUseAmmo")
    TNT1 A 0 A_GiveInventory ("FDDoom2PistolAmmo", 15)
	Goto ReloadFinish
  ReloadUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDDoom2PistolAmmo", 0, "ReloadFinish")
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Bullets", 1, 1)
	Goto ReloadFinish
    TNT1 A 0 A_GiveInventory ("FDDoom2PistolAmmo", 1)
    TNT1 A 0 A_TakeInventory ("FDDoom2Bullets", 1)
	Goto ReloadUseAmmo
  ReloadFinish:
    TNT1 A 0 A_PlaySound("doom2pistol/load", 5, 1, 0, 1.25)
    DPIS LMNOPQ 2
    Goto Ready
	
  HoldToReload:
    DPIS CBI 1
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    Goto Reloadery2
  ReadyReset:
    DPIS CBI 1
    TNT1 A 0 A_ClearRefire
    TNT1 A 0 A_TakeInventory ("FDWeaponAimState", 1)
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
		
	// no don't look it's crime
  CrimeForReloading:
    DPIS CCCCCCCCCCCCCCCCCCCCCCCCCCCCCCCCCCC 1 A_WeaponReady (WRF_NOBOB)
	Goto CeaseFire
  }
}
