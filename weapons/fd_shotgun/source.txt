 
ACTOR FDDoom2Shotgun : FDWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2SHOTGU"
  Weapon.UpSound ""
  Weapon.AmmoType "FDDoom2Shells"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 3
  Obituary "%o couldn't escape %k's sight."
  Decal "BulletChip"
  Tag "Shotgun"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Shells", 1, "ReadyAmmo")
    DSHT A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    DSHT A 1 A_WeaponReady
    Loop
	
  Select:
    DSHT A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    DSHT A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    DSHT A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDDoom2Shells", 1, 1)
	Goto Dryfire
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,0,6)
    TNT1 A 0 A_PlaySound("doom2shotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("doom2shotgun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla")
    TNT1 A 0 A_FireBullets (5.6, 0.8, 9, 5, "FDBulletPuff", FBF_USEAMMO)
  FiringComplete:
    DSHT I 1 Bright A_SetPitch (pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    DSHT I 1 Bright A_SetPitch (pitch - (0.4 * CallACS("FD_recoil")/65536.0))
    DSHT J 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    DSHT J 1 Bright A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    DSHT K 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    DSHT K 1 A_SetPitch (pitch + (1.05 * CallACS("FD_recoil")/65536.0))
    DSHT L 1 A_SetPitch (pitch + (0.25 * CallACS("FD_recoil")/65536.0))
  GonnaReload:
    DSHT B 1
    DSHT C 1
	DSHT D 2
	DSHT E 3
	DSHT F 4
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDShellCasingSpawner", 0,0,-14,-6, FPF_NOAUTOAIM)
    TNT1 A 0 A_PlaySound("doom2shotgun/pump", 5, 1, 0, 1.25)
	DSHT G 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	DSHT H 1 A_SetPitch(pitch + (0.23 * CallACS("FD_recoil")/65536.0))
	DSHT H 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	DSHT H 1 A_SetPitch(pitch - (0.06 * CallACS("FD_recoil")/65536.0))
	DSHT G 1 A_SetPitch(pitch - (0.28 * CallACS("FD_recoil")/65536.0))
	DSHT G 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
	DSHT FEDCB 2
	DSHT A 3
	DSHT AAAAAAA 1 A_Refire // Gun fires at a speed of 37 tics
    Goto Ready
  FiringVanilla:
    TNT1 A 0 A_FireBullets (5.6, 0, 7, 5, "FDBulletPuff", FBF_USEAMMO)
	Goto FiringComplete
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}
