ACTOR FDHellboundPistol : FDWeapon
{
  Inventory.RestrictedTo "FDHellboundPlayer"
  Inventory.Icon "HBPISTOL"
  Weapon.UpSound ""
  Weapon.AmmoType "FDHellboundPistolAmmo"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 2
  Obituary "%o got %g innards blown out by %k's revolvers."
  Decal "BulletChip"
  Tag "Dual Magnums"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "ToggleAmmoHax2") // Reloading is disabled if you're a bastard that uses double ammo I hate you
	TNT1 A 0 A_TakeInventory("FDHellboundPistolDoubleAmmo", 1)
    TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReadyUseAmmo")
    HPIS A 1 A_WeaponReady
    Loop
  ReadyFull:
    HPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Goto Ready
	
  ReadyUseAmmo: // Used with the pistol ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, 1)
	Goto ReadyFull
    HPIS A 1 A_WeaponReady
    Goto Ready
	
  ReadyUseAmmo2: // Used with the double ammo cvar
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, 1)
	Goto ReadyFull
    HPIS A 1 A_WeaponReady (WRF_NOSECONDARY)
    Goto Ready
	
  Select:
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "ToggleAmmoHax")
	TNT1 A 0 A_TakeInventory("FDHellboundPistolDoubleAmmo", 1)
  SelectLoop:
    HPIS A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    HPIS A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    HPIS A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  ToggleAmmoHax:
	TNT1 A 0 A_GiveInventory("FDHellboundPistolDoubleAmmo", 1)
    Goto SelectLoop
  ToggleAmmoHax2:
	TNT1 A 0 A_GiveInventory("FDHellboundPistolDoubleAmmo", 1)
    Goto ReadyUseAmmo2

  Fire:
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireNoReload")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireNoReload") // If reloading is disabled, we need to go to dryfire, not altfire
    TNT1 A 0 A_JumpIfNoAmmo ("AltFire")
    Goto FireNoAmmoCheckDone
  FireNoReload:
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, 3)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2) // If ammo is also enabled, we gotta check for the base bullet ammo for that gun
    TNT1 A 0 A_JumpIfNoAmmo ("Dryfire")
    Goto FireNoAmmoCheckDone
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, "FireNoAmmoCheckDone")
    Goto Dryfire
	
  FireNoAmmoCheckDone:
	TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 1, 1)
	Goto Altfire
	TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolState", 1, "FireRight")
	Goto FireLeft
	
  FireLeftContinueH:
	HPIS K 1
  FireLeftContinueI:
	HPIS L 1
  FireLeftContinueJ:
	TNT1 A 0 A_GiveInventory("FDHellboundPistolState", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolLeftShots", 1)
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftContinueNoMagConsumption")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireLeftContinueNoMagConsumption") // No reload means no mag consumption
	TNT1 A 0 A_FireBullets (2.5, 0, -1, randompick(28,32), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", CHAN_WEAPON, 1.0, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,4)
	HPIS M 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
	Goto FireLeftQuickComplete
	
  FireLeftContinueNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireLeftContinueNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(22,24), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireLeftContinueNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftContinueNoMagButAmmo")
    Goto FireLeftContinueNoMag
  FireLeftContinueNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireLeftContinueNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftContinueNoMagButAmmo")
    Goto FireLeftContinueNoMag
  FireLeftContinueNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDHellboundBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireLeftContinueNoMag
  FireLeftContinueNoMag:
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", CHAN_WEAPON, 1.0, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,4)
	HPIS M 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    Goto FireLeftQuickComplete
	
  FireLeftNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireLeftNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(22,24), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireLeftNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftNoMagButAmmo")
    Goto FireLeftComplete
  FireLeftNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireLeftNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftNoMagButAmmo")
    Goto FireLeftComplete
  FireLeftNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDHellboundBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireLeftComplete
	
  FireLeft: // Left Gun
	HPIS BC 1
	TNT1 A 0 A_GiveInventory("FDHellboundPistolState", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolLeftShots", 1)
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftNoMagConsumption")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireLeftNoMagConsumption") // No reload means no mag consumption
	TNT1 A 0 A_FireBullets (2.5, 0, -1, randompick(28,32), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FireLeftComplete:
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", CHAN_WEAPON, 1.0, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-5,4)
	HPIS D 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
  FireLeftQuickComplete:
	HPIS E 1 Bright A_SetPitch (pitch - (1.3  * CallACS("FD_recoil")/65536.0))
	HPIS G 1 A_SetPitch (pitch + (0.35 * CallACS("FD_recoil")/65536.0))
	HPIS F 1 A_SetPitch (pitch + (1.0 * CallACS("FD_recoil")/65536.0))
	HPIS G 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireLeftEndCheck")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireLeftEndCheck")
	TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 1, "FireLeftEnd")
	Goto FireLeftNoContinue
	
  FireLeftEndCheck:
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, 3)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2)
	TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 1, "FireLeftEnd")
	Goto FireLeftNoContinue
	TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, "FireLeftEnd")
	Goto FireLeftNoContinue
  FireLeftEnd:
	HPIS H 1 A_JumpIfInventory ("FDSynthFireFire", 1, "FireRightContinueH")
	HPIS I 1 A_JumpIfInventory ("FDSynthFireFire", 1, "FireRightContinueI")
	HPIS J 1 A_JumpIfInventory ("FDSynthFireFire", 1, "FireRightContinueJ")
	Goto Ready
  FireLeftNoContinue:
	HPIS HIJ 1
	Goto Ready
	
  FireRightContinueH:
	HPI2 K 1
  FireRightContinueI:
	HPI2 L 1
  FireRightContinueJ:
	TNT1 A 0 A_TakeInventory("FDHellboundPistolState", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolRightShots", 1)
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightContinueNoMagConsumption")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireRightContinueNoMagConsumption") // No reload means no mag consumption
	TNT1 A 0 A_FireBullets (2.5, 0, -1, randompick(28,32), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 6, 1.0, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 5, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,4)
	HPI2 M 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
	Goto FireRightQuickComplete
	
  FireRightContinueNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireRightContinueNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(22,24), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireRightContinueNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightContinueNoMagButAmmo")
    Goto FireRightContinueNoMag
  FireRightContinueNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireRightContinueNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightContinueNoMagButAmmo")
    Goto FireRightContinueNoMag
  FireRightContinueNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDHellboundBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireRightContinueNoMag
  FireRightContinueNoMag:
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 6, 1.0, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 5, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,4)
	HPI2 M 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    Goto FireRightQuickComplete
	
  FireRightNoMagConsumption: // These states are for when reloading is disabled
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") == 2, "FireRightNoMagConsumptionBalanced")
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(22,24), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireRightNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightNoMagButAmmo")
    Goto FireRightComplete
  FireRightNoMagConsumptionBalanced: // This attack has reduced damage to compensate for not needing to reload
    TNT1 A 0 A_FireBullets (0.5, 0.5, -1, randompick(11,12), "FDBulletPuff", FBF_NORANDOM)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "FireRightNoMagButAmmo") // If ammo is enabled and reload is disabled, gotta take bullets directly
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightNoMagButAmmo")
    Goto FireRightComplete
  FireRightNoMagButAmmo: // eat pant
    TNT1 A 0 A_TakeInventory ("FDHellboundBullets", 1, TIF_NOTAKEINFINITE)
    Goto FireRightComplete
	
  FireRight: // Right Gun
	HPI2 BC 1
	TNT1 A 0 A_TakeInventory("FDHellboundPistolState", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolRightShots", 1)
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightNoMagConsumption")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireRightNoMagConsumption") // No reload means no mag consumption
	TNT1 A 0 A_FireBullets (2.5, 0, -1, randompick(28,32), "FDBulletPuff", FBF_NORANDOM | FBF_USEAMMO)
  FireRightComplete:
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 6, 1.0, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundpistol/fire", 5, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,5,4)
	HPI2 D 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
  FireRightQuickComplete:
	HPI2 E 1 Bright A_SetPitch (pitch - (1.3  * CallACS("FD_recoil")/65536.0))
	HPI2 G 1 A_SetPitch (pitch + (0.35 * CallACS("FD_recoil")/65536.0))
	HPI2 F 1 A_SetPitch (pitch + (1.0 * CallACS("FD_recoil")/65536.0))
	HPI2 G 1 A_SetPitch (pitch + (0.15 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, "FireRightEndCheck")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolreload") >= 1, "FireRightEndCheck")
	TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 1, "FireRightEnd")
	Goto FireRightNoContinue
	
  FireRightEndCheck:
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 1, 3)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, 2)
	TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 1, "FireRightEnd")
	Goto FireRightNoContinue
	TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, "FireRightEnd")
	Goto FireRightNoContinue
  FireRightEnd:
	HPI2 H 1 A_JumpIfInventory ("FDSynthFireFire", 1, "FireLeftContinueH")
	HPI2 I 1 A_JumpIfInventory ("FDSynthFireFire", 1, "FireLeftContinueI")
	HPI2 J 1 A_JumpIfInventory ("FDSynthFireFire", 1, "FireLeftContinueJ")
	Goto Ready
  FireRightNoContinue:
	HPI2 HIJ 1
	Goto Ready
	
  Altfire:
    TNT1 A 0 A_JumpIfInventory ("FDHellboundPistolAmmo", 0, "ReadyFull")
	TNT1 A 0 A_JumpIf(CallACS("FD_DoubleAmmoCheck") == 0, "Reloadery")
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 0, "Reloadery")
    TNT1 A 0 A_JumpIfInventory ("FDHellboundBullets", 1, "Reloadery")
	Goto DryFire // Normally we'd return to Ready, but we disable altfire if full anyway, so we can go to dryfire without problems
  Reloadery:
	HPIS A 1 A_Lower
	HPIS A 1 A_Lower
	TNT1 A 0 A_Lower
	HPIS A 1 A_Lower
	TNT1 A 0 A_Lower
	HPIS A 1 A_Lower
	TNT1 AA 0 A_Lower
	HPIS A 1 A_Lower
	TNT1 AA 0 A_Lower
	HPIS A 1 A_Lower
	TNT1 AA 0 A_Lower
	TNT1 A 2 A_WeaponReady (WRF_NOFIRE | WRF_NOBOB)
	TNT1 A 0 A_JumpIf(CallACS("FD_pistolammo") == 1, "ReloadLoopAmmoUseBegin")
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolAmmo", 0, "ReloadFinish")
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolLeftShots", 1, "ReloadLoopLeftBegin")
	Goto ReloadLoopRightBegin
	
  ReloadLoopLeftBegin:
	TNT1 A 8 A_PlaySound("hellboundpistol/open", 5, 0.7, 0, 1.4)
  ReloadLoopLeft:
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolLeftShots", 1, 1)
	Goto ReloadLeftFinish
	TNT1 A 0 A_TakeInventory("FDHellboundPistolLeftShots", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolAmmo", 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, "ReloadLoopLeft")
	TNT1 A 0 A_FireCustomMissile ("FDRevolverCasingSpawner", 0,0,frandom(-8,-7.5),-12, FPF_NOAUTOAIM)
	Goto ReloadLoopLeft
  ReloadLeftFinish:
	TNT1 A 4
	TNT1 A 12 A_PlaySound("hellboundpistol/load", 6, 0.7, 0, 1.4)
	TNT1 A 12 A_PlaySound("hellboundpistol/close", 5, 0.7, 0, 1.4)
	Goto ReloadLoopRightBegin
  ReloadLoopRightBegin:
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolAmmo", 0, "ReloadFinish")
	TNT1 A 8 A_PlaySound("hellboundpistol/open", 5, 0.7, 0, 1.4)
  ReloadLoopRight:
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolRightShots", 1, 1)
	Goto ReloadRightFinish
	TNT1 A 0 A_TakeInventory("FDHellboundPistolRightShots", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolAmmo", 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, "ReloadLoopRight")
	TNT1 A 0 A_FireCustomMissile ("FDRevolverCasingSpawner", 0,0,frandom(8,7.5),-12, FPF_NOAUTOAIM)
	Goto ReloadLoopRight
  ReloadRightFinish:
	TNT1 A 0 A_GiveInventory("FDHellboundPistolAmmo", 12) // just in case it somehow fucks up
	TNT1 A 4
	TNT1 A 12 A_PlaySound("hellboundpistol/load", 6, 0.7, 0, 1.4)
	TNT1 A 12 A_PlaySound("hellboundpistol/close", 5, 0.7, 0, 1.4)
	TNT1 A 2
	Goto ReloadFinish
	
  ReloadLoopAmmoUseBegin:
	TNT1 A 24 A_PlaySound("hellboundpistol/open", 5, 0.7, 0, 1.4)
  ReloadLoopAmmoUse: // Used with the pistol ammo cvar
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolAmmo", 0, "ReloadLoopAmmoUseFinish")
	TNT1 A 0 A_JumpIfInventory("FDHellboundBullets", 1, 1)
	Goto ReloadLoopAmmoUseFinish
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolState", 0, "ReloadLoopAmmoUseState")
	TNT1 A 0 A_GiveInventory("FDHellboundPistolState", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolAmmo", 1)
	TNT1 A 0 A_TakeInventory("FDHellboundBullets", 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, "ReloadLoopAmmoUse")
	TNT1 A 0 A_FireCustomMissile ("FDRevolverCasingSpawner", 0,0,frandom(8,7.5),-12, FPF_NOAUTOAIM)
	Goto ReloadLoopAmmoUse
  ReloadLoopAmmoUseState:
	TNT1 A 0 A_TakeInventory("FDHellboundPistolState", 1)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolAmmo", 1)
	TNT1 A 0 A_TakeInventory("FDHellboundBullets", 1)
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, "ReloadLoopAmmoUse")
	TNT1 A 0 A_FireCustomMissile ("FDRevolverCasingSpawner", 0,0,frandom(-8,-7.5),-12, FPF_NOAUTOAIM)
	Goto ReloadLoopAmmoUse
  ReloadLoopAmmoUseFinish:
	TNT1 A 12
	TNT1 A 24 A_PlaySound("hellboundpistol/load", 6, 0.7, 0, 1.4)
	TNT1 A 28 A_PlaySound("hellboundpistol/close", 5, 0.7, 0, 1.4)
	TNT1 A 2
	Goto ReloadFinish
	
  ReloadFinishState:
	TNT1 A 0 A_JumpIfInventory("FDHellboundPistolState", 1, 2)
	TNT1 A 0 A_GiveInventory("FDHellboundPistolState", 1)
	Goto ReloadFinish
	TNT1 A 0 A_TakeInventory("FDHellboundPistolState", 1)
	Goto ReloadFinish
  ReloadFinish:
	HPIS A 1 A_Raise
	TNT1 AA 0 A_Raise
	HPIS A 1 A_Raise
	TNT1 AA 0 A_Raise
	HPIS A 1 A_Raise
	TNT1 AA 0 A_Raise
	HPIS A 1 A_Raise
	TNT1 A 0 A_Raise
	HPIS A 1 A_Raise
	TNT1 A 0 A_Raise
	HPIS A 1 A_Raise
	Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
