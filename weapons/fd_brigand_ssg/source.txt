 
ACTOR FDHellboundSuperShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDHellboundPlayer"
  Inventory.Icon "HBSUPERS"
  Weapon.UpSound ""
  Weapon.AmmoType "FDHellboundShells"
  Weapon.AmmoUse 2
  Weapon.SlotNumber 3
  Obituary "%k doubled %o's fun by doubling their gun."
  Decal "BulletChip"
  Tag "Brigand SSG"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDHellboundShells", 2, "ReadyAmmo")
    HSSG A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    HSSG A 1 A_WeaponReady
    Loop
	
  Select:
    HSSG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    HSSG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    HSSG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
	TNT1 A 0 A_JumpIfInventory ("FDHellboundShells", 2, 1)
	Goto Dryfire
	TNT1 A 0 A_AlertMonsters
	TNT1 A 0 A_GunFlash
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 3)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,1,6)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-1,6)
	TNT1 A 0 A_PlaySound("hellboundsupershotgun/fire", CHAN_WEAPON, 0.9, 0, 0.7)
	TNT1 A 0 A_PlaySound("hellboundsupershotgun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_FireBullets (11.2, 7.1, 20, 5, "FDHellboundSuperShotgunBulletPuff", FBF_USEAMMO)
	HSSG R 1 Bright A_SetPitch (pitch - (1.6 * CallACS("FD_recoil")/65536.0))
	HSSG R 1 Bright A_SetPitch (pitch - (1.1 * CallACS("FD_recoil")/65536.0))
	HSSG S 1 Bright A_SetPitch (pitch - (0.5 * CallACS("FD_recoil")/65536.0))
	HSSG S 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
	HSSG T 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0)) // 3.5
	HSSG U 1 A_SetPitch (pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	HSSG A 1 A_SetPitch (pitch + (0.75 * CallACS("FD_recoil")/65536.0))
	HSSG A 0 A_JumpIfInventory ("FDHellboundShells", 2, "GonnaReload")
	HSSG A 1 A_SetPitch (pitch + (1.5 * CallACS("FD_recoil")/65536.0))
	HSSG A 1 A_SetPitch (pitch + (1.0 * CallACS("FD_recoil")/65536.0))
	HSSG A 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
	Goto Ready
  GonnaReload:
	HSSG B 1 A_SetPitch (pitch + (1.5 * CallACS("FD_recoil")/65536.0))
	HSSG B 1 A_SetPitch (pitch + (1.0 * CallACS("FD_recoil")/65536.0))
	HSSG C 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
	HSSG C 1
	HSSG D 3
	HSSG E 4
	HSSG F 2
	HSSG G 2
	HSSG H 2
	TNT1 A 0 A_PlaySound("hellboundsupershotgun/open", 5, 0.9, 0, 1.25)
	TNT1 A 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 AA 0 A_FireCustomMissile ("FDShellCasingSpawner2", 0,0,frandom(-3.6,-3.2),-9, FPF_NOAUTOAIM)
	HSSG I 2
	HSSG J 2
	HSSG K 3
	TNT1 A 0 A_PlaySound("hellboundsupershotgun/load", 6, 0.9, 0, 1.25)
	HSSG L 2
	HSSG M 2
	TNT1 A 2
	TNT1 A 1 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_Refire ("QuickFire")
	TNT1 A 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	TNT1 A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_PlaySound("hellboundsupershotgun/close", 5, 0.9, 0, 1.25)
	HSSG N 2
	TNT1 A 0 A_Refire ("Fire")
	HSSG O 2
	HSSG P 2
	HSSG Q 2
	HSSG A 1
	HSSG AAAAA 1 A_Refire
	Goto Ready
  QuickFire:
	TNT1 A 0 A_PlaySound("hellboundsupershotgun/close", 5, 1, 0, 1.25)
	HSSG N 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
	HSSG O 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	HSSG PQ 1
	Goto Fire
	
  Flash:
    TNT1 A 2 A_Light1
    TNT1 A 2 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}
