ACTOR FDBTSXChaingun : FDWeapon
{
  Inventory.RestrictedTo "FDBTSXPlayer"
  Inventory.Icon "BXCHAING"
  Weapon.UpSound ""
  Weapon.AmmoType "FDBTSXBullets"
  Weapon.AmmoUse 1
  Weapon.SlotNumber 4
  Obituary "%o was subjected to intense pulsing by %k."
  Decal ""
  Tag "Pulse Rifle"
  States
  {
  Ready:
    TNT1 A 0 A_GunFlash ("Flash", GFF_NOEXTCHANGE)
  ReadyMain:
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "ReadyRecharge")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "ReadyRecharge")
    XCHG A 1 A_WeaponReady
    Goto ReadyMain
  ReadyRecharge:
    XCH2 A 1 A_WeaponReady
    Goto ReadyMain
	
  Select:
	TNT1 A 0
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "SelectRechargeLoop")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "SelectRechargeLoop")
  SelectLoop:
    XCHG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  SelectRechargeLoop:
    XCH2 A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_GunFlash ("FlashStop", GFF_NOEXTCHANGE)
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "DeselectRechargeLoop")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "DeselectRechargeLoop")
  DeselectLoop:
    XCHG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
  DeselectRechargeLoop:
    XCH2 A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "DryfireRecharge")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "DryfireRecharge")
    XCHG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto ReadyMain
  DryfireRecharge:
    XCH2 A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto ReadyMain
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXBullets", 1, 1)
	Goto Dryfire
  ShotCheck:
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "RechargeShot")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "RechargeShot")
	Goto PowerShot
  NormalShot:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_Light1
    TNT1 A 0 A_PlaySound("btsxchaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("btsxchaingun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFire")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXChaingunProjectile", frandom(-0.2,0.2), 1, 0, 2)
  NormalShotFinish:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "NormalOverdriveFinish")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "NormalOverdriveFinish")
    XCHG B 1 Bright A_SetPitch(pitch - (0.35 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light2
    XCHG C 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light0
    XCHG D 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    XCHG E 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
    XCHG A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	Goto NormalShotRefire
  NormalOverdriveFinish:
    XCHG B 1 Bright A_SetPitch(pitch - (0.35 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light2
    XCHG C 1 Bright A_SetPitch(pitch + (0.20 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light0
    XCHG D 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
    XCHG E 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	Goto NormalShotRefire
	
  NormalShotRefire:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXBullets", 1, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "PumpTime")
	Goto ReadyMain
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "NormalShotRefirePump")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "NormalShotRefireCooldown")
    XCHG AAAAAAAAAAAAAAA 1 A_Refire ("NormalShot")
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, "ReadyMain")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "PumpTime")
    Goto ReadyMain
  NormalShotRefireCooldown:
    XCH2 AAAAAAAAAAAAAAA 1 A_Refire ("RechargeShot")
    Goto ReadyMain
  NormalShotRefirePump:
    XCH2 AAAAAAAAAAAAAAA 1 A_Refire ("RechargeShot")
    Goto PumpTime
  NormalShotRefireAlternate:
    XCHG AAAAAAAAAAAAAAA 1 A_Refire ("RechargeShot")
    Goto ReadyMain
	
  RechargeShot:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_Light1
    TNT1 A 0 A_PlaySound("btsxchaingun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("btsxchaingun/fire", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "RechargeFancyFire")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXChaingunProjectile", frandom(-0.2,0.2), 1, 0, 2)
  RechargeShotFinish:
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePowerAlt", 1, "RechargeOverdriveFinish")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXOverdrivePower", 1, "RechargeOverdriveFinish")
    XCH2 B 1 Bright A_SetPitch(pitch - (0.35 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light2
    XCH2 C 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light0
    XCH2 D 1 A_SetPitch(pitch + (0.15 * CallACS("FD_recoil")/65536.0))
    XCH2 E 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
    XCH2 A 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	Goto NormalShotRefire
  RechargeOverdriveFinish:
    XCH2 B 1 Bright A_SetPitch(pitch - (0.35 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light2
    XCH2 C 1 Bright A_SetPitch(pitch + (0.20 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light0
    XCH2 D 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
    XCH2 E 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	Goto NormalShotRefire
	
  PowerShot:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_Light2
    TNT1 A 0 A_PlaySound("btsxchaingun/firestrong", CHAN_WEAPON, 1, 0, 0.5)
    TNT1 A 0 A_PlaySound("btsxchaingun/firestrong", 7, 0.05, 0, 0.20)
	TNT1 A 0 A_JumpIf(CallACS("FD_fancyprojectiles") == 1, "FancyFirePower")
    TNT1 A 0 A_FireCustomMissile ("FDBTSXChaingunProjectileStrong", 0, 1, 0, 2)
  PowerShotFinish:
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_GiveInventory ("FDBTSXChaingunPump", 1)
    TNT1 A 0 A_GiveInventory ("FDBTSXChaingunCooldown", 1)

	XCHG F 1 Bright A_SetPitch(pitch - (0.5 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light (3)
	XCHG G 1 Bright A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	XCHG G 1 Bright A_SetPitch(pitch + (0.25 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light1
	XCHG H 1 Bright A_SetPitch(pitch + (0.7 * CallACS("FD_recoil")/65536.0))
	XCHG H 1 Bright A_SetPitch(pitch + (0.43 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_Light0
	XCHG I 1 A_SetPitch(pitch + (0.31 * CallACS("FD_recoil")/65536.0))
	XCHG I 1 A_SetPitch(pitch + (0.29 * CallACS("FD_recoil")/65536.0))
	XCHG I 1 A_SetPitch(pitch + (0.17 * CallACS("FD_recoil")/65536.0))
	XCHG I 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
    XCHG J 3
    XCHG K 2
    TNT1 A 0 A_JumpIfInventory ("FDBTSXBullets", 1, 1)
	Goto PumpTime
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump") == 0, 2)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunPump", 1, "PowerShotRefirePump")
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "PowerShotRefireCooldown")
    XCHG AAAAAAAAAAAAAAA 1 A_Refire ("NormalShot")
    Goto PumpTime
  PowerShotRefireCooldown:
    XCH2 AAAAAAAAAAAAAAA 1 A_Refire ("RechargeShot")
    Goto ReadyMain
  PowerShotRefirePump:
    XCH2 AAAAAAAAAAAAAAA 1 A_Refire ("RechargeShot")
    Goto PumpTime
	
  PumpTimeCooldown:
    XCHG LMN 2
    TNT1 A 0 A_GiveInventory ("FDBTSXChaingunCooldown", 1)
	Goto PumpTimeCooldownDone
  PumpTime:
	TNT1 A 0 A_JumpIf(CallACS("FD_btsxpulsepump"), "PumpTimeCooldown")
    XCHG LMN 2
  PumpTimeCooldownDone:
    TNT1 A 0 A_TakeInventory ("FDBTSXChaingunPump", 0)
    TNT1 A 0 A_PlaySound("btsxchaingun/pumpdown", 6, 1, 0, 1.25)
    XCHG O 1 A_SetPitch(pitch + (0.8 * CallACS("FD_recoil")/65536.0))
    XCHG O 1 A_SetPitch(pitch + (0.28 * CallACS("FD_recoil")/65536.0))
    XCHG P 1 A_SetPitch(pitch + (0.1 * CallACS("FD_recoil")/65536.0))
    XCHG P 1 A_SetPitch(pitch + (0.02 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_PlaySound("btsxchaingun/pumpup", 5, 1, 0, 1.25)
    XCHG Q 1 A_SetPitch(pitch - (0.8 * CallACS("FD_recoil")/65536.0))
    XCHG Q 1 A_SetPitch(pitch - (0.28 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_PlaySound("btsxchaingun/beep", 6, 1, 0, 1.25)
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "PumpWhileCooldown")
    XCHG R 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    XCHG R 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
    XCHG ST 2
	Goto ReadyMain
  PumpWhileCooldown:
    XCH2 R 1 A_SetPitch(pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    XCH2 R 1 A_SetPitch(pitch - (0.02 * CallACS("FD_recoil")/65536.0))
    XCH2 ST 2
	Goto ReadyMain
	
  Flash: // Runs alongside normal weapon functionality, beeps when cooldown is finished
    TNT1 A 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "FlashCooldown")
	Loop
  FlashCooldown:
    TNT1 A 1
    TNT1 A 0 A_JumpIfInventory ("FDBTSXChaingunCooldown", 1, "FlashCooldown")
	Goto FlashBeep
  FlashBeep:
    TNT1 A 0 A_PlaySound("btsxchaingun/recharged", 6, 1, 0, 1.25)
    Goto Flash
	
  FlashStop:
    TNT1 A 0
    Goto LightDone
	
  FancyFire:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyChaingunProjectile", frandom(-0.2,0.2), 1, 0, 2)
    Goto NormalShotFinish
  FancyFirePower:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyChaingunProjectileStrong", 0, 1, 0, 2)
    Goto PowerShotFinish
  RechargeFancyFire:
    TNT1 A 0 A_FireCustomMissile ("FDBTSXFancyChaingunProjectile", frandom(-0.2,0.2), 1, 0, 2)
    Goto RechargeShotFinish
  }
}
