 
ACTOR FDAlienVendettaSuperShotgun : FDWeapon
{
  Inventory.RestrictedTo "FDAlienVendettaPlayer"
  Inventory.Icon "AVSUPERS"
  Weapon.UpSound ""
  Weapon.AmmoType "FDAlienVendettaShells"
  Weapon.AmmoUse 2
  Weapon.SlotNumber 3
  Obituary "%o discovered that %k had shattered them."
  Decal "BulletChip"
  Tag "Shattergun"
  States
  {
  Ready:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShells", 2, "ReadyAmmo")
    VSSG A 1 A_WeaponReady (WRF_NOFIRE)
    Loop
  ReadyAmmo:
    VSSG A 1 A_WeaponReady
    Loop
	
  Select:
    VSSG A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    VSSG A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Dryfire:
    VSSG A 6 A_PlaySound("weapons/dryfire1", 5, 1, 0, 1.25)
    Goto Ready
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShells", 2, 1)
	Goto Dryfire
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunShotCounter", 10)
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaShells", 2, TIF_NOTAKEINFINITE)
	
  FireStartLoop:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("alienvendettasupershotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
    TNT1 A 0 A_PlaySound("alienvendettasupershotgun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (7.5, 2.2, 2, 5, "FDBulletPuff", 0)
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunShotCounter", 1)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunState", 1, "FireStartRight")
	Goto FireStartLeft
  FireStartLeft:
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunState", 1)
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,1,6)
    VSSG B 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
	Goto FireLoopCheck
  FireStartRight:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunState", 1)
    TNT1 A 0 A_GunFlash ("Flash2")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-1,6)
    VSSG C 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
	Goto FireLoopCheck
	
  FireLoop:
    TNT1 A 0 A_AlertMonsters
//    TNT1 A 0 A_PlaySound("alienvendettasupershotgun/fire", CHAN_WEAPON, 1, 0, 0.7)
//    TNT1 A 0 A_PlaySound("alienvendettasupershotgun/fire", 7, 0.05, 0, 0.20)
    TNT1 A 0 A_FireBullets (7.5, 2.2, 2, 5, "FDBulletPuff", 0)
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunShotCounter", 1)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunState", 1, "FireLoopRight")
	Goto FireLoopLeft
  FireLoopLeft:
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunState", 1)
    TNT1 A 0 A_GunFlash ("Flash")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,1,6)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunPitchState", 1, 3)
    VSSG D 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
	Goto FireLoopCheck
    VSSG D 1 Bright A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
	Goto FireLoopCheck
  FireLoopRight:
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunState", 1)
    TNT1 A 0 A_GunFlash ("Flash2")
	TNT1 A 0 A_JumpIf(CallACS("FD_gunsmoke") == 0, 2)
	TNT1 A 0 A_FireCustomMissile ("FDGunSmokeSpawner", 0,0,-1,6)
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunPitchState", 1, 3)
    VSSG E 1 Bright A_SetPitch (pitch - (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_GiveInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
	Goto FireLoopCheck
    VSSG E 1 Bright A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
	Goto FireLoopCheck
	
  FireLoopCheck:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunShotCounter", 1, "FireLoop")
	Goto FireEnd
	
  FireEnd:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunPitchState", 1, 2)
	Goto FireEndSide
    TNT1 A 0 A_TakeInventory ("FDAlienVendettaSuperShotgunPitchState", 1)
    TNT1 A 0 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
	Goto FireEndSide
  FireEndSide:
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaSuperShotgunState", 1, "FireEndLeft")
	Goto FireEndRight
  FireEndLeft:
    VSSG G 1 Bright A_SetPitch (pitch - (0.6 * CallACS("FD_recoil")/65536.0))
    VSSG I 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    VSSG A 1 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
    VSSG A 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShells", 2, "GonnaReload")
    Goto Ready
  FireEndRight:
    VSSG F 1 Bright A_SetPitch (pitch - (0.6 * CallACS("FD_recoil")/65536.0))
    VSSG H 1 A_SetPitch (pitch - (0.1 * CallACS("FD_recoil")/65536.0))
    VSSG A 1 A_SetPitch (pitch + (0.5 * CallACS("FD_recoil")/65536.0))
    VSSG A 1 A_SetPitch (pitch + (0.2 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_JumpIfInventory ("FDAlienVendettaShells", 2, "GonnaReload")
    Goto Ready
  GonnaReload:
    VSSG A 3
    TNT1 A 0 A_PlaySound("alienvendettasupershotgun/pumpin", 5, 1, 0, 1.25)
	VSSG J 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	VSSG J 1 A_SetPitch(pitch - (0.08 * CallACS("FD_recoil")/65536.0))
	VSSG J 1 A_SetPitch(pitch - (0.15 * CallACS("FD_recoil")/65536.0))
	VSSG K 1 A_SetPitch(pitch - (0.08 * CallACS("FD_recoil")/65536.0))
	VSSG K 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	TNT1 A 0 A_JumpIf(CallACS("FD_guncasings") == 0, 3)
	TNT1 AA 0 A_FireCustomMissile ("FDTNTShellCasingSpawner3", 0,0,frandom(3.2,3.6),-9, FPF_NOAUTOAIM)
	VSSG K 2
	VSSG L 7
	VSSG M 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0))
	VSSG M 1 A_SetPitch(pitch + (0.03 * CallACS("FD_recoil")/65536.0))
	VSSG M 1 A_SetPitch(pitch + (0.08 * CallACS("FD_recoil")/65536.0))
	VSSG M 1 A_SetPitch(pitch + (0.20 * CallACS("FD_recoil")/65536.0))
	VSSG N 1 A_SetPitch(pitch + (0.35 * CallACS("FD_recoil")/65536.0))
	VSSG N 1 A_SetPitch(pitch + (0.22 * CallACS("FD_recoil")/65536.0))
	VSSG N 1 A_SetPitch(pitch + (0.13 * CallACS("FD_recoil")/65536.0))
    TNT1 A 0 A_PlaySound("alienvendettasupershotgun/pumpout", 5, 1, 0, 1.25)
	VSSG O 1 A_SetPitch(pitch + (0.10 * CallACS("FD_recoil")/65536.0))
	VSSG O 1 A_SetPitch(pitch + (0.05 * CallACS("FD_recoil")/65536.0))
	VSSG O 1 A_SetPitch(pitch + (0.01 * CallACS("FD_recoil")/65536.0)) // 1.18
	VSSG P 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0))
	VSSG P 1 A_SetPitch(pitch - (0.07 * CallACS("FD_recoil")/65536.0))
	VSSG Q 1 A_SetPitch(pitch - (0.17 * CallACS("FD_recoil")/65536.0))
	VSSG Q 1 A_SetPitch(pitch - (0.20 * CallACS("FD_recoil")/65536.0))
	VSSG A 1 A_SetPitch(pitch - (0.16 * CallACS("FD_recoil")/65536.0))
	VSSG A 1 A_SetPitch(pitch - (0.07 * CallACS("FD_recoil")/65536.0))
	VSSG A 1 A_SetPitch(pitch - (0.05 * CallACS("FD_recoil")/65536.0)) // 51
    VSSG AAAAA 1 A_Refire // Gun fires at a speed of 51 tics
    Goto Ready
	
  Flash:
    TNT1 A 1 A_Light1
    TNT1 A 1 A_Light2
    TNT1 A 0 A_Light0
	Goto LightDone
  Flash2:
    TNT1 A 1 A_Light2
    TNT1 A 1 A_Light1
    TNT1 A 0 A_Light0
	Goto LightDone
  }
}

Actor FDAlienVendettaSuperShotgunShotCounter : FDCounter
{
inventory.maxamount 10
}

Actor FDAlienVendettaSuperShotgunState : FDBoolean {}
Actor FDAlienVendettaSuperShotgunPitchState : FDBoolean {}
