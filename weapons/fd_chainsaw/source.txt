 
ACTOR FDDoom2Chainsaw : FDWeapon
{
  Inventory.RestrictedTo "FDDoom2Player"
  Inventory.Icon "D2CHAINS"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Obituary "%k forgot to yell TIMBER before cutting down %o."
  Decal ""
  +WEAPON.MELEEWEAPON
  Tag "Chainsaw"
  States
  {
  Ready:
    TNT1 A 0 A_PlaySound("doom2chainsaw/idle", 7, 1.0, 1, 1.5)
    TNT1 A 0 A_AlertMonsters (random(256,368))
    DSAW CCDD 1 A_WeaponReady
    TNT1 A 0 A_AlertMonsters (random(256,368))
    DSAW EEFF 1 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0 A_PlaySound("doom2chainsaw/draw", 5, 1.0, 0, 1.5)
  SelectLoop:
    DSAW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_PlaySound("plutchainsaw/idlestop", 7, 1.0, 0, 1.5)
    TNT1 A 0 A_StopSound(CHAN_WEAPON)
  DeselectLoop:
    DSAW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_PlaySound("doom2chainsaw/start", 7, 1.0, 0, 0.9)
  Hold:
	TNT1 A 0 A_JumpIf(CallACS("FD_vanillabalance"), "FiringVanilla")
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("doom2chainsaw/fire", CHAN_WEAPON, 0.7, 1, 0.9)
    TNT1 A 0 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    DSAW A 1 A_Saw ("", "", randompick(10,20), "FDChainsawPuff", SF_NORANDOM, 65, 2.8125, 0, 0, 0)
	DSAW A 1 A_SetPitch(pitch - (-0.25 * CallACS("FD_recoil")/65536.0))
	DSAW B 2
    TNT1 A 0 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    DSAW A 1 A_Saw ("", "", randompick(10,20), "FDChainsawPuff", SF_NORANDOM, 65, 2.8125, 0, 0, 0)
	DSAW A 1 A_SetPitch(pitch - (-0.25 * CallACS("FD_recoil")/65536.0))
	DSAW B 2
    TNT1 A 0 A_ReFire
    TNT1 A 0 A_PlaySound("doom2chainsaw/stop", CHAN_WEAPON, 1.0, 0, 0.9)
    Goto Ready
  FiringVanilla:
    TNT1 A 0 A_AlertMonsters
    TNT1 A 0 A_PlaySound("doom2chainsaw/fire", CHAN_WEAPON, 0.7, 1, 0.9)
    TNT1 A 0 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    DSAW A 1 A_Saw ("", "", 2, "FDChainsawPuff", 0, 65, 2.8125, 0, 0, 0)
	DSAW A 1 A_SetPitch(pitch - (-0.25 * CallACS("FD_recoil")/65536.0))
	DSAW B 2
    TNT1 A 0 A_SetPitch(pitch - (0.25 * CallACS("FD_recoil")/65536.0))
    DSAW A 1 A_Saw ("", "", 2, "FDChainsawPuff", 0, 65, 2.8125, 0, 0, 0)
	DSAW A 1 A_SetPitch(pitch - (-0.25 * CallACS("FD_recoil")/65536.0))
	DSAW B 2
    TNT1 A 0 A_ReFire
    TNT1 A 0 A_PlaySound("doom2chainsaw/stop", CHAN_WEAPON, 1.0, 0, 0.9)
    Goto Ready
  }
}
