 
ACTOR FDTNTChainsaw : FDWeapon
{
  Inventory.RestrictedTo "FDTNTPlayer"
  Inventory.Icon "TNCHAINS"
  Weapon.UpSound ""
  Weapon.SlotNumber 1
  Weapon.SlotPriority 0
  Obituary "RIP Ty, you will be missed."
  Decal ""
  Tag "Halderman Device"
  States
  {
  Ready:
	TNT1 A 0 A_JumpIf(CallACS("FD_weaponambience") == 0, 2)
    TNT1 A 0 A_PlaySound("tntchainsaw/idle", 7, 0.8, 1, 1.5)
    TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanType", 2, "ReadyVision")
    TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanType", 1, "ReadyHazard")
  ReadyDefence:
    TSA1 AB 2 A_WeaponReady
    Loop
  ReadyHazard:
    TNT1 A 0 A_JumpIfInventory ("PowerIronFeet", 1, 3)
    TSA2 AB 2 A_WeaponReady
    Loop
    TSA2 AB 2 A_WeaponReady (WRF_NOPRIMARY)
    Loop
  ReadyVision:
    TSA3 AB 2 A_WeaponReady
    Loop
	
  Select:
    TNT1 A 0// A_PlaySound("tntchainsaw/select", 5, 0.8, 0, 1.5)
  SelectLoop:
    TSAW A 1 A_Raise
    "####" "#" 0 A_Raise
    Loop
  Deselect:
    TNT1 A 0 A_Light0
    TNT1 A 0 A_TakeInventory ("FDTNTHaldermanVisionActive", 1)
    TNT1 A 0 A_TakeInventory ("FDTNTHaldermanPlsNoBreakRadsuits", 1)
    TNT1 A 0 A_PlaySound("tntchainsaw/deselect", 7, 0.8, 0, 1.5)
  DeselectLoop:
    TSAW A 1 A_Lower
    "####" "#" 0 A_Lower
    Loop
	
  Fire:
    TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanType", 2, "FireVision")
    TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanType", 1, "FireHazard")
  FireDefence:
    TNT1 A 0 A_PlaySound("tntchainsaw/on", 7, 1, 0, 0.8)
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanDefenceGiver", 1)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanDefenceLight", 1,0,32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
	
	TNT1 A 0 A_JumpIf(CallACS("FD_tnthaldermanicons") == 0, 4)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanDefenceIcon", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanDefenceIcon2", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanDefenceIcon3", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TSA1 C 1 Bright
  FireDefenceLoop:
    TNT1 A 0 A_PlaySound("tntchainsaw/active", CHAN_WEAPON, 1, 1, 0.9)
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanDefenceGiver", 1)
    TSA1 DEFG 1 Bright
    TNT1 A 0 A_Refire ("FireDefenceLoop")
    Goto CooldownDeactivateDefence
	
  FireHazard:
    TNT1 A 0 A_JumpIfInventory ("PowerIronFeet", 1, "EffectAlreadyActive")
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanPlsNoBreakRadsuits", 1)
    TNT1 A 0 A_PlaySound("tntchainsaw/on", 7, 1, 0, 0.8)
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanHazardGiver", 1)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanHazardLight", 1,0,32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
	
	TNT1 A 0 A_JumpIf(CallACS("FD_tnthaldermanicons") == 0, 5)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanHazardIcon", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanHazardIcon2", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanHazardIcon3", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanHazardIcon4", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TSA2 C 1 Bright
  FireHazardLoop:
    TNT1 A 0 A_PlaySound("tntchainsaw/active", CHAN_WEAPON, 1, 1, 0.9)
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanHazardGiver", 1)
    TSA2 DEFG 1 Bright Light("FDHaldermanHazard")
    TNT1 A 0 A_Refire ("FireHazardLoop")
    Goto CooldownDeactivateHazard
	
  FireVision:
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanVisionActive", 1)
    TNT1 A 0 A_PlaySound("tntchainsaw/on",7, 1, 0, 0.8)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanVisionLight", 1,0,32, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
	
	TNT1 A 0 A_JumpIf(CallACS("FD_tnthaldermanicons") == 0, 4)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanLightIcon", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanLightIcon2", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TNT1 A 0 A_SpawnItemEx ("FDTNTHaldermanLightIcon3", 0,0,0, 0,0,0, 0, SXF_NOCHECKPOSITION | SXF_SETMASTER)
    TSA3 C 1 Bright A_Light2
  FireVisionLoop:
    TNT1 A 0 A_PlaySound("tntchainsaw/active", CHAN_WEAPON, 1, 1, 0.9)
    TSA3 DEFG 1 Bright Light("FDHaldermanVision") A_Light (random(8,10))
    TNT1 A 0 A_Refire ("FireVisionLoop")
    Goto CooldownDeactivateVision
	
  CooldownDeactivateDefence:
    TNT1 A 0 A_TakeInventory ("FDTNTHaldermanDefence", 1)
    Goto Cooldown
  CooldownDeactivateHazard:
    TNT1 A 0 A_TakeInventory ("FDTNTHaldermanPlsNoBreakRadsuits", 1)
    TNT1 A 0 A_TakeInventory ("PowerIronFeet", 1)
    Goto Cooldown
  CooldownDeactivateVision:
    TNT1 A 0 A_TakeInventory ("FDTNTHaldermanVisionActive", 1)
    TNT1 A 0 A_Light0
    Goto Cooldown
  Cooldown:
    TNT1 A 0 A_PlaySound("tntchainsaw/off", CHAN_WEAPON, 0.7, 0, 0.9)
    TSAW AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 1 A_WeaponReady (WRF_NOFIRE)
    Goto Ready
	
  EffectAlreadyActive:
    TNT1 A 0 A_Print ("\cgYOU ALREADY HAVE\n\cdHAZARD \cgPROTECTION", 2)
    Goto Cooldown
	
  AltFire:
    TSAW A 2
    TNT1 A 0 A_PlaySound("tntchainsaw/mode", 6, 1, 0, 1.25)
    TSAW A 2
    TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanType", 2, "DefenceMode")
    TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanType", 1, "VisionMode")
	Goto HazardMode
  HazardMode:
    TNT1 A 0 A_Print ("\cdHAZARD MODE", 2)
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanType", 1)
    TSAW A 2
    Goto Ready
  VisionMode:
    TNT1 A 0 A_Print ("\cfVISION MODE", 2)
    TNT1 A 0 A_GiveInventory ("FDTNTHaldermanType", 1)
    TSAW A 2
    Goto Ready
  DefenceMode:
    TNT1 A 0 A_Print ("\ctDEFENCE MODE", 2)
    TNT1 A 0 A_TakeInventory ("FDTNTHaldermanType", 0)
    TSAW A 2
    Goto Ready
  }
}

  Alpha 0.0

  {
  Spawn:
	TNT1 A 0

	TNT1 A 0 A_JumpIfInventory ("FDTNTHaldermanDefence", 1, "LightTime", AAPTR_MASTER)
	Stop
  LightTime:
	TNT1 A 0 A_SetUserVar("user_rotation", user_rotation + 2)
	THDI A 1 Bright Light("FDHaldermanDefenceTiny") A_FadeTo (frandom(0.38,0.42), 0.01, 0)
	TNT1 A 0 A_Jump (256, "SpawnLoop")
	Goto SpawnLoop
  }
}

actor FDTNTHaldermanHazardIcon : FDTNTHaldermanDefenceIcon
{
  var int user_lightpulse;
  States
  {
  Spawn:
	TNT1 A 0
	TNT1 A 0 A_SetUserVar("user_rotation", 0)
	TNT1 A 0 A_Jump (256, "SpawnLoop")
	Goto SpawnLoop
  SpawnLoop:
	TNT1 A 0 A_Warp (AAPTR_MASTER, 26,0,12, user_rotation, WARPF_NOCHECKPOSITION | WARPF_COPYVELOCITY | WARPF_INTERPOLATE, "MaxHax")
	Stop
  MaxHax:
	TNT1 A 0 A_JumpIfInventory ("PowerIronFeet", 1, "LightTime", AAPTR_MASTER)
	Stop
  LightTime:
	TNT1 A 0 A_SetUserVar("user_rotation", user_rotation + 5)
	TNT1 A 0 A_JumpIf (alpha >= 0.95, "GoingDownTime")
	TNT1 A 0 A_JumpIf (alpha <= 0.05, "GoingUpTime")
	TNT1 A 0 A_Jump (256, "LightCheckDone")
	Goto LightCheckDone
  LightCheckDone:
	TNT1 A 0 A_JumpIf (user_lightpulse == 1, "LightUp")
	TNT1 A 0 A_Jump (256, "LightDown")
	Goto LightDown
	
  GoingUpTime:
	TNT1 A 0 A_SetUserVar("user_lightpulse", 1)
	TNT1 A 0 A_Jump (256, "LightCheckDone")
	Goto LightCheckDone
  GoingDownTime:
	TNT1 A 0 A_SetUserVar("user_lightpulse", 0)
	TNT1 A 0 A_Jump (256, "LightCheckDone")
	Goto LightCheckDone
  LightUp:
	THDI B 1 Bright Light("FDHaldermanHazardTiny") A_FadeTo (1.0, 0.02, 0)
	TNT1 A 0 A_Jump (256, "SpawnLoop")
	Goto SpawnLoop
  LightDown:
	THDI B 1 Bright Light("FDHaldermanHazardTiny") A_FadeTo (0.0, 0.02, 0)
	TNT1 A 0 A_Jump (256, "SpawnLoop")
	Goto SpawnLoop
  }
}
